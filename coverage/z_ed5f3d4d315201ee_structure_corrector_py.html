26def apply_chainer(structure: 'Structure', chainer: dict, missing_any_bonds: bool) -> set[int]: 

27 """Set chains according to the 'chainer' input. 

28 Keys are chain names or 'auto' and values are VMD selections. 

29 The 'auto' key sets a new chain per fragment in its selection. 

30 Return the indices of all atoms whose chain was set. 

31 """ 

32 if type(chainer) is not dict: 

33 raise InputError(f'The "chainer" input must be a dict of chain names and selections, not: {chainer}') 

34 # Names explicitly requested by the user must not be used by the 'auto' logic 

35 requested_names = set(key for key in chainer.keys() if key != 'auto') 

36 forced_atom_indices = set() 

37 for key, selection_string in chainer.items(): 

38 if key != 'auto' and not key.isalnum(): 

39 raise InputError(f'Wrong chain "{key}" in the "chainer" input. Keys must be letters and numbers or "auto".') 

40 selection = structure.select(selection_string, syntax='vmd') 

41 if not selection: 

42 raise InputError(f'Empty selection for chain "{key}" in the "chainer" input: {selection_string}') 

43 # Note that chains are set per whole residue so we save all atoms in the selected residues 

44 for residue_index in structure.get_selection_residue_indices(selection): 

45 forced_atom_indices.update(structure.residues[residue_index].atom_indices) 

46 if key != 'auto': 

47 print(f' Setting chain {key} to {len(selection)} atoms ({selection_string})') 

48 structure.set_selection_chain_name(selection, key) 

49 continue 

50 # Set a new chain per fragment 

51 if missing_any_bonds: 

52 raise InputError('Cannot use the "auto" key in the "chainer" input with missing bonds.\n' 

53 ' Please either set chain letters explicitly or provide a topology including bonds.') 

54 for fragment in structure.find_fragments(selection): 

55 current_names = set(chain.name for chain in structure.chains) 

56 unavailable_names = current_names | requested_names 

57 index = 0 

58 while get_chain_name(index) in unavailable_names: index += 1 

59 chain_name = get_chain_name(index) 

60 print(f' Setting chain {chain_name} to a fragment of {len(fragment)} atoms ({selection_string})') 

61 structure.set_selection_chain_name(fragment, chain_name) 

62 return forced_atom_indices

65def get_excluded_atoms_indices( 

66 structure: 'Structure', 

67 pbc_selection: 'Selection', 

68 cg_selection: 'Selection') -> 'Selection': 

69 """Set some atoms which are to be skipped from bonding tests given their "fake" nature.""" 

70 # Get a selection of ion atoms which are not in PBC 

71 # These ions are usually "tweaked" to be bonded to another atom although there is no real covalent bond 

72 # They are not taken in count when testing coherent bonds or looking for the reference frame 

73 non_pbc_ions_selection = structure.select_ions() - pbc_selection 

74 # We also exclude coarse grain atoms since their bonds will never be found by a distance/radius guess 

75 # Also dummy atoms are excluded since they are not real atoms 

76 excluded_atoms_selection = non_pbc_ions_selection + cg_selection + structure.select_dummy() 

77 return set(excluded_atoms_selection.atom_indices)

80def get_mismatching_bonds( 

81 clashes: list[tuple[int, tuple[int, ...], tuple[int, ...]]], 

82) -> tuple[list[tuple[int, int]], list[tuple[int, int]]]: 

83 """Split atom bond mismatches in bonds which are missing and bonds which should not be there. 

84 Every bond is a sorted pair of atom indices. 

85 """ 

86 missing_bonds, unexpected_bonds = set(), set() 

87 for atom_index, actual_bonds, expected_bonds in clashes: 

88 for bonded_index in set(expected_bonds) - set(actual_bonds): 

89 missing_bonds.add(tuple(sorted((atom_index, bonded_index)))) 

90 for bonded_index in set(actual_bonds) - set(expected_bonds): 

91 unexpected_bonds.add(tuple(sorted((atom_index, bonded_index)))) 

92 return sorted(missing_bonds), sorted(unexpected_bonds)

115def write_stable_bonds_view_script( 

116 script_filename: str, 

117 structure_filename: str, 

118 trajectory_filename: str, 

119 frame: int, 

120 failing_atoms: 'Selection', 

121 missing_bonds: list[tuple[int, int]], 

122 unexpected_bonds: list[tuple[int, int]], 

123): 

124 """Write a VMD script to see the wrong bonds in a frame. 

125 

126 Bonds are guessed by VMD from the coordinates in this frame, as the test does. 

127 The failing atom and the atoms it is or should be bonded to are shown as spheres and their residues in licorice. 

128 Missing bonds are drawn as red dashed lines and bonds which should not be there are highlighted in orange. 

129 """ 

130 commands = [ 

131 f'# Wrong bonds in frame {frame}', 

132 f'mol new {{{abspath(structure_filename)}}} waitfor all', 

133 # Frames from the structure file are removed so frame numbers match the trajectory 

134 'animate delete all', 

135 f'mol addfile {{{abspath(trajectory_filename)}}} type xtc first {frame} last {frame} waitfor all', 

136 'set main [molinfo top]', 

137 # Bonds are guessed again from the coordinates in this frame 

138 'mol bondsrecalc $main', 

139 # Residues of the failing atoms 

140 'mol representation Licorice 0.2', 

141 'mol color Name', 

142 f'mol selection {{same residue as ({failing_atoms.to_vmd()})}}', 

143 'mol material Opaque', 

144 'mol addrep $main', 

145 # The failing atom and the atoms it is or should be bonded to 

146 'mol representation VDW 0.4', 

147 f'mol selection {{{failing_atoms.to_vmd()}}}', 

148 'mol addrep $main', 

149 # Draw a bond between the two atoms in a pair 

150 'proc draw_bond {bond args} {', 

151 ' global main', 

152 ' set atoms [atomselect $main "index $bond"]', 

153 ' lassign [$atoms get {x y z}] start end', 

154 ' $atoms delete', 

155 ' graphics $main {*}[linsert $args 1 $start $end]', 

156 '}', 

157 # Bonds which are missing 

158 'graphics $main color red', 

159 f'foreach bond {to_tcl_bonds(missing_bonds)} {{ draw_bond $bond line width 4 style dashed }}', 

160 # Bonds which should not be there 

161 'graphics $main color orange', 

162 'graphics $main material Transparent', 

163 f'foreach bond {to_tcl_bonds(unexpected_bonds)} {{ draw_bond $bond cylinder radius 0.4 }}', 

164 # Center the view in the failing atoms and zoom in 

165 *get_focus_commands(failing_atoms), 

166 'puts "The failing atom and the atoms it is or should be bonded to are shown as spheres and their residues in licorice"', 

167 'puts "Missing bonds are drawn as red dashed lines and bonds which should not be there are highlighted in orange"', 

168 ] 

169 with open(script_filename, 'w') as script_file: 

170 script_file.write('\n'.join(commands) + '\n')

173def write_coherent_bonds_view_script( 

174 script_filename: str, 

175 structure_filename: str, 

176 incoherent_atoms: 'Selection', 

177 bonds: list[tuple[int, int]], 

178): 

179 """Write a VMD script to see the atoms with a wrong number of bonds. 

180 

181 Note that VMD guesses bonds from distances instead of reading them from the topology. 

182 For this reason bonds in the topology of these atoms are passed and drawn apart. 

183 Atoms with a wrong number of bonds are shown as big spheres and the atoms bonded to them as small spheres. 

184 Their residues are shown in lines, with the bonds guessed by VMD. 

185 """ 

186 bonded_atoms = Selection(sorted(set(index for bond in bonds for index in bond))) 

187 commands = [ 

188 '# Atoms with a wrong number of bonds', 

189 f'mol new {{{abspath(structure_filename)}}} waitfor all', 

190 'set main [molinfo top]', 

191 # Residues of the atoms with wrong bonds, with the bonds guessed by VMD 

192 'mol representation Lines 2', 

193 'mol color Name', 

194 f'mol selection {{same residue as ({incoherent_atoms.to_vmd()})}}', 

195 'mol material Opaque', 

196 'mol addrep $main', 

197 # Atoms with a wrong number of bonds 

198 'mol representation VDW 0.4', 

199 f'mol selection {{{incoherent_atoms.to_vmd()}}}', 

200 'mol addrep $main', 

201 ] 

202 # Atoms bonded to them in the topology 

203 if bonded_atoms: 

204 commands += [ 

205 'mol representation VDW 0.2', 

206 f'mol selection {{{bonded_atoms.to_vmd()}}}', 

207 'mol addrep $main', 

208 ] 

209 commands += [ 

210 # Bonds in the topology 

211 'graphics $main color yellow', 

212 f'foreach bond {to_tcl_bonds(bonds)} {{', 

213 ' set atoms [atomselect $main "index $bond"]', 

214 ' lassign [$atoms get {x y z}] start end', 

215 ' $atoms delete', 

216 ' graphics $main cylinder $start $end radius 0.1', 

217 '}', 

218 # Center the view in the atoms with wrong bonds and zoom in 

219 *get_focus_commands(incoherent_atoms), 

220 'puts "Atoms with a wrong number of bonds are shown as big spheres and the atoms bonded to them as small spheres"', 

221 'puts "Bonds in the topology are drawn in yellow, while lines show the bonds guessed by VMD from distances"', 

222 ] 

223 with open(script_filename, 'w') as script_file: 

224 script_file.write('\n'.join(commands) + '\n')

227def structure_corrector( 

228 # Note that this is an early provisional structure 

229 structure: 'Structure', 

230 input_trajectory_file: Optional['File'], 

231 input_topology_file: Union['File', Exception], 

232 output_structure_file: 'File', 

233 output_trajectory_file: Optional['File'], 

234 # Files which will hold the processed structure and trajectory once the processing is over 

235 # They are used in the VMD scripts to see failed tests, since current files are renamed further 

236 final_structure_file: 'File', 

237 final_trajectory_file: 'File', 

238 MD: 'MD', 

239 # Note that this is an early provisional atom selection 

240 pbc_selection: 'Selection', 

241 cg_selection: 'Selection', 

242 ua_selection: 'Selection', 

243 dummy_selection: 'Selection', 

244 snapshots: int, 

245 register: 'Register', 

246 mercy: list[str], 

247 trust: list[str], 

248 guess_bonds: bool, 

249 ignore_bonds: bool, 

250 # Bonds already mined from the topology file, so they are not mined again 

251 topology_bonds: Optional[list[list[int]]], 

252 chainer: Optional[dict] = None, 

253) -> bool: 

254 """Analyze the structure looking for irregularities and then modify the structure to standarize the format. 

255 

256 Supported cases: 

257 

258 * Missing/Non-standard elements -> Atom elements are guessed when missing and standarized (e.g. ZN -> Zn) 

259 * Unstable atom bonds: A bond is formed / broken because its 2 atoms are too close / far in the structure 

260 -> Coordinates in the pdb file are replaced by those of the first frame in the trajectory where bonds are stable 

261 * Incoherent atom bonds -> If an atom has unexpected bonds then the simulation may be wrong 

262 -> Stop the workflow and warn the user 

263 -> Atoms in the united atom selection may have less bonds, since their hydrogens may be absorbed 

264 * Forced chains -> Chains are set according to the 'chainer' input, if passed 

265 -> Missing chains are not fixed and forced chains must not be modified by further corrections 

266 * Missing chains -> Chains are added through VMD 

267 * Splitted chain -> Kill the process and warn the user. This should never happen by just reading a pdb, but after correcting 

268 * Repeated chains -> Chains are renamed (e.g. A, G, B, G, C, G -> A, G, B, H, C, I) 

269 * Splitted residues -> Atoms are sorted together by residue. Trajectory coordinates are also sorted 

270 * Repeated residues -> Residues are renumerated (e.g. 1, 2, 3, 1, 2, 1, 2 -> 1, 2, 3, 4, 5, 6, 7) 

271 * Repeated atoms -> Atoms are renamed with their numeration (e.g. C, C, C, O, O -> C1, C2, C3, O1, O2) 

272 

273 Note that the 'mercy' flag may be passed for critical checkings to not kill the process on fail. 

274 Note that the 'trust' flag may be passed for critical checkings to skip them. 

275 

276 This function also sets some values in the passed MD. 

277 

278 Return True if the chain of any atom was changed. 

279 

280 """ 

281 # Write the initial output structure file which will be overwritten several times further 

282 write_updated_structure(structure, output_structure_file, 'copied') 

283 

284 # ------------------------------------------------------------------------------------------ 

285 # Missing/Non-standard elements ------------------------------------------------------------ 

286 # ------------------------------------------------------------------------------------------ 

287 

288 # It is important to fix elements before trying to fix bonds, since elements have an impact on bonds 

289 # VMD logic to find bonds relies in the atom element to set the covalent bond distance cutoff 

290 if structure.fix_atom_elements(trust=(CORRECT_ELEMENTS in trust)): 

291 # Update the structure file using the corrected structure 

292 write_updated_structure(structure, output_structure_file, 'modified (fix atom elements)') 

293 

294 # ------------------------------------------------------------------------------------------ 

295 # Unstable atom bonds ---------------------------------------------------------------------- 

296 # ------------------------------------------------------------------------------------------ 

297 

298 # Set if stable bonds have to be checked 

299 # Note that we must not skip this even if the test already passed 

300 # It may be a corrected structure the one which passed the structure, while this structure comes from the raw input 

301 must_check_stable_bonds = STABLE_BONDS_FLAG not in trust 

302 

303 # Get safe bonds 

304 # Use topology bonds if possible 

305 # Otherwise guess bonds by guessing bonds according to coordinates and atom radius for 10 frames along the trajectory 

306 safe_bonds = find_safe_bonds( 

307 input_topology_file, 

308 input_trajectory_file, 

309 must_check_stable_bonds, 

310 snapshots, 

311 structure, 

312 register, 

313 cg_selection, 

314 guess_bonds, 

315 ignore_bonds, 

316 topology_bonds=topology_bonds, 

317 ) 

318 

319 # Check if the structure is missing bonds 

320 missing_any_bonds = any(bond == MISSING_BONDS for bond in safe_bonds) 

321 

322 # Failed tests to be reported at the end, once atoms are not to be sorted anymore 

323 # Note that atom indices in warnings and VMD scripts would be wrong otherwise 

324 pending_reports = [] 

325 

326 # If safe bonds do not match structure bonds then we have to fix it 

327 def check_stable_bonds(): 

328 # Save the current structure bonds to further compare with the safe bonds 

329 current_bonds = structure.bonds 

330 # Now force safe bonds in the structure even if they "match" already 

331 # Note that some bonds are excluded from the check and they may be wrong in the structure 

332 # e.g. coarse grain atom bonds 

333 structure.bonds = safe_bonds 

334 # If we have been requested to skip this test then we are done 

335 if not must_check_stable_bonds or missing_any_bonds: return 

336 # Reset warnings related to this analysis 

337 register.remove_warnings(STABLE_BONDS_FLAG) 

338 stable_bonds_view_filename = MD.pathify(STABLE_BONDS_VIEW_FILENAME) 

339 remove_view_script(stable_bonds_view_filename) 

340 # Set some atoms which are to be skipped from these test given their "fake" nature 

341 excluded_atom_indices = get_excluded_atoms_indices(structure, pbc_selection, cg_selection) 

342 # If bonds match from the begining we are done as well 

343 print(f'Checking default structure bonds ({STABLE_BONDS_FLAG})') 

344 if do_bonds_match(current_bonds, safe_bonds, excluded_atom_indices, verbose=True, atoms=structure.atoms): 

345 register.update_test(STABLE_BONDS_FLAG, True) 

346 print(' They are good') 

347 return 

348 # Explain the problem to the user 

349 # Make clear this is not a problem but very normal or you will get mails from users 

350 print(' There may be an atom clash or a bond overstretch in the default structure but this is normal.\n' 

351 ' An atom clash means that two non-bonded atoms are too close.\n' 

352 ' A bond overstretch means that two bonded atoms are too far away.\n' 

353 ' Now we will look for a frame which has no clashes/overstretches at all.\n' 

354 ' This frame will be considered as the "reference frame".') 

355 # Find the first frame in the whole trajectory where safe bonds are respected 

356 bonds_reference_frame, bonds_are_fine, bonds_failure = find_bonds_reference_frame( 

357 structure=structure, 

358 trajectory_file=input_trajectory_file, 

359 snapshots=snapshots, 

360 reference_bonds=safe_bonds, 

361 excluded_atom_indices=excluded_atom_indices, 

362 ) 

363 # Update the task output so it does not have to be repeated further 

364 # IMPORTANT: Note that this is not always run, but only when default structure bonds are wrong 

365 # IMPORTANT: Thus it is NORMAL when the reference frame is calculated further in the workflow 

366 MD.get_reference_frame.prefill(MD, (bonds_reference_frame, bonds_are_fine), { 

367 'structure_file': output_structure_file, 

368 'trajectory_file': input_trajectory_file, 

369 'snapshots': snapshots, 

370 'reference_bonds': safe_bonds, 

371 'structure': structure, 

372 'pbc_selection': pbc_selection 

373 }) 

374 # If there is no reference frame then stop here since there must be a problem 

375 if not bonds_are_fine: 

376 print('There is no reference frame for the safe bonds. Is the trajectory not imaged?') 

377 message = 'Could not find a frame in the trajectory respecting all bonds if bonds were guessed according to atom coordinates and radius.' 

378 # Most frequent wrong bonds, in the first frame where they happen 

379 clash, clash_frames, checked_frames, failing_frame = bonds_failure 

380 missing_bonds, unexpected_bonds = get_mismatching_bonds([clash]) 

381 # Get the distance between the atoms of every wrong bond in this frame 

382 frame_filename = get_pdb_frame(output_structure_file.path, input_trajectory_file.path, failing_frame) 

383 frame_coords = np.array([ atom.coords for atom in Structure.from_pdb_file(frame_filename).atoms ]) 

384 remove(frame_filename) 

385 # Keep atoms instead of indices, since atoms may be sorted further 

386 def to_atom_bonds(bonds: list[tuple[int, int]]) -> list[tuple['Atom', 'Atom', float]]: 

387 return [ (structure.atoms[a], structure.atoms[b], float(np.linalg.norm(frame_coords[a] - frame_coords[b]))) 

388 for a, b in bonds ] 

389 missing_atom_bonds = to_atom_bonds(missing_bonds) 

390 unexpected_atom_bonds = to_atom_bonds(unexpected_bonds) 

391 atom_index, actual_bonds, expected_bonds = clash 

392 failing_atom = structure.atoms[atom_index] 

393 # The failing atom along with the atoms it is bonded to and the atoms it should be bonded to 

394 clash_atoms = [ structure.atoms[index] for index in sorted({ atom_index, *actual_bonds, *expected_bonds }) ] 

395 must_be_killed = STABLE_BONDS_FLAG not in mercy 

396 def report_stable_bonds(): 

397 description = ('The main PDB structure is a default structure and it would be considered to have wrong bonds if they were predicted as previously stated.\n' 

398 f'The most failing atom is {failing_atom.label}, with the same wrong bonds in {clash_frames} of {checked_frames} checked frames.' 

399 f' First in frame {failing_frame}:') 

400 for bonds, header in [(missing_atom_bonds, 'Atoms which should be bonded but are too far:'), 

401 (unexpected_atom_bonds, 'Atoms which should not be bonded but are too close:')]: 

402 if not bonds: continue 

403 description += f'\n {header}' 

404 for atom_a, atom_b, distance in bonds: 

405 description += f'\n {atom_a.label} — {atom_b.label} ({distance:.2f} Å)' 

406 # Atom indices are read now, once atoms are sorted 

407 def to_index_pairs(bonds: list[tuple['Atom', 'Atom', float]]) -> list[tuple[int, int]]: 

408 return [ tuple(sorted((atom_a.index, atom_b.index))) for atom_a, atom_b, _ in bonds ] 

409 missing_pairs, unexpected_pairs = to_index_pairs(missing_atom_bonds), to_index_pairs(unexpected_atom_bonds) 

410 failing_atoms = Selection(sorted(atom.index for atom in clash_atoms)) 

411 # Frame and atoms to be shown in the client 

412 data = { 

413 'frame': failing_frame, 

414 'selection': format_selection(structure, failing_atoms), 

415 } 

416 # Write a VMD script for the user to see the wrong bonds 

417 # This is not part of the warning since it is only useful for the workflow user 

418 # If the workflow is to be killed then current files will remain, otherwise they will be renamed 

419 view_structure_file = output_structure_file if must_be_killed else final_structure_file 

420 view_trajectory_file = input_trajectory_file if must_be_killed else final_trajectory_file 

421 write_stable_bonds_view_script(stable_bonds_view_filename, view_structure_file.path, 

422 view_trajectory_file.path, failing_frame, failing_atoms, missing_pairs, unexpected_pairs) 

423 view_hint = f'To see the wrong bonds run:\n vmd -e {stable_bonds_view_filename}' 

424 if must_be_killed: 

425 raise TestFailure(f'{message}\n{description}\n{view_hint}') 

426 register.add_warning(STABLE_BONDS_FLAG, message, description, data) 

427 print(f' {view_hint}') 

428 if must_be_killed: report_stable_bonds() 

429 register.update_test(STABLE_BONDS_FLAG, False) 

430 pending_reports.append(report_stable_bonds) 

431 return 

432 # Set also the stable bonds frame structure to mine its coordinates 

433 stable_bonds_frame_filename = get_pdb_frame(output_structure_file.path, input_trajectory_file.path, bonds_reference_frame) 

434 stable_bonds_frame_structure = Structure.from_pdb_file(stable_bonds_frame_filename) 

435 # Set all coordinates in the main structure by copying the stable bonds frame coordinates 

436 for atom_1, atom_2 in zip(structure.atoms, stable_bonds_frame_structure.atoms): 

437 atom_1.coords = atom_2.coords 

438 # Remove the stable bonds frame since it is not required anymore 

439 remove(stable_bonds_frame_filename) 

440 write_updated_structure(structure, output_structure_file, 'modified (stable bonds)') 

441 # Set the test as passed 

442 register.update_test(STABLE_BONDS_FLAG, True) 

443 check_stable_bonds() 

444 

445 # Write safe bonds back to the MD 

446 MD.get_reference_bonds.prefill(MD, safe_bonds, { 

447 'topology_file': input_topology_file, 

448 'structure_file': output_structure_file, 

449 'trajectory_file': input_trajectory_file, 

450 'must_check_stable_bonds': must_check_stable_bonds, 

451 'snapshots': snapshots, 

452 'structure': structure, 

453 'register': register, 

454 'cg_selection': cg_selection, 

455 'guess_bonds': guess_bonds, 

456 'ignore_bonds': ignore_bonds, 

457 }) 

458 

459 # ------------------------------------------------------------------------------------------ 

460 # Incoherent residue length ---------------------------------------------------------------- 

461 # ------------------------------------------------------------------------------------------ 

462 

463 # Make sure every known residue has an expected number of atoms 

464 # This number was obtained after checking the number of atoms pero aminoacid in many topologies 

465 # DANI: No es muy fiable de esta forma, pero sirve para ligandos mal puestos 

466 for residue in structure.residues: 

467 if not residue.is_large_aa(): continue 

468 msg = f'Residue {residue.name} in chain {residue.chain.name} has an aminoacid' + \ 

469 f' name but has {residue.atom_count} atoms (residue {residue.index}).' 

470 if LARGE_AMINOACID_FLAG in mercy: 

471 warn(msg) 

472 else: 

473 raise TestFailure(msg + f'\nUse the "-m {LARGE_AMINOACID_FLAG}" flag to ignore this warning and continue anyway.') 

474 

475 # ------------------------------------------------------------------------------------------ 

476 # Incoherent residue bonds ----------------------------------------------------------------- 

477 # ------------------------------------------------------------------------------------------ 

478 

479 # Set a variable to store guessed atoms, in case we need them 

480 guessed_atoms_structure = None 

481 

482 # Make sure there are no disconnected groups of atoms in every residue 

483 for residue in structure.residues: 

484 # If the residue is missing bonds then we can not check if it is coherent 

485 # We assume it is and continue 

486 if residue.is_missing_any_bonds(): continue 

487 # If the residue is coherent then continue 

488 # WARNING: We cannot exclude dummy atoms here, since a dummy atom may connect groups of non-dummy atoms 

489 # e.g. Martini Coarse Grain POPC residues 

490 if residue.is_coherent(): continue 

491 # Otherwise we report the problem 

492 residue_selection = residue.get_selection() - dummy_selection 

493 # WARNING: At this point we must get rid of fragments which are all made of dummy atoms 

494 # These dummy atoms may be dissconnected from the residue entirely 

495 fragments = list(structure.find_fragments(residue_selection, exclude_dummy_fragments=True)) 

496 n_fragments = len(fragments) 

497 if n_fragments == 0: raise RuntimeError('Do we have an empty residue?') 

498 if n_fragments == 1: raise RuntimeError('Test failed but should not') 

499 warn(f'Multiple fragments ({n_fragments}) in residue {residue.index}: {residue}') 

500 for f, fragment in enumerate(fragments, 1): 

501 fragment_atoms = [structure.atoms[index] for index in fragment.atom_indices] 

502 atom_names = [atom.label for atom in fragment_atoms] 

503 print(f' Fragment {f}: {", ".join(atom_names)}') 

504 # DANI: There are two possible scenarios here: 

505 # 1 - Independent atom fragments are in the same residue by mistake 

506 # 2 - Atom bonds are wrong for some atom(s) in this residue 

507 # In the first case we could simply split the residue in multiple residues to fix the problem 

508 # In the second scenario we must surrender 

509 # We check if we are in the first scenario by looking at the distance between atoms fragments 

510 # Guess atom bonds by distance and radii 

511 if not guessed_atoms_structure: 

512 guessed_atoms_structure = structure.copy() 

513 guessed_bonds = get_most_stable_bonds( 

514 structure, 

515 input_trajectory_file.path, 

516 snapshots) 

517 guessed_atoms_structure.bonds = guessed_bonds 

518 # Check if the number of fragments is the same or 1 lower using the guessed bonds 

519 # If the number of guessed fragments matches perfectly the actual number of fragments then we assume it is scenario 1 

520 # In any other scenario we complain 

521 guessed_fragments = list(guessed_atoms_structure.find_fragments(residue_selection, exclude_dummy_fragments=True)) 

522 n_guessed_fragments = len(guessed_fragments) 

523 if n_guessed_fragments != n_fragments: 

524 raise TestFailure(f'Residue {residue.index}: {residue} is not coherent: some atoms are disconnected') 

525 print('Guessed fragments match actual fragments, so we assume bonds are correct.') 

526 print('This means that multiple residues are numbered as the same residue by mistake.') 

527 print('In order to solve this issue, this residue will be splitted in new renumerated residues.') 

528 # New residues keep the same number and get the next free insertion code 

529 # Increasing the number instead could clash with the next residue in the chain 

530 last_icode = residue.icode 

531 for fragment in fragments[1:]: 

532 last_icode = get_next_letter(last_icode) 

533 while last_icode and residue.chain.find_residue(residue.number, last_icode): 

534 last_icode = get_next_letter(last_icode) 

535 if last_icode is None: raise RuntimeError(f'No more letters in the alphabet for icodes in residue {residue}') 

536 # Set the new residue 

537 new_residue = Residue(residue.name, residue.number, last_icode) 

538 structure.set_new_residue(new_residue) 

539 residue.chain.add_residue(new_residue) 

540 # Add fragment atoms to the new residue 

541 fragment_atoms = [structure.atoms[index] for index in fragment.atom_indices] 

542 for atom in fragment_atoms: 

543 new_residue.add_atom(atom) 

544 write_updated_structure(structure, output_structure_file, 'modified (splitted independent residue)') 

545 

546 # ------------------------------------------------------------------------------------------ 

547 # Incoherent atom bonds --------------------------------------------------------------- 

548 # ------------------------------------------------------------------------------------------ 

549 

550 # Set if coherent bonds have to be checked 

551 must_check_coherent_bonds = COHERENT_BONDS_FLAG not in trust 

552 if missing_any_bonds: must_check_coherent_bonds = False 

553 

554 # If this analysis has been already passed then we skip the process 

555 if register.tests.get(COHERENT_BONDS_FLAG, None) is True: 

556 must_check_coherent_bonds = False 

557 

558 # Run the coherent bonds analysis if necessary 

559 if must_check_coherent_bonds: 

560 # Reset warnings related to this analysis 

561 register.remove_warnings(COHERENT_BONDS_FLAG) 

562 coherent_bonds_view_filename = MD.pathify(COHERENT_BONDS_VIEW_FILENAME) 

563 remove_view_script(coherent_bonds_view_filename) 

564 # If the test is not passed then report it 

565 incoherent_atom_indices = structure.check_incoherent_bonds(ua_selection) 

566 if incoherent_atom_indices: 

567 print('FAIL: Uncoherent bonds were found') 

568 message = 'Some atoms may have a higher or lower number of bonds than they should according to their element.' 

569 # Keep atoms instead of indices, since atoms may be sorted further 

570 incoherent_atoms = [ structure.atoms[index] for index in incoherent_atom_indices ] 

571 atom_bonded_atoms = [ (atom, [ structure.atoms[index] for index in atom.get_bonds() ]) for atom in incoherent_atoms ] 

572 # Carbons with a single bond are usually methyl groups whose hydrogens were absorbed in a united atom force field 

573 single_bond_carbons = any(atom.element == 'C' and len(bonded_atoms) == 1 for atom, bonded_atoms in atom_bonded_atoms) 

574 must_be_killed = COHERENT_BONDS_FLAG not in mercy 

575 def report_coherent_bonds(): 

576 description = f'{len(incoherent_atoms)} atoms have a wrong number of bonds:' 

577 MAX_REPORTED_ATOMS = 5 

578 for atom, bonded_atoms in atom_bonded_atoms[:MAX_REPORTED_ATOMS]: 

579 description += f'\n {atom.label} is a {atom.element} atom with {len(bonded_atoms)} bonds' 

580 if bonded_atoms: description += ': ' + ', '.join(bonded_atom.label for bonded_atom in bonded_atoms) 

581 if len(incoherent_atoms) > MAX_REPORTED_ATOMS: 

582 description += f'\n etc. ({len(incoherent_atoms)} atoms)' 

583 if single_bond_carbons: 

584 description += ('\nCarbons with a single bond may be methyl groups in a united atom force field (e.g. GROMOS, Berger lipids).' 

585 f'\nIf so, add them to the united atom selection (e.g. set the "ua_selection" input to "{AUTOMATIC_FLAG}").') 

586 # Atom indices are read now, once atoms are sorted 

587 incoherent_selection = Selection(sorted(atom.index for atom in incoherent_atoms)) 

588 bonds = sorted(set( tuple(sorted((atom.index, bonded_atom.index))) 

589 for atom, bonded_atoms in atom_bonded_atoms for bonded_atom in bonded_atoms )) 

590 # Atoms to be shown in the client 

591 data = { 'selection': format_selection(structure, incoherent_selection) } 

592 # Write a VMD script for the user to see the atoms with wrong bonds 

593 # This is not part of the warning since it is only useful for the workflow user 

594 # If the workflow is to be killed then the current file will remain, otherwise it will be renamed 

595 view_structure_file = output_structure_file if must_be_killed else final_structure_file 

596 write_coherent_bonds_view_script(coherent_bonds_view_filename, view_structure_file.path, 

597 incoherent_selection, bonds) 

598 view_hint = f'To see the atoms with wrong bonds run:\n vmd -e {coherent_bonds_view_filename}' 

599 if must_be_killed: 

600 raise TestFailure(f'{message}\n{description}\n{view_hint}') 

601 register.add_warning(COHERENT_BONDS_FLAG, message, description, data) 

602 print(f' {view_hint}') 

603 if must_be_killed: report_coherent_bonds() 

604 register.update_test(COHERENT_BONDS_FLAG, False) 

605 pending_reports.append(report_coherent_bonds) 

606 # Set the test as succeed if all was good 

607 else: 

608 register.update_test(COHERENT_BONDS_FLAG, True) 

609 

610 # ------------------------------------------------------------------------------------------ 

611 # Forced chains ---------------------------------------------------------------------------- 

612 # ------------------------------------------------------------------------------------------ 

613 

614 # Save the chain of every atom to further know if chains were changed 

615 initial_atom_chains = [ atom.chain.name for atom in structure.atoms ] 

616 

617 # Set chains as requested by the user, if requested 

618 # Save the forced chains so we can make sure they are not modified by further corrections 

619 forced_atom_chains = None 

620 if chainer: 

621 forced_atom_indices = apply_chainer(structure, chainer, missing_any_bonds) 

622 forced_atom_chains = { index: structure.atoms[index].chain.name for index in forced_atom_indices } 

623 write_updated_structure(structure, output_structure_file, 'modified (chainer input)') 

624 

625 # ------------------------------------------------------------------------------------------ 

626 # Missing chains --------------------------------------------------------------------------- 

627 # ------------------------------------------------------------------------------------------ 

628 

629 # Check if chains are missing. If so, create a new chainned structure and set it as the reference 

630 chains = structure.chains 

631 

632 # In case there are not chains at all 

633 # If chains were forced then we respect them, even if there is a single chain 

634 if not chainer and len(chains) == 1 and (chains[0].name == ' ' or chains[0].name == 'X'): 

635 warn('Chains are missing and they will be added') 

636 # Stop here if we have bonds guessed from coarse grain (i.e. we have no topology) 

637 # Note that we rely in fragments (and thus in bonds) to guess chains 

638 if structure.is_missing_any_bonds(): 

639 raise InputError('We cannot guess chains with bonds guessed from coarse grain.\n' 

640 ' Please either provide a topology including bonds or set chains in the structure PDB file.') 

641 # Run the chainer 

642 structure.auto_chainer() 

643 write_updated_structure(structure, output_structure_file, 'modified (no chains)') 

644 

645 else: 

646 # In case there are some missing chains 

647 # Note that atoms with no chain are not a problem for the workflow but they are for the web client 

648 unlettered_chain = next((chain for chain in chains if chain.name == ' '), None) 

649 if unlettered_chain: 

650 new_chain_name = structure.get_available_chain_name() 

651 warn(f'Some chains are missing -> Unchained regions will be chained as {new_chain_name}') 

652 unlettered_chain.name = new_chain_name 

653 write_updated_structure(structure, output_structure_file, 'modified (missing chains)') 

654 

655 # ------------------------------------------------------------------------------------------ 

656 # Splitted chains -------------------------------------------------------------------------- 

657 # ------------------------------------------------------------------------------------------ 

658 

659 # Note that this will never happen because of the pdb itself, duplicated chains are handled further 

660 # This will only happen if chains were missing and guessed 

661 # This may mean there is something wrong in the structure 

662 # Check fragments with the VMD and searh for wrong bonds 

663 if structure.check_splitted_chains(fix_chains=True, display_summary=True): 

664 write_updated_structure(structure, output_structure_file, 'modified (splitted chains)') 

665 

666 # ------------------------------------------------------------------------------------------ 

667 # Repeated chains -------------------------------------------------------------------------- 

668 # ------------------------------------------------------------------------------------------ 

669 

670 if structure.check_repeated_chains(fix_chains=True, display_summary=True): 

671 write_updated_structure(structure, output_structure_file, 'modified (repeated chains)') 

672 

673 # ------------------------------------------------------------------------------------------ 

674 # Coherent chains -------------------------------------------------------------------------- 

675 # ------------------------------------------------------------------------------------------ 

676 

677 # Makes sure polymers with sequence are all bonded 

678 # In other words, make sure there are not multiple proteins/nucleics labeled with the same chain 

679 # Note that their sequences otherwise will be merged silently, thus not reflecting reality 

680 # This would make protein mapping impossible for instance 

681 sequence_polymers_selection = structure.select_protein() + structure.select_nucleic() 

682 

683 # We also want to support having a solution of free aminoacids/nucleotides 

684 # So if residues in the chain are not bonded or bonded in very small fragments we accept it 

685 N_RESIDUES_CUTOFF = 3 

686 

687 # Save if we splitted chains 

688 had_to_split_chains = False 

689 

690 # Iterate sequence polymer chains 

691 for chain in structure.get_selection_chains(sequence_polymers_selection): 

692 # If the chain is missing bonds then we can not check if it is coherent 

693 # We assume it is and continue 

694 if chain.is_missing_any_bonds(): continue 

695 # If there is only one fragment we are good 

696 chain_selection = chain.get_selection() 

697 fragments = list(structure.find_fragments(chain_selection, atom_bonds=safe_bonds)) 

698 if len(fragments) == 0: raise ValueError(f'No fragments found in chain {chain.name}') 

699 if len(fragments) == 1: continue 

700 # We also want to support having a solution of free aminoacids/nucleotides 

701 # So if residues in the chain are not bonded or bonded in very small fragments we accept it 

702 # Make a single fragments out of all small fragments 

703 independent_fragments = [] 

704 merged_fragment = None 

705 for fragment in fragments: 

706 # If the fragments reaches the size cutoff then send it to the independent fragments list 

707 residues = structure.get_selection_residues(fragment) 

708 if len(residues) > N_RESIDUES_CUTOFF: 

709 independent_fragments.append(fragment) 

710 continue 

711 # Otherwise merge it with other small fragments 

712 if merged_fragment: merged_fragment += fragment 

713 else: merged_fragment = fragment 

714 # Add the merged fragment to the list as well 

715 if merged_fragment: 

716 independent_fragments.append(merged_fragment) 

717 # If we have only one fragment after the merging then we are good 

718 if len(independent_fragments) == 1: continue 

719 # Otherwise rename fragments 

720 warn(f'Chain {chain.name} has sequence polymer(s) but not all atoms are bonded.') 

721 # We can skip the first fragment, so it can keep the original chain name 

722 for independent_fragment in independent_fragments[1:]: 

723 print(f' A fragment of this chain including {len(independent_fragment)} atoms will be splitted appart.') 

724 had_to_split_chains = True 

725 # Make a new chain for the current fragment 

726 new_chain_name = structure.get_next_available_chain_name(chain.name) 

727 print(f' This fragment will be renamed as chain "{new_chain_name}"') 

728 structure.set_selection_chain_name(independent_fragment, new_chain_name) 

729 

730 # If we did any change then save the structure 

731 if had_to_split_chains: 

732 write_updated_structure(structure, output_structure_file, 'modified (coherent chains)') 

733 

734 # Make sure forced chains were not modified by any of the previous corrections 

735 # Note that this must be checked before atoms are resorted 

736 if forced_atom_chains: 

737 modified_atom_indices = [ index for index, chain_name in forced_atom_chains.items() 

738 if structure.atoms[index].chain.name != chain_name ] 

739 if len(modified_atom_indices) > 0: 

740 modified_atoms_vmd_selection = Selection(modified_atom_indices).to_vmd() 

741 raise InputError(f'Chains set by the "chainer" input were modified for {len(modified_atom_indices)} atoms' 

742 ' when correcting the structure (see the logs above).\n' 

743 f' Modified atoms: {modified_atoms_vmd_selection}\n' 

744 ' Please make sure every chain is a continuous region and its polymers are bonded.') 

745 

746 # Warn if chains look wrongly assigned (e.g. every lipid has its own chain) 

747 structure.check_suspicious_chains() 

748 

749 # Check if chains were changed 

750 # Note that this must be checked before atoms are resorted 

751 chains_changed = initial_atom_chains != [ atom.chain.name for atom in structure.atoms ] 

752 

753 # ------------------------------------------------------------------------------------------ 

754 # Merged residues ------------------------------------------------------------------------ 

755 # ------------------------------------------------------------------------------------------ 

756 

757 # There may be residues which contain unconnected (unbonded) atoms. They are not allowed. 

758 # They may come from a wrong parsing and be indeed duplicated residues. 

759 # NEVER FORGET: Merged residues may be generated when calling the structure.auto_chainer 

760 # Note that we need bonds to check for merged residues 

761 # If bonds are missing then we skip this check 

762 if not missing_any_bonds and structure.check_merged_residues(fix_residues=True, display_summary=True): 

763 write_updated_structure(structure, output_structure_file, 'modified (merged residues)') 

764 

765 # ------------------------------------------------------------------------------------------ 

766 # Splitted residues ------------------------------------------------------------------------ 

767 # ------------------------------------------------------------------------------------------ 

768 

769 # Note that we need bonds to check for splitted residues 

770 # If bonds are missing then we skip this check 

771 if not missing_any_bonds and structure.check_repeated_residues(fix_residues=True, display_summary=True): 

772 write_updated_structure(structure, output_structure_file, 'modified (repeated residues)') 

773 

774 # Sort trajectory coordinates in case atoms were sorted 

775 if input_trajectory_file.path and structure.trajectory_atom_sorter: 

776 # Save a warning in the register 

777 warn('Atoms have been sorted to solve splitted residues') 

778 print('Creating resorted files for atom bonds and charges') 

779 # Bonds are already resorted 

780 save_json(safe_bonds, MD.project.resorted_bonds_file.path, indent=4) 

781 # Charges are to be resorted 

782 # DANI: Esto no se prueba desde hace tiempo y ha sufrido cambios 

783 # DANI: Debería funcionar pero puede tener algun bug 

784 charges = get_charges(input_topology_file) 

785 resorted_charges = [charges[index] for index in structure.new_atom_order] 

786 MD.get_charges.prefill(MD, resorted_charges, { 

787 'topology_file': input_topology_file 

788 }) 

789 save_json(resorted_charges, MD.project.resorted_charges_file.path, indent=4) 

790 print('Sorting trajectory coordinates to fit the new structure atom sort...') 

791 structure.trajectory_atom_sorter(output_structure_file, input_trajectory_file, output_trajectory_file) 

792 

793 # ------------------------------------------------------------------------------------------ 

794 # Repeated atoms --------------------------------------------------------------------------- 

795 # ------------------------------------------------------------------------------------------ 

796 

797 if structure.check_repeated_atoms(fix_atoms=True, display_summary=True): 

798 write_updated_structure(structure, output_structure_file, 'modified (repeated atoms)') 

799 

800 # ------------------------------------------------------------------------------------------ 

801 # Failed tests ----------------------------------------------------------------------------- 

802 # ------------------------------------------------------------------------------------------ 

803 

804 # Report failed tests with mercy now that atoms are in their final order 

805 for report in pending_reports: report() 

806 

807 return chains_changed
