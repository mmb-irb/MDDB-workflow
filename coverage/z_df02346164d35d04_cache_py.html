12 def __init__(self, cache_file: 'File', project_uuid: str = None, uuid: str = None): 

13 """Initialize the cache. 

14 

15 Args: 

16 cache_file: The cache file. 

17 project_uuid: The uuid of the parent project, for MD caches. 

18 uuid: The uuid to use if the cache has none (e.g. the one already registered in a dataset). 

19 If not passed and the cache has no uuid, a new one is generated. 

20 

21 """ 

22 # Save the previous cache file 

23 self.file = cache_file 

24 # Set an auxiliar file which is used to not overwrite directly the previous cache 

25 self._aux_file = cache_file.get_prefixed_file('.aux') 

26 # Set a dict to store the actual cached data 

27 self.data = {} 

28 # Set a variable to store the hash of the cache file 

29 self.filehash = None 

30 # Load data from the cache file, if it already exists 

31 self.load() 

32 if not self.data.get('uuid'): 

33 # Create a unique id for the Project/MD, unless we already know which one it must be 

34 self.data['uuid'] = uuid if uuid else str(uuid_module.uuid4()) 

35 if project_uuid is not None: 

36 # Add Project uuid for this MD 

37 self.data['project_uuid'] = project_uuid 

38 # Save the entry for the first time 

39 self.save()

41 def refresh(self): 

42 """Check if the cache file has changed right before we modify its content. 

43 If so, then we must read it again before. 

44 This is checked in case the workflow is running several times in parallel. 

45 """ 

46 if self.get_filehash() != self.filehash: self.load()

94def get_cached_function(function: Callable, cache: Cache) -> Callable: 

95 """Create a cached function. 

96 The cached function will store input cksums and output in cache. 

97 The cached function will return cached output when input cksums are identical. 

98 If there is no cached output for the given input cksums then the function will 

99 be executed normally and the output will be stored in cache. 

100 """ 

101 cache_key = function.__name__ 

102 def cached_function(*inputs): 

103 # Get all input cksums 

104 current_result = {} 

105 for i, input in enumerate(inputs, 1): 

106 input_key = f'input_{i}_cksum' 

107 input_cksum = get_cksum_id(input) 

108 current_result[input_key] = input_cksum 

109 # Find if there is a cached result with identical input keys 

110 cached_results = cache.retrieve(cache_key, []) 

111 for cached_result in cached_results: 

112 # Make sure every input cksum matches 

113 missmatch = False 

114 for input_key, new_input_cksum in current_result.items(): 

115 cached_input_cksum = cached_result[input_key] 

116 if cached_input_cksum != new_input_cksum: 

117 missmatch = True 

118 break 

119 if missmatch: continue 

120 # If we make it this far with no missmatch then it means we have a previous result which matches 

121 # Just return the cached output 

122 return cached_result['output'] 

123 # Otherwise we must run the function 

124 current_result['output'] = function(*inputs) 

125 # Save the result in the cache 

126 cached_results.append(current_result) 

127 cache.update(cache_key, cached_results) 

128 # And finally return the output 

129 return current_result['output'] 

130 return cached_function
