306def generate_md_metadata ( 

307 md_inputs : dict, 

308 structure : 'Structure', 

309 snapshots : int, 

310 reference_frame : int, 

311 global_rmsd : Optional[tuple[float, float, float]], 

312 warnings : dict, 

313 output_file : 'File' 

314 ): 

315 """Produce the MD metadata file to be uploaded to the database.""" 

316 # Mine name and directory from MD inputs 

317 name = md_inputs.get('name', None) 

318 directory = md_inputs.get(MD_DIRECTORY, None) 

319 

320 # Write the metadata file 

321 md_metadata = { 

322 'name': name, 

323 'frames': snapshots, 

324 'atoms': len(structure.atoms), # Should be always the same but we better have explicit confirmation 

325 'refframe': reference_frame, 

326 'warnings': warnings, 

327 } 

328 

329 # Add the global RMSD values (against the first frame), if any 

330 # Note that the tuple becomes a list when it is retrieved from the cache 

331 if global_rmsd: 

332 rmsd_mean, rmsd_stdv, rmsd_max = global_rmsd 

333 md_metadata['rmsd_mean'] = rmsd_mean 

334 md_metadata['rmsd_stdv'] = rmsd_stdv 

335 md_metadata['rmsd_max'] = rmsd_max 

336 

337 # Get other MD inputs than the name and the directory 

338 other_md_inputs = { k: v for k, v in md_inputs.items() } 

339 # Remove name from MD inputs to not further overwrite project metadata 

340 if name: 

341 del other_md_inputs['name'] 

342 # Remove the directory name form MD inputs since it is not to be uploaded to the database 

343 if directory: 

344 del other_md_inputs[MD_DIRECTORY] 

345 

346 # Search for other metadata inputs specific for this MD 

347 metadata = {} 

348 for field in METADATA_FIELDS: 

349 input_field = field.lower() 

350 field_value = other_md_inputs.get(input_field, None) 

351 if field_value: 

352 metadata[field] = field_value 

353 

354 # Add the matadata field only if there is at least one value 

355 if len(metadata) > 0: 

356 md_metadata['metadata'] = metadata 

357 

358 # Write metadata to a file 

359 save_json(md_metadata, output_file.path)
