16# All 26 neighbour cell offsets, in units of box vectors 

17NEIGHBOUR_CELL_SHIFTS = np.array(

32def find_frame_escaped_fragments( 

33 reference_center : np.ndarray, 

34 fragment_centers : np.ndarray, 

35 box_vectors : np.ndarray, 

36 margin : float, 

37) -> list[tuple[int, float, float]]: 

38 """Find fragments whose center is considerably closer to a periodic image of the reference than to the reference itself. 

39 

40 The reference is the geometric center of all non-PBC atoms, so it does not depend on where the system is in the box. 

41 Return a list of (fragment index, distance to the reference, distance to the closest reference image). 

42 A fragment is escaped when its distance to a reference image is shorter than the distance to the reference minus the margin. 

43 Only images in the first neighbour cells are checked, but a fragment further away is still closer to one of them. 

44 """ 

45 shifts = NEIGHBOUR_CELL_SHIFTS @ box_vectors 

46 deltas = fragment_centers - reference_center 

47 own_distances = np.linalg.norm(deltas, axis=1) 

48 # The distance to a reference image shifted by s is the norm of delta - s 

49 image_distances = np.linalg.norm(deltas[:, np.newaxis, :] - shifts[np.newaxis, :, :], axis=2).min(axis=1) 

50 escaped_indices = np.where(own_distances - image_distances > margin)[0] 

51 return [ (int(index), own_distances[index], image_distances[index]) for index in escaped_indices ]

54def find_frame_cross_contacts( 

55 positions : np.ndarray, 

56 box_vectors : np.ndarray, 

57 cutoff : float, 

58) -> tuple[np.ndarray, np.ndarray, np.ndarray]: 

59 """Find contacts across periodic boundaries in a frame, for any (orthogonal or triclinic) box. 

60 

61 Return the contact pairs as an (n, 2) array of position indices, sorted by 

62 distance, along with their distances through the periodic boundary and the 

63 cell shifts as an (n, 3) array in units of box vectors. 

64 The second atom of a pair shifted by its cell shift is in contact with the first atom. 

65 

66 Positions are used as they are, without wrapping them into the unit cell, 

67 so the system does not need to be centred in the box and any imaging 

68 representation works (triclinic, compact, rectangular). 

69 Periodic copies of the atoms are generated for the 26 neighbour cells and 

70 only those falling within the cutoff of the atoms bounding box are kept. 

71 A pair is a cross-PBC contact when an atom is within the cutoff of the 

72 periodic copy of another atom but not within the cutoff of the atom itself. 

73 Images beyond the first neighbour cells are not checked. 

74 Molecules are expected to be whole and imaged together at this point. 

75 

76 Made by Claude. 

77 """ 

78 lower_bound = positions.min(axis=0) - cutoff 

79 upper_bound = positions.max(axis=0) + cutoff 

80 copy_positions = [] 

81 copy_sources = [] 

82 copy_cell_shifts = [] 

83 for cell_shift, shift in zip(NEIGHBOUR_CELL_SHIFTS, NEIGHBOUR_CELL_SHIFTS @ box_vectors): 

84 shifted = positions + shift 

85 near_system = np.all((shifted >= lower_bound) & (shifted <= upper_bound), axis=1) 

86 if not near_system.any(): continue 

87 copy_positions.append(shifted[near_system]) 

88 copy_sources.append(np.where(near_system)[0]) 

89 copy_cell_shifts.append(np.tile(cell_shift, (near_system.sum(), 1))) 

90 no_contacts = (np.empty((0, 2), dtype=np.int64), np.empty(0), np.empty((0, 3), dtype=np.int64)) 

91 if not copy_positions: return no_contacts 

92 copy_positions = np.concatenate(copy_positions) 

93 copy_sources = np.concatenate(copy_sources) 

94 copy_cell_shifts = np.concatenate(copy_cell_shifts).astype(np.int64) 

95 # Find real atoms within the cutoff of every periodic copy 

96 neighbour_lists = cKDTree(positions).query_ball_point(copy_positions, cutoff, workers=-1) 

97 counts = [len(neighbours) for neighbours in neighbour_lists] 

98 total = sum(counts) 

99 if total == 0: return no_contacts 

100 pair_copies = np.repeat(np.arange(len(copy_positions)), counts) 

101 pair_sources = copy_sources[pair_copies] 

102 pair_neighbours = np.fromiter(itertools.chain.from_iterable(neighbour_lists), dtype=np.int64, count=total) 

103 # Discard pairs which are also in contact directly 

104 delta = positions[pair_sources] - positions[pair_neighbours] 

105 direct_distances = np.sqrt(np.einsum('ij,ij->i', delta, delta)) 

106 cross = direct_distances >= cutoff 

107 if not cross.any(): return no_contacts 

108 pbc_delta = copy_positions[pair_copies[cross]] - positions[pair_neighbours[cross]] 

109 pbc_distances = np.sqrt(np.einsum('ij,ij->i', pbc_delta, pbc_delta)) 

110 # The copied atom goes second, so the cell shift moves the second atom towards the first one 

111 pairs = np.stack([pair_neighbours[cross], pair_sources[cross]], axis=1) 

112 cell_shifts = copy_cell_shifts[pair_copies[cross]] 

113 # Sort atoms in every pair, so the cell shift is reversed when atoms are swapped 

114 swapped = pairs[:, 0] > pairs[:, 1] 

115 pairs[swapped] = pairs[swapped][:, ::-1] 

116 cell_shifts[swapped] *= -1 

117 # Every pair is found twice (from the copy of each atom), so keep only its shortest distance 

118 by_distance = np.argsort(pbc_distances, kind='stable') 

119 pairs, pbc_distances, cell_shifts = pairs[by_distance], pbc_distances[by_distance], cell_shifts[by_distance] 

120 _, first_index = np.unique(pairs, axis=0, return_index=True) 

121 first_index = np.sort(first_index) 

122 return pairs[first_index], pbc_distances[first_index], cell_shifts[first_index]

130def check_cross_periodic_contacts( 

131 input_structure_filename : str, 

132 input_trajectory_filename : str, 

133 view_script_filename : str, 

134 structure : 'Structure', 

135 pbc_selection : 'Selection', 

136 input_pbc_selection : Optional[str], 

137 update_pbc_selection : Callable, 

138 chains_changed : bool, 

139 mercy : list[str], 

140 trust: list[str], 

141 register : 'Register', 

142 check_selection : str, 

143 distance_cutoff : float, 

144 snapshots : int, 

145 simulation_box : Optional[tuple | str] = None, 

146 ignore_box : bool = False, 

147) -> Optional[bool]: 

148 """Check if non-PBC atoms contact each other across periodic boundaries. 

149 

150 Every frame is checked with find_frame_cross_contacts, which works for any 

151 box shape and does not need the system to be centred in the box. 

152 

153 Also check if any non-PBC fragment goes away from the rest of the system (e.g. an unbound ligand). 

154 If the input PBC selection is automatic then these fragments are added to the PBC selection of this MD 

155 in the inputs file and the workflow must be run again. 

156 

157 distance_cutoff is kept as a fixed parameter (default 5 Å in mwf.py). 

158 No dynamic scaling is needed: the comparison d_PBC < cutoff ≤ d_direct 

159 adapts naturally to the box geometry. 

160 """ 

161 

162 # If it is to be skipped 

163 if CROSS_PBC_FLAG in trust: 

164 return True 

165 

166 # Skip if the box is to be ignored 

167 if ignore_box: 

168 print('Skipping cross-PBC contacts check: simulation box is ignored') 

169 register.remove_warnings(CROSS_PBC_FLAG) 

170 remove_view_script(view_script_filename) 

171 register.update_test(CROSS_PBC_FLAG, 'na') 

172 return True 

173 

174 # If the test was passed already 

175 # Note that a not applicable result is checked again, since it may be applicable now (e.g. box is not ignored anymore) 

176 if register.tests.get(CROSS_PBC_FLAG, None) is True: 

177 return True 

178 

179 register.remove_warnings(CROSS_PBC_FLAG) 

180 remove_view_script(view_script_filename) 

181 

182 # Skip if the system has no box at all for there is no way to run this test without it 

183 if simulation_box is None: 

184 print('Skipping cross-PBC contacts check: missing simulation box') 

185 register.update_test(CROSS_PBC_FLAG, 'na') 

186 return True 

187 

188 full_selection = structure.select(check_selection, syntax='vmd') 

189 non_pbc_selection = full_selection - pbc_selection 

190 

191 if len(non_pbc_selection) < 2: 

192 print('No non-PBC atoms found; skipping cross-PBC contacts check') 

193 register.update_test(CROSS_PBC_FLAG, 'na') 

194 return True 

195 

196 cutoff_nm = distance_cutoff / 10.0 # Å → nm (MDtraj unit) 

197 non_pbc_indices = np.array(non_pbc_selection.atom_indices, dtype=np.int32) 

198 n_nonpbc = len(non_pbc_indices) 

199 

200 print(f'Checking cross-PBC contacts ({n_nonpbc} non-PBC atoms, cutoff {distance_cutoff} Å)') 

201 

202 # Split non-PBC atoms in fragments to check none of them goes away from the rest of the system 

203 # The geometric center of every fragment is compared against the geometric center of all non-PBC atoms 

204 # If bonds are missing then we can not find fragments so we use chains instead 

205 if structure.is_missing_any_bonds(): 

206 fragments = [ chain.get_selection() & non_pbc_selection for chain in structure.chains ] 

207 fragments = [ fragment for fragment in fragments if fragment ] 

208 else: 

209 fragments = list(structure.find_fragments(non_pbc_selection)) 

210 fragments_atom_indices = [ np.array(fragment.atom_indices) for fragment in fragments ] 

211 # Frames where every fragment escaped, and the distances in the first escaped frame 

212 escape_frames = [ [] for _ in fragments ] 

213 first_escape_distances = [ None for _ in fragments ] 

214 

215 violation_frames = [] 

216 # Frame with the largest number of atoms in cross-PBC contact, to be reported as example 

217 example_frame = None 

218 example_atoms_count = 0 

219 example_pairs = example_distances = example_cell_shifts = example_box_vectors = None 

220 # Shortest cross-PBC contact distance along the whole trajectory 

221 closest_distance = np.inf 

222 

223 if not input_trajectory_filename.endswith('.xtc'): 

224 raise ValueError(f'Cross-PBC contacts check is only supported for xtc trajectories: {input_trajectory_filename}') 

225 frames = iterate_xtc_frames(input_trajectory_filename) 

226 pbar = tqdm(frames, total=snapshots, desc=' Frame', unit='frame') 

227 

228 # Iterate frames 

229 for frame_idx, (frame_positions, frame_box) in enumerate(pbar): 

230 # Ignore frames with no unit vectors 

231 # DANI: Si ya hemos comprobado que tiene caja esto puede pasar? 

232 if not frame_box.any(): raise RuntimeError('Missing unitcell vectors') 

233 # Check if there are atoms too close to its period images in this frame 

234 box_vectors = frame_box.astype(np.float64) # (3, 3) nm, rows are vectors 

235 all_positions = frame_positions.astype(np.float64) 

236 positions = all_positions[non_pbc_indices] # (n_nonpbc, 3) nm 

237 # Check if any fragment is closer to a periodic image of the non-PBC center than to the center itself 

238 fragment_centers = np.array([ all_positions[atom_indices].mean(axis=0) for atom_indices in fragments_atom_indices ]) 

239 escaped = find_frame_escaped_fragments(positions.mean(axis=0), fragment_centers, box_vectors, ESCAPE_MARGIN) 

240 for fragment_index, own_distance, image_distance in escaped: 

241 escape_frames[fragment_index].append(frame_idx) 

242 if first_escape_distances[fragment_index] is None: 

243 first_escape_distances[fragment_index] = (own_distance, image_distance) 

244 # Check if there are atoms too close to its period images in this frame 

245 pairs, distances, cell_shifts = find_frame_cross_contacts(positions, box_vectors, cutoff_nm) 

246 if len(pairs) == 0: continue 

247 violation_frames.append(frame_idx) 

248 closest_distance = min(closest_distance, distances[0]) 

249 atoms_count = len(np.unique(pairs)) 

250 if atoms_count > example_atoms_count: 

251 example_frame = frame_idx 

252 example_atoms_count = atoms_count 

253 example_pairs, example_distances, example_cell_shifts = pairs, distances, cell_shifts 

254 example_box_vectors = box_vectors 

255 

256 # Report any problem 

257 escaped_fragment_indices = [ index for index, frames in enumerate(escape_frames) if frames ] 

258 if not violation_frames and not escaped_fragment_indices: 

259 print(' Test passed: no cross-PBC contacts detected') 

260 register.update_test(CROSS_PBC_FLAG, True) 

261 return True 

262 # Each problem has a short title and a description with the details 

263 titles, descriptions = [], [] 

264 # Frames and atoms to be shown in the client 

265 data = {} 

266 # Fragments which went away from the rest of the system 

267 if escaped_fragment_indices: 

268 titles.append('Some non-PBC fragments go away from the rest of the system') 

269 description = '' 

270 for fragment_index in escaped_fragment_indices: 

271 fragment_name = name_fragment(structure, fragments[fragment_index]) 

272 frames = escape_frames[fragment_index] 

273 own_distance, image_distance = first_escape_distances[fragment_index] 

274 description += (f'{fragment_name} in {len(frames)}/{snapshots} frames.' 

275 f' First in frame {frames[0]}, where it is {own_distance * 10:.1f} Å away from the rest of the system' 

276 f' and {image_distance * 10:.1f} Å away from its periodic image\n') 

277 # If the PBC selection is automatic then add the escaped fragments to the PBC selection of this MD 

278 is_automatic, additional_selection_string = parse_automatic_selection(input_pbc_selection) 

279 if is_automatic: 

280 escaped_fragments = [ fragments[fragment_index] for fragment_index in escaped_fragment_indices ] 

281 escaped_selection_string = get_fragments_selection_string(structure, escaped_fragments, chains_changed) 

282 if additional_selection_string: 

283 new_pbc_selection = f'{AUTOMATIC_FLAG} + ({additional_selection_string}) or {escaped_selection_string}' 

284 else: 

285 new_pbc_selection = f'{AUTOMATIC_FLAG} + {escaped_selection_string}' 

286 # If the inputs file is updated then the workflow is run again with the new PBC selection 

287 if update_pbc_selection(new_pbc_selection): 

288 raise InputsUpdated(f'{titles[-1]}\n{description}' 

289 f'These fragments have been added to the PBC selection of this MD in the inputs file: "{new_pbc_selection}"') 

290 # Otherwise the PBC selection may be set from the command line, so the user must change it 

291 raise TestFailure(f'{titles[-1]}\n{description}' 

292 f'These fragments must be part of the PBC selection: "{new_pbc_selection}"') 

293 description += 'If these fragments diffuse freely (e.g. an unbound ligand) then they must be part of the PBC selection' 

294 descriptions.append(description) 

295 # Contacts across periodic boundaries 

296 if violation_frames: 

297 n_violations = len(violation_frames) 

298 titles.append(f'Cross-PBC contacts detected in {n_violations}/{snapshots} frames') 

299 description = ( 

300 'Non-PBC molecules appear to contact each other across periodic boundaries.\n' 

301 f'Frame {example_frame} has the most atoms in contact ({example_atoms_count}).' 

302 f' Closest pairs in this frame:' 

303 ) 

304 MAX_REPORTED_PAIRS = 5 

305 for (atom_a, atom_b), distance in zip(example_pairs[:MAX_REPORTED_PAIRS], example_distances): 

306 label_a = structure.atoms[int(non_pbc_indices[atom_a])].label 

307 label_b = structure.atoms[int(non_pbc_indices[atom_b])].label 

308 description += f'\n {label_a} — {label_b} ({distance * 10:.2f} Å)' 

309 if len(example_pairs) > MAX_REPORTED_PAIRS: 

310 description += f'\n etc. ({len(example_pairs)} pairs)' 

311 # Atoms overlapping through the boundary are not physical, so the box is likely wrong 

312 if closest_distance < OVERLAP_DISTANCE: 

313 description += (f'\nSome atoms overlap their periodic neighbours (down to {closest_distance * 10:.2f} Å).' 

314 ' This is not physical: the simulation box in the trajectory is probably wrong.') 

315 descriptions.append(description) 

316 # Atoms in contact in the example frame 

317 structure_pairs = non_pbc_indices[example_pairs] 

318 contact_selection = Selection([ int(index) for index in np.unique(structure_pairs) ]) 

319 # A pair in contact through a cell shift is also in contact through the opposite shift with its atoms swapped 

320 # Flip shifts so their first non-zero value is positive and opposite shifts are the same 

321 flipped = example_cell_shifts[np.arange(len(example_cell_shifts)), np.argmax(example_cell_shifts != 0, axis=1)] < 0 

322 cell_shifts = example_cell_shifts.copy() 

323 cell_shifts[flipped] *= -1 

324 structure_pairs[flipped] = structure_pairs[flipped][:, ::-1] 

325 # Keep only the pairs in contact through the same cell shift as the closest pair 

326 # Pairs are sorted by distance, so the closest pair is the first one 

327 cell_shift = cell_shifts[0] 

328 shift_pairs = structure_pairs[np.all(cell_shifts == cell_shift, axis=1)] 

329 selection_a = Selection([ int(index) for index in np.unique(shift_pairs[:, 0]) ]) 

330 selection_b = Selection([ int(index) for index in np.unique(shift_pairs[:, 1]) ]) 

331 # Atoms in selection b moved by the shift (in box vectors) are in contact with atoms in selection a 

332 data['frame'] = example_frame 

333 data['selection_a'] = format_selection(structure, selection_a) 

334 data['selection_b'] = format_selection(structure, selection_b) 

335 data['images'] = cell_shift.tolist() 

336 # Write a VMD script for the user to see the contacts 

337 # This is not part of the warning since it is only useful for the workflow user 

338 write_view_script(view_script_filename, input_structure_filename, input_trajectory_filename, 

339 example_frame, example_box_vectors, contact_selection, [ (cell_shift, selection_b) ]) 

340 message = '. '.join(titles) 

341 description = '\n'.join(descriptions) 

342 view_hint = f'To see the atoms in contact run:\n vmd -e {view_script_filename}' if violation_frames else '' 

343 if CROSS_PBC_FLAG in mercy: 

344 register.add_warning(CROSS_PBC_FLAG, message, description, data) 

345 if view_hint: print(f' {view_hint}') 

346 register.update_test(CROSS_PBC_FLAG, False) 

347 return False 

348 raise TestFailure(f'{message}\n{description}\n{view_hint}')

351def write_view_script( 

352 script_filename : str, 

353 structure_filename : str, 

354 trajectory_filename : str, 

355 frame : int, 

356 box_vectors : np.ndarray, 

357 contact_selection : 'Selection', 

358 images : list[tuple[np.ndarray, 'Selection']], 

359): 

360 """Write a VMD script to see the atoms in cross-PBC contact in a frame. 

361 

362 Atoms in contact are shown in CPK and their periodic images are shown in transparent yellow. 

363 The system is loaded again for every image and moved by the cell shift, and the box is drawn with lines. 

364 The whole image is shown in yellow lines. 

365 Note that the VMD periodic display is not used since VMD would misplace images and the box if the box is rotated. 

366 """ 

367 load_commands = [ 

368 f'mol new {{{abspath(structure_filename)}}} waitfor all', 

369 # Frames from the structure file are removed so frame numbers match the trajectory 

370 'animate delete all', 

371 f'mol addfile {{{abspath(trajectory_filename)}}} type xtc first {frame} last {frame} waitfor all', 

372 ] 

373 commands = [ f'# Cross-PBC contacts in frame {frame}', *load_commands, 'set main [molinfo top]' ] 

374 # Box vectors in Angstroms 

375 box_vectors = box_vectors * 10 

376 commands.append('draw color blue') 

377 # Every corner of the box is connected to the three next corners 

378 for shift in itertools.product((0, 1), repeat=3): 

379 corner = np.array(shift) @ box_vectors 

380 for axis in range(3): 

381 if shift[axis] == 1: continue 

382 next_corner = corner + box_vectors[axis] 

383 commands.append(f'draw line {to_tcl_vector(corner)} {to_tcl_vector(next_corner)}') 

384 # Atoms in contact 

385 commands += [ 

386 'mol representation CPK', 

387 'mol color Name', 

388 f'mol selection {{{contact_selection.to_vmd()}}}', 

389 'mol material Opaque', 

390 'mol addrep $main', 

391 ] 

392 # Periodic images of the atoms in contact, with one representation per cell shift 

393 for cell_shift, image_selection in images: 

394 commands += [ 

395 *load_commands, 

396 'mol modcolor 0 top ColorID 4', 

397 'mol representation CPK', 

398 'mol color ColorID 4', 

399 f'mol selection {{{image_selection.to_vmd()}}}', 

400 'mol material Transparent', 

401 'mol addrep top', 

402 f'[atomselect top all] moveby {to_tcl_vector(cell_shift @ box_vectors)}', 

403 ] 

404 # Center the view in the atoms in contact and zoom in 

405 # Note that every molecule has its own center, so all of them are centered together 

406 commands += [ 

407 'mol top $main', 

408 f'set contact [atomselect $main {{{contact_selection.to_vmd()}}}]', 

409 'proc get_size {atoms} {', 

410 ' lassign [measure minmax $atoms] lowest highest', 

411 ' return [tcl::mathfunc::max {*}[vecsub $highest $lowest]]', 

412 '}', 

413 'display resetview', 

414 'set center [list [transoffset [vecscale -1 [measure center $contact]]]]', 

415 'foreach molecule [molinfo list] { molinfo $molecule set center_matrix $center }', 

416 f'set zoom [expr {{[get_size [atomselect $main all]] / max([get_size $contact], {MIN_FOCUS_SIZE})}}]', 

417 'if {$zoom > 1} { scale by $zoom }', 

418 'puts "Atoms in contact are shown in CPK and their periodic images in transparent yellow"', 

419 ] 

420 with open(script_filename, 'w') as script_file: 

421 script_file.write('\n'.join(commands) + '\n')

424def to_tcl_vector(vector : np.ndarray) -> str: 

425 """Format a vector as a Tcl list, rounded to thousandths.""" 

426 return '{' + ' '.join(f'{value:.3f}' for value in vector) + '}'

435def get_fragments_selection_string( 

436 structure : 'Structure', 

437 fragments : list['Selection'], 

438 chains_changed : bool, 

439) -> str: 

440 """Write a VMD selection string for some fragments, as readable as possible. 

441 

442 A fragment which is a whole chain is selected by its chain name, unless chains were changed when correcting the structure. 

443 A fragment which is a single residue with a unique name is selected by its residue name. 

444 Otherwise fragments are selected by their residue indices. 

445 """ 

446 chain_name_counts = Counter(chain.name for chain in structure.chains) 

447 residue_name_counts = Counter(residue.name for residue in structure.residues) 

448 selection_strings = [] 

449 residue_indices = [] 

450 for fragment in fragments: 

451 atom_indices = set(fragment.atom_indices) 

452 # Chains are set in the input structure, while this structure is already corrected 

453 # If chains were changed then chain names may select different atoms in the input structure 

454 if not chains_changed: 

455 chains = structure.get_selection_chains(fragment) 

456 if len(chains) == 1: 

457 chain = chains[0] 

458 if chain_name_counts[chain.name] == 1 and set(chain.get_selection().atom_indices) == atom_indices: 

459 selection_strings.append(f'chain {chain.name}') 

460 continue 

461 fragment_residue_indices = structure.get_selection_residue_indices(fragment) 

462 if len(fragment_residue_indices) == 1: 

463 residue = structure.residues[fragment_residue_indices[0]] 

464 if residue_name_counts[residue.name] == 1 and set(residue.get_selection().atom_indices) == atom_indices: 

465 selection_strings.append(f'resname {residue.name}') 

466 continue 

467 residue_indices += fragment_residue_indices 

468 if residue_indices: 

469 selection_strings.append('residue ' + ranger(sorted(residue_indices), unit_separator=' ', range_separator=' to ')) 

470 selection_string = ' or '.join(selection_strings) 

471 # Make sure the selection string selects exactly the fragment atoms 

472 fragments_atom_indices = set(itertools.chain.from_iterable(fragment.atom_indices for fragment in fragments)) 

473 selected_atom_indices = set(structure.select(selection_string).atom_indices) 

474 if selected_atom_indices != fragments_atom_indices: 

475 raise RuntimeError(f'Selection "{selection_string}" selects {len(selected_atom_indices)} atoms' 

476 f' while fragments have {len(fragments_atom_indices)} atoms') 

477 return selection_string

480def name_fragment(structure : 'Structure', fragment : 'Selection') -> str: 

481 """Name a fragment by its residue if it has only one, or by its chains otherwise.""" 

482 residue_indices = structure.get_selection_residue_indices(fragment) 

483 if len(residue_indices) == 1: return structure.residues[residue_indices[0]].label 

484 return structure.name_selection(fragment)
