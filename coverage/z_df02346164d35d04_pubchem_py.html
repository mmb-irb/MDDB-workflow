17def make_pubchem_request( 

18 url: str, 

19 payload: Optional[dict] = None, 

20 error_context: str = 'PubChem', 

21 no_result_codes: tuple[int, ...] = (), 

22) -> Optional[dict]: 

23 """Make a request to PubChem API, with retries and error handling. 

24 

25 Args: 

26 url (str): The URL to make the request to. 

27 payload (Optional[dict]): The payload to send in a POST request. If None, a GET request will be made. 

28 error_context (str): Context for error messages. 

29 no_result_codes (tuple[int, ...]): Status codes which mean PubChem has no result for this request. 

30 In these cases return None. Any other error status raises an error. 

31 

32 """ 

33 if payload is None: 

34 r = _make_get_request(url) 

35 else: 

36 r = _make_post_request(url, payload) 

37 # Add a small sleep if we are in the green zone to avoid hitting the rate limit 

38 # https://pubchem.ncbi.nlm.nih.gov/docs/dynamic-request-throttling 

39 throttling = r.headers.get('X-Throttling-Control', 'No throttling').split(',')[0] 

40 

41 if 'Yellow' in throttling: 

42 time.sleep(0.2) 

43 elif 'Red' in throttling: 

44 time.sleep(0.5) 

45 elif 'Black' in throttling: 

46 time.sleep(1) 

47 if r.status_code == 200: 

48 return r.json() 

49 if r.status_code in no_result_codes: 

50 return None 

51 if r.status_code == 429: 

52 raise RateLimitError(f"Too many requests in the {error_context} request. Throttling: {throttling}", 

53 retry_after=get_retry_after(r.headers)) 

54 if 'temporarily unable' in r.text: 

55 raise requests.exceptions.ConnectionError("PubChem is temporarily unable to process requests. Please try again later.") 

56 raise RequestError(f"Something went wrong with the {error_context} request (error " + str(r.status_code) + "): " + r.text + "\nThrottling: " + throttling)

60def pubchem_standardization(inchi: str) -> Optional[list[dict]]: 

61 """Try PubChem standardization service to write things like the bonds from aromatic rings in the same order. 

62 You can see examples on the paper https://jcheminf.biomedcentral.com/articles/10.1186/s13321-018-0293-8. 

63 """ 

64 url = "https://pubchem.ncbi.nlm.nih.gov/rest/pug/standardize/inchi/JSON" 

65 # PubChem answers 400 when it is not able to standardize the structure 

66 data = make_pubchem_request(url, payload={'inchi': inchi}, 

67 error_context='PubChem standardization', no_result_codes=(400,)) 

68 if data: 

69 results = [] 

70 for compound in data['PC_Compounds']: 

71 if compound['id']['id']['cid']: 

72 cid = str(compound['id']['id']['cid']) 

73 else: 

74 continue 

75 standard_inchi = None 

76 for prop in compound['props']: 

77 if prop['urn']['label'] == 'InChI': 

78 standard_inchi = prop['value']['sval'] 

79 results.append({'pubchem': cid, 'inchi': standard_inchi}) 

80 return results 

81 return None
