39def add_leading_and_trailing_gaps(alignment: Alignment) -> Alignment: 

40 coords = alignment.coordinates 

41 

42 # We need to add two columns on each side of the coordinates matrix. 

43 # The first column will always point to the start of the sequences, 

44 # hence filled with zeros. 

45 new_coords = np.zeros((2, coords.shape[1] + 4), dtype=int) 

46 

47 # The last w column will always point to the end of sequences, 

48 # hence filled with the length of the respective sequences. 

49 target_len = len(alignment.sequences[0]) 

50 query_len = len(alignment.sequences[1]) 

51 last_col = np.array([target_len, query_len]) 

52 new_coords[:, -1] = last_col 

53 

54 # The middle columns will contain the original coordinates. 

55 new_coords[:, 2:-2] = coords 

56 

57 # The second and one before last columns will point to the end of sequence, 

58 # with less unaligned positions. 

59 new_coords[:, 1] = coords[:, 0] - coords[:, 0].min() 

60 new_coords[:, -2] = coords[:, -1] + (last_col - coords[:, -1]).min() 

61 return Alignment(sequences=alignment.sequences, coordinates=new_coords)

105class ProteinMapper: 

106 """Map structure aminoacid sequences against UniProt reference sequences. 

107 

108 Reference candidates are gathered from several sources, tried in priority order: 

109 forced (inputs file) -> imported (references.json) -> remote project -> PDB ids -> blast. 

110 After each source contributes new candidates we re-run the matching and stop as soon 

111 as every protein chain has a reference. 

112 """

114 def __init__( 

115 self, 

116 structure: 'Structure', 

117 protein_references_file: 'File', 

118 database: 'Database', 

119 remote: 'Remote', 

120 cache: 'Cache', 

121 register: dict, 

122 mercy: list[str] = [], 

123 input_protein_references: list | dict = [], 

124 pdb_ids: list[str] = [], 

125 local_blast: bool = False, 

126 ): 

127 self.structure = structure 

128 self.protein_references_file = protein_references_file 

129 self.database = database 

130 self.remote = remote 

131 self.cache = cache 

132 self.register = register 

133 self.mercy = mercy 

134 self.pdb_ids = pdb_ids 

135 self.local_blast = local_blast

137 # Forced references must be list or dict 

138 # If it is none then we set it as an empty list 

139 if input_protein_references is None: 

140 input_protein_references = [] 

141 # If forced references is a list of dictionaries then it means the input is wrongly formatted 

142 # This may happen since the inputs file is in YAML format, and a simple hyphen makes the difference 

143 # We can fix it from here anyway 

144 if type(input_protein_references) is list and len(input_protein_references) > 0 and type(input_protein_references[0]) is dict: 

145 input_protein_references = {k: v for fr in input_protein_references for k, v in fr.items()}

147 # Check if the forced references are strict (i.e. reference per chain, as a dictionary) or flexible (list of references) 

148 self.strict_references = type(input_protein_references) is dict 

149 # Check the "no referable" flag not to be passed when references are not strict 

150 if not self.strict_references and NO_REFERABLE_FLAG in input_protein_references: 

151 raise InputError(' The "no referable" flag cannot be passed in a list.' 

152 f' You must use a chain keys dictionary (e.g. {"A":"{NO_REFERABLE_FLAG}"})')

154 # Store all the references which are got through this process 

155 # Note that not all references may be used at the end 

156 self.references = {} 

157 # For each input forced reference, get the reference sequence 

158 self.reference_sequences = {}

160 self.imported_references = None 

161 # Structure chains, and the subset which are protein 

162 self.parsed_chains = [] 

163 self.protein_parsed_chains = [] 

164 # Save already tried alignments to not repeat the alignment further 

165 self.tried_alignments = {}

167 # Cache wrappers for getters/logic which connect to the internet (or are otherwise expensive) 

168 self.cached_pdb_to_uniprot = get_cached_function(pdb_to_uniprot, cache) 

169 self.cached_blast = get_cached_function(run_blastp, cache)

185 def add_reference(self, uniprot_id: str, check_isoforms: bool = True): 

186 """Given a uniprot id, add its reference to this list of available references. 

187 Get also the reference of every uniform.

190 reference = self.get_reference(uniprot_id)

198 if not isoforms: return

227 def match_sequences(self) -> bool: 

228 """Try to match all protein sequences with the available reference sequences. 

229 In case of match, objects in the 'protein_parsed_chains' list are modified by adding the result. 

230 Finally, return True if all protein sequences were matched with the available reference sequences or False if not. 

231 """ 

232 self._align_chains_to_references() 

233 self._print_reference_summary() 

234 # Export already matched references 

235 export_references(self.protein_parsed_chains, self.protein_references_file) 

236 # Finally, return True if all protein sequences were matched with the available reference sequences or False if not 

237 allright = all([chain_data['match']['ref'] for chain_data in self.protein_parsed_chains]) 

238 # If we match all chains then make sure there is no forced reference missing which did not match 

239 # Otherwise stop here and force the user to remove these forced uniprot ids from the inputs file 

240 # WARNING: Although we could move on it is better to stop here and warn the user to prevent a future silent problem 

241 if allright and self.input_protein_references: 

242 self._check_forced_references_matched() 

243 return allright

245 def _align_chains_to_references(self): 

246 """For each protein chain, align the available reference sequences and keep the best match.""" 

247 # Track each chain-reference alignment match and keep the score of successful alignments 

248 # Now for each structure sequence, align all reference sequences and keep the best alignment (if it meets the minimum) 

249 for chain_data in self.protein_parsed_chains: 

250 chain = chain_data['name'] 

251 chain_tried_alignments = self.tried_alignments[chain] 

252 # In case references are forced per chain check if there is a reference for this chain and match according to this 

253 if self.strict_references: 

254 # Get the forced specific chain for this sequence, if any 

255 forced_reference = self.input_protein_references.get(chain, None) 

256 if forced_reference: 

257 # If the chain has a specific forced reference then we must align it just once 

258 # Skip this process in further matches 

259 if chain_data['match']['ref']: 

260 continue 

261 # In case the forced reference is the "no referable" flag 

262 # Thus it has no reference sequence and we must not try to match it 

263 # Actually, any match would be accidental and not correct 

264 if forced_reference == NO_REFERABLE_FLAG: 

265 chain_data['match'] = {'ref': NO_REFERABLE_FLAG} 

266 continue 

267 # In case the forced reference is the "not found" flag 

268 # This should not happend but we may be using references as forced references, so just in case 

269 if forced_reference == NOT_FOUND_FLAG: 

270 chain_data['match'] = {'ref': NOT_FOUND_FLAG} 

271 continue 

272 # Get the forced reference sequence and align it to the chain sequence in order to build the map 

273 reference_sequence = self.reference_sequences[forced_reference] 

274 print(f' Aligning chain {chain} with {forced_reference} reference sequence') 

275 align_results = align(reference_sequence, chain_data['sequence']) 

276 # The align must match or we stop here and warn the user 

277 if not align_results: 

278 raise InputError(f'Forced reference {chain} -> {forced_reference} does not match in sequence') 

279 sequence_map, align_score = align_results 

280 reference = self.references[forced_reference] 

281 chain_data['match'] = {'ref': reference, 'map': sequence_map, 'score': align_score} 

282 continue 

283 # If the chain has already a match and this match is among the forced references then stop here 

284 # Forced references have priority and this avoids having a match with a not forced reference further 

285 # Same behaviour if the match is with an unreferable sequence 

286 if chain_data['match']['ref'] and (chain_data['match']['ref'] == NO_REFERABLE_FLAG 

287 or chain_data['match']['ref'] == NOT_FOUND_FLAG 

288 or chain_data['match']['ref']['uniprot'] in self.input_protein_references): 

289 continue 

290 # Iterate over the different available reference sequences 

291 for uniprot_id, reference_sequence in self.reference_sequences.items(): 

292 # If this alignment has been tried already then skip it 

293 if uniprot_id in chain_tried_alignments: 

294 continue 

295 # Align the structure sequence with the reference sequence 

296 # NEVER FORGET: This system relies on the fact that topology chains are not repeated 

297 print(f' Aligning chain {chain} with {uniprot_id} reference sequence') 

298 align_results = align(reference_sequence, chain_data['sequence']) 

299 self.tried_alignments[chain].append(uniprot_id) # Save the alignment try, no matter if it works or not 

300 if not align_results: 

301 continue 

302 # In case we have a valid alignment, check the alignment score is better than the current reference score (if any) 

303 sequence_map, align_score = align_results 

304 current_reference = chain_data['match'] 

305 # IMPORTANT: It is normal having multiple alignments with identical score, due to isoforms 

306 # We want the original reference to prevail if has identical score with isoforms 

307 # The original reference goes always first, so we must stop here if the score is the same 

308 if current_reference['score'] >= align_score: 

309 continue 

310 # If the match is a 'no referable' exception then set a no referable flag 

311 if isinstance(uniprot_id, NoReferableException): 

312 chain_data['match'] = {'ref': NO_REFERABLE_FLAG, 'map': sequence_map, 'score': align_score} 

313 continue 

314 # Proceed to set the corresponding reference otherwise 

315 reference = self.references[uniprot_id] 

316 # If the alignment is better then we impose the new reference 

317 chain_data['match'] = {'ref': reference, 'map': sequence_map, 'score': align_score}

341 def _check_forced_references_matched(self): 

342 """Make sure every forced reference ended up matching some protein sequence. 

343 Otherwise stop here and force the user to remove these forced uniprot ids from the inputs file. 

344 """ 

345 # Get forced uniprot ids 

346 forced_uniprot_ids = set(list(self.input_protein_references.values()) if self.strict_references else self.input_protein_references) 

347 forced_uniprot_ids -= {NOT_FOUND_FLAG, NO_REFERABLE_FLAG} 

348 # forced_uniprot_ids.remove(NO_REFERABLE_FLAG) 

349 # forced_uniprot_ids.remove(NOT_FOUND_FLAG) 

350 # Get matched uniprot ids 

351 matched_references = [chain_data['match']['ref'] for chain_data in self.protein_parsed_chains] 

352 matched_uniprot_ids = set([ref['uniprot'] for ref in matched_references if type(ref) == dict]) 

353 # Isoforms of the forced references are aligned too and may score better than the canonical sequence 

354 # A chain matching an isoform (e.g. P12345-2) is then a match of its entry (P12345) 

355 matched_uniprot_ids |= {uniprot_id.split('-')[0] for uniprot_id in matched_uniprot_ids} 

356 # Check the difference 

357 unmatched_uniprot_ids = forced_uniprot_ids - matched_uniprot_ids 

358 if len(unmatched_uniprot_ids) > 0: 

359 log = ', '.join(unmatched_uniprot_ids) 

360 raise InputError(f'Some forced references were not matched with any protein sequence: {log}\n' 

361 ' Please remove them from the inputs file')

365 def _load_forced_references(self) -> bool: 

366 """Load the references forced through the inputs file. Return True if matching should be attempted.""" 

367 if not self.input_protein_references: 

368 return False 

369 forced_uniprot_ids = list(self.input_protein_references.values()) if self.strict_references else self.input_protein_references 

370 for uniprot_id in forced_uniprot_ids: 

371 # If instead of a uniprot id there is a 'no referable' flag then we skip this process 

372 if uniprot_id == NO_REFERABLE_FLAG: 

373 continue 

374 # If instead of a uniprot id there is a 'not found' flag then we skip this process 

375 # This should not happend but we may be using references as forced references, so just in case 

376 if uniprot_id == NOT_FOUND_FLAG: 

377 continue 

378 # If reference is already in the list (i.e. it has been imported) then skip this process 

379 reference = self.references.get(uniprot_id, None) 

380 if reference: 

381 self.reference_sequences[uniprot_id] = reference['sequence'] 

382 continue 

383 # Find the reference data for the given uniprot id 

384 self.add_reference(uniprot_id) 

385 # Now that we have all forced references data perform the matching 

386 print(' Using forced references from the inputs file') 

387 return True

389 def _load_imported_references(self) -> bool: 

390 """Add references imported from references.json to the matchable pool. 

391 It was not done before since we want forced references to have priority. 

392 Return True if any new reference was added and matching should be attempted. 

393 """ 

394 if not self.imported_references: 

395 return False 

396 need_rematch = False 

397 for uniprot_id, reference in self.imported_references.items(): 

398 # If the imported reference has been aligned already (i.e. it was a forced reference) 

399 if uniprot_id in self.reference_sequences: 

400 continue

403 self.reference_sequences[uniprot_id] = reference['sequence'] 

404 if need_rematch: 

405 print(' Using references imported from references.json') 

406 return need_rematch

408 def _load_remote_references(self) -> bool: 

409 """Import remote references, in case we have a project id. 

410 Return True if any new reference was added and matching should be attempted. 

411 """ 

412 if not self.remote: 

413 return False 

414 need_rematch = False 

415 # Get remote references 

416 remote_references = self.remote.project_data['metadata']['REFERENCES'] 

417 # If there are no referables we will have to make a bit of extra work 

418 if NO_REFERABLE_FLAG in remote_references: 

419 # Get the remote standard topology 

420 topology = self.remote.get_standard_topology() 

421 # Find which chain is the no referable 

422 # Even if multiple chains are no referable, the reference will be only one 

423 topology_references = topology['references'] 

424 no_referable_reference_index = next(( 

425 index for index, reference in enumerate(topology_references) 

426 if reference == NO_REFERABLE_FLAG), None) 

427 if no_referable_reference_index is None: 

428 raise RuntimeError(f'Data inconsistency in project {self.remote.accession}:\n' 

429 ' Metadata references include "noref" while the topology has none.') 

430 # Get all residues which are flagged as no referable 

431 residue_references = topology['residue_reference_indices'] 

432 reference_residue_indices = [ 

433 residue_index for residue_index, reference_index in enumerate(residue_references) 

434 if reference_index == no_referable_reference_index] 

435 # Classify these residues in the different chains they belong to 

436 topology_residue_chains_indices = topology['residue_chain_indices'] 

437 chain_residue_indices = {} 

438 for residue_index in reference_residue_indices: 

439 chain_index = topology_residue_chains_indices[residue_index] 

440 current_chain = chain_residue_indices.get(chain_index, []) 

441 if len(current_chain) == 0: 

442 chain_residue_indices[chain_index] = current_chain 

443 current_chain.append(residue_index) 

444 # Get the amino-acid sequence of every no referable chain and add it to the references 

445 topology_residue_names = topology['residue_names'] 

446 for residue_indices in chain_residue_indices.values(): 

447 residue_names = [topology_residue_names[index] for index in residue_indices] 

448 residue_letters = [protein_residue_name_to_letter(name) for name in residue_names] 

449 chain_sequence = ''.join(residue_letters) 

450 no_referable_exception = NoReferableException(chain_sequence) 

451 self.reference_sequences[no_referable_exception] = chain_sequence 

452 need_rematch = True 

453 # Iterate remote references 

454 for uniprot_id in remote_references: 

455 # If the imported reference has been aligned already (i.e. it was a forced reference) 

456 if uniprot_id in self.reference_sequences: continue 

457 # Skip the no referable flag, which has been handled already 

458 if uniprot_id == NO_REFERABLE_FLAG: continue 

459 need_rematch = True 

460 self.add_reference(uniprot_id) 

461 if need_rematch: 

462 print(f' Using references from remote project {self.remote.accession}') 

463 return need_rematch

465 def _load_pdb_references(self) -> bool: 

466 """Retrieve all uniprot ids associated to the input pdb ids, if any. 

467 Return True if any PDB reference was found and matching should be attempted. 

468 """ 

469 if not self.pdb_ids: 

470 return False 

471 # Track if we added any reference 

472 any_pdb_reference = False 

473 for pdb_id in self.pdb_ids: 

474 # Ask PDB 

475 uniprot_ids = self.cached_pdb_to_uniprot(pdb_id) 

476 if uniprot_ids: any_pdb_reference = True 

477 # Iterate UniProt ids we just found 

478 for uniprot_id in uniprot_ids: 

479 # If this is not an actual UniProt, but a no referable exception, then handle it 

480 # We must find the matching sequence and set the corresponding chain as no referable 

481 if isinstance(uniprot_id, NoReferableException): 

482 self.reference_sequences[uniprot_id] = uniprot_id.sequence 

483 continue 

484 # Build a new reference from the resulting uniprot 

485 self.add_reference(uniprot_id) 

486 # If we found any reference from our search in the PDB then try to match every chain again 

487 pdb_ids_label = ', '.join(self.pdb_ids) 

488 if any_pdb_reference: 

489 print(f' Using references related to PDB ids from the inputs file: {pdb_ids_label}') 

490 else: 

491 print(f' Failed to find any reference related to PDB ids from the inputs file: {pdb_ids_label}') 

492 return any_pdb_reference

494 def _match_via_blast(self) -> bool: 

495 """Last resort: run a blast with each orphan chain sequence, re-matching after each hit. 

496 Return True if all protein chains ended up matched. 

497 """ 

498 # If there are still any chain which is not matched with a reference then we need more references 

499 # To get them, we run a blast with each orphan chain sequence 

500 for chain_data in self.protein_parsed_chains: 

501 # Skip already references chains 

502 if chain_data['match']['ref']: continue 

503 # Get the chain sequence 

504 sequence = chain_data['sequence'] 

505 # Run the blast 

506 uniprot_id = self.cached_blast(sequence, self.local_blast) 

507 if not uniprot_id: 

508 chain_data['match'] = {'ref': NOT_FOUND_FLAG} 

509 continue 

510 # Build a new reference from the resulting uniprot 

511 self.add_reference(uniprot_id) 

512 # If we have every protein chain matched with a reference already then we stop here 

513 print(' Using references from blast') 

514 if self.match_sequences(): 

515 return True 

516 return False 

517 

518 def _handle_unmatched_chains(self): 

519 """At this point we should have matched all sequences. 

520 If not, kill the process unless mercy was given. 

521 """ 

522 unmatched_chains = [chain_data['name'] 

523 for chain_data in self.protein_parsed_chains 

524 if chain_data['match']['ref'] == NOT_FOUND_FLAG or chain_data['match']['ref'] is None] 

525 must_be_killed = REFERENCE_SEQUENCE_FLAG not in self.mercy 

526 if must_be_killed: 

527 raise InputError(f'BLAST failed to find a matching reference sequence for chains: {", ".join(unmatched_chains)}.\n' 

528 ' If your system has antibodies or synthetic constructs please consider marking these chains as "no referable" in the inputs file:\n'+ 

529 f'\t- forced_references:\n\t\t{unmatched_chains[0]}: noref\n' + 

530 ' If your system has exotic proteins whose sequences are not found in the Swiss-Prot database you may force non-curated UniProt ids.\n' + 

531 ' If your system has very exotic proteins whose sequence are not in UniProt you can use the "--mercy refseq" flag to skip this error.') 

532 warn('BLAST failed to find a matching reference sequence for at least one protein sequence') 

533 self.register.add_warning(REFERENCE_SEQUENCE_FLAG, 'There is at least one protein region which is not mapped to any reference sequence') 

534 

535 # -- Orchestration ---------------------------------------------------------------------------- 

536 

537 def run(self) -> dict: 

538 """Run the whole mapping pipeline and return the protein parsed chains.""" 

539 # Remove previous warnings, if any 

540 self.register.remove_warnings(REFERENCE_SEQUENCE_FLAG) 

541 # Import local references, in case the references json file already exists 

542 self._import_local_references() 

543 # Collect protein chains from the structure 

544 self._collect_protein_chains() 

545 # If there are no protein sequences then there is no need to map anything 

546 if len(self.protein_parsed_chains) == 0: 

547 print(' There are no protein sequences') 

548 return self.protein_parsed_chains 

549 # Save already tried alignments to not repeat the alignment further 

550 self.tried_alignments = {chain_data['name']: [] for chain_data in self.protein_parsed_chains} 

551 

552 # Reference sources are tried in priority order. 

553 # After each source adds candidate references we re-run matching and 

554 # stop as soon as every protein chain has a reference. 

555 # Forced references go first so they have priority over the rest. 

556 reference_sources = [ 

557 self._load_forced_references, 

558 self._load_imported_references, 

559 self._load_remote_references, 

560 self._load_pdb_references, 

561 ] 

562 # Iterate until all sequences are matched or all sources have been tried 

563 for load_source in reference_sources: 

564 if load_source() and self.match_sequences(): 

565 return self.protein_parsed_chains 

566 

567 # Blast is the last resort: it is done per orphan chain, re-matching after each hit 

568 if self._match_via_blast(): 

569 return self.protein_parsed_chains 

570 

571 # At this point we should have matched all sequences; deal with those we could not 

572 self._handle_unmatched_chains() 

573 return self.protein_parsed_chains

576def export_references(mapping_data: list, protein_references_file: 'File'): 

577 """Export reference objects data to a json file. 

578 This file is used by the loader to load new references to the database. 

579 Note that all references are saved to this json file, even those which are already in the database. 

580 It is the loader who is the responsible to check which references must be loaded and which ones are loaded already. 

581 Note that mapping data (i.e. which residue belongs to each reference) is not saved. 

582 """ 

583 final_references = [] 

584 final_uniprots = [] 

585 for data in mapping_data: 

586 match = data['match'] 

587 ref = match['ref'] 

588 if not ref or ref == NO_REFERABLE_FLAG or ref == NOT_FOUND_FLAG: 

589 continue 

590 uniprot = ref['uniprot'] 

591 if uniprot in final_uniprots: 

592 continue 

593 final_references.append(ref) 

594 final_uniprots.append(uniprot) 

595 # If there are no references to exported then do not egenrate the json file 

596 if len(final_references) == 0: 

597 return 

598 # Write references to a json file 

599 save_json(final_references, protein_references_file.path, indent=4)

625def align(ref_sequence: str, new_sequence: str, verbose: bool = False) -> Optional[tuple[list, float]]: 

626 """Align two aminoacid sequences. 

627 

628 Return a list with the reference residue indexes (values) 

629 which match each new sequence residues indexes (indexes) 

630 Return also the score of the alignment 

631 Return None when there is not valid alignment at all 

632 Set verbose = True to see a visual summary of the sequence alignments in the logs. 

633 

634 """ 

635 if verbose: print('- REFERENCE\n' + ref_sequence + '\n- NEW\n' + new_sequence) 

636 

637 # If the new sequence is all 'X' stop here, since this would make the alignment infinite 

638 # Then an array filled with None is returned 

639 if all([letter == 'X' for letter in new_sequence]): 

640 return None 

641 

642 # Make sure all characters in both sequences belong to valid aminoacids or are X 

643 # Otherwise stop here because the aligner will complain -> e.g. sequences containing 'U' 

644 ref_seq_non_alphabet_letter = next((letter for letter in ref_sequence if letter not in PROTEIN_RESIDUE_LETTERS), None) 

645 if ref_seq_non_alphabet_letter is not None: 

646 raise ValueError(f'Found non supported letter ({ref_seq_non_alphabet_letter}) in reference sequence: {ref_sequence}') 

647 new_seq_non_alphabet_letter = next((letter for letter in new_sequence if letter not in PROTEIN_RESIDUE_LETTERS), None) 

648 if new_seq_non_alphabet_letter is not None: 

649 raise ValueError(f'Found non supported letter ({new_seq_non_alphabet_letter}) in new sequence: {new_sequence}') 

650 

651 # Return the new sequence as best aligned as possible with the reference sequence 

652 # alignments = pairwise2.align.localds(ref_sequence, new_sequence, MatrixInfo.blosum62, -10, -0.5) 

653 # DANI: Habría que hacerlo de esta otra forma según el deprecation warning (arriba hay más código) 

654 # DANI: El problema es que el output lo tiene todo menos la sequencia en formato alienada 

655 # DANI: i.e. formato '----VNLTT' (con los gaps identificados con guiones), que es justo el que necesito 

656 # Actualized code for biopython >= 1.80 

657 alignments = aligner.align(ref_sequence, new_sequence) 

658 

659 # In case there are no alignments it means the current chain has nothing to do with this reference 

660 # Then an array filled with None is returned 

661 if len(alignments) == 0: 

662 return None 

663 

664 # Several alignments may be returned, specially when it is a difficult or impossible alignment 

665 # Output format example: '----VNLTT' 

666 best_alignment = alignments[0] 

667 best_alignment_with_gaps = add_leading_and_trailing_gaps(best_alignment) 

668 # Extract the allignment sequences 

669 # aligned_ref, aligned_sequence = best_alignment_with_gaps.sequences 

670 # aligned_ref = best_alignment_with_gaps[0] # AGUS: Puede ser útil en el futuro 

671 aligned_sequence = best_alignment_with_gaps[1] 

672 # Obtain the score of the alignment 

673 score = best_alignment.score 

674 # Calculate the normalized score 

675 normalized_score = score / len(new_sequence) 

676 # Make it beautiful for the logs 

677 beautiful_normalized_score = round(normalized_score * 100) / 100 

678 

679 # If the normalized score does not reaches the minimum we consider the alignment is not valid 

680 # It may happen when the reference goes for a specific chain but we must map all chains 

681 # This 1 has been found experimentally 

682 # Non maching sequence may return a 0.1-0.3 normalized score 

683 # Matching sequence may return >4 normalized score 

684 if normalized_score < 2: 

685 if verbose: print(f' Not valid alignment -> Normalized score = {beautiful_normalized_score}') 

686 return None 

687 

688 # Tell the user about the success 

689 if verbose: print(f' Valid alignment -> Normalized score = {beautiful_normalized_score}') 

690 

691 # Match each residue 

692 aligned_mapping = [] 

693 aligned_index = 0 

694 for l, letter in enumerate(aligned_sequence): 

695 # Guions are skipped 

696 if letter == '-': 

697 continue 

698 # Get the current residue of the new sequence 

699 equivalent_letter = new_sequence[aligned_index] 

700 if not letter == equivalent_letter: 

701 raise RuntimeError(f'Something was wrong at position {l} :S') 

702 # 'X' residues cannot be mapped since reference sequences should never have any 'X' 

703 if letter == 'X': 

704 aligned_mapping.append(None) 

705 aligned_index += 1 

706 continue 

707 # Otherwise add the equivalent aligned index to the mapping 

708 # WARNING: Add +1 since uniprot residue counts start at 1, not 0 

709 aligned_mapping.append(l + 1) 

710 aligned_index += 1 

711 

712 return aligned_mapping, normalized_score

715def normalize_protein_sequence(sequence: str) -> str: 

716 """Given a protein sequence, make sure al letters ar standard aminoacids. Replace anything else with X.""" 

717 # WARNING: some PDB polymers/chains may have a "special" sequence 

718 # It may combine one-letter code with 2/3-letter code in parenthesis 

719 # These are special aminoacids or other type of residues such as nucleotides 

720 # e.g. 5JMO, entity 3 -> (DKA)RVK(AR7)(0QE) 

721 # e.g. 6ME2, entity 1 -> ... DRYLYI(YCM)HSLKYD ... 

722 # e.g. nucleic acids -> (DC)(DA)(DA)(DC)(DC)(DG)(DC)(DA)(DA)(DC) 

723 # We simply replace these special residues by X in the sequence 

724 normalized_sequence = re.sub(r'\([0-9A-Z]{2,3}\)', 'X', sequence) 

725 # Also, single letters aimed for nucleic acids with no amino acid equivalent must be removed 

726 # Otheriwse they will make the aligner fail -> e.g. 'U' 

727 # Also you may encounter non-standard letters in UniProt, although this is not usual 

728 # e.g. P02678 -> letter 'B' 

729 sequence_letters = set(normalized_sequence) 

730 for letter in sequence_letters: 

731 if letter not in PROTEIN_RESIDUE_LETTERS: 

732 normalized_sequence = normalized_sequence.replace(letter, 'X') 

733 return normalized_sequence

769def get_uniprot_reference(uniprot_accession: str) -> Optional[dict]: 

770 """Given a uniprot accession, use the uniprot API to request its data and then mine what is needed for the database.""" 

771 # Request Uniprot 

772 request_url = 'https://www.ebi.ac.uk/proteins/api/proteins/' + uniprot_accession 

773 parsed_response = None 

774 request = urllib.request.Request(request_url) 

775 # One day the API was returning a 'Secure Connection Failed' error and this header fixed the problem 

776 request.add_header('Referer', 'https://www.uniprot.org/') 

777 try: 

778 parsed_response = read_uniprot_response(request) 

779 # If the accession is not found in UniProt then the id is not valid 

780 except urllib.error.HTTPError as error: 

781 if error.code == 404: 

782 warn('Cannot find UniProt entry for accession ' + uniprot_accession) 

783 return None 

784 print('Error when requesting ' + request_url) 

785 raise RequestError(f'Something went wrong with the Uniprot request (error {error.code})') 

786 except Exception as error: 

787 print('Error when requesting ' + request_url) 

788 raise RequestError(f'Something went very wrong with the Uniprot request ({str(error)})') 

789 # If we have not a response at this point then it may mean we are trying to access an obsolete entry (e.g. P01607) 

790 if parsed_response == None: 

791 warn('Cannot find UniProt entry for accession ' + uniprot_accession) 

792 return None 

793 # Get the full protein name 

794 protein_data = parsed_response['protein'] 

795 protein_name_data = protein_data.get('recommendedName', None) 

796 # DANI: It is possible that the 'recommendedName' is missing if it is not a reviewed UniProt entry 

797 if not protein_name_data: 

798 warn(f'The UniProt accession {uniprot_accession} is missing the recommended name. You should consider changing the reference.') 

799 protein_name_data = protein_data.get('submittedName', None)[0] 

800 if not protein_name_data: 

801 raise ValueError('Unexpected structure in UniProt response for accession ' + uniprot_accession) 

802 protein_name = protein_name_data['fullName']['value'] 

803 # Get the gene names as a single string 

804 gene_names = [] 

805 # WARNING: Some uniprot entries are missing gene names (e.g. P00718) 

806 # WARNING: Some uniprot entries have names in non-canonical keys (e.g. orfNames, olnNames) 

807 # These names are not shown event in the uniprot web page so we also skip them 

808 genes = parsed_response.get('gene', []) 

809 for gene in genes: 

810 gene_name = gene.get('name', None) 

811 if not gene_name: 

812 continue 

813 gene_names.append(gene_name['value']) 

814 gene_names = ', '.join(gene_names) if len(gene_names) > 0 else None 

815 # Get the organism name 

816 organism = parsed_response['organism']['names'][0]['value'] 

817 # Get the aminoacids sequence 

818 sequence = parsed_response['sequence']['sequence'] 

819 if sequence: 

820 sequence = normalize_protein_sequence(sequence) 

821 # Get interesting regions to be highlighted in the client 

822 domains = [] 

823 # WARNING: Some uniprot entries are missing features (e.g. O27908) 

824 features = parsed_response.get('features', []) 

825 # Get comments data which may be useful to further build domain descriptions 

826 comments_data = parsed_response.get('comments', None) 

827 # There are many types of features but in this case we will focus on domanins and similar 

828 target_types = ['CHAIN', 'REGION', 'DOMAIN', 'MOTIF', 'SITE'] 

829 for feature in features: 

830 # Skip features of other types 

831 if feature['type'] not in target_types: 

832 continue 

833 # Get the domain name 

834 name = feature['description'] 

835 # Build the domain description from the coments data 

836 description = None 

837 if comments_data: 

838 comments = [comment for comment in parsed_response['comments'] if name == comment.get('molecule', None)] 

839 comment_text = [] 

840 for comment in comments: 

841 if not comment.get('text', False): continue 

842 text = comment.get('text', None) 

843 if text is None: raise ValueError('Unexpected UniProt response format: no text in comment') 

844 # DANI: el comment 'text' casi siempre es una lista 

845 # DANI: solo tengo constancia de una vez en que era un string directamente 

846 # DANI: en uno de los comentarios de https://www.ebi.ac.uk/proteins/api/proteins/Q15465 

847 if type(text) is str: comment_text.append(text) 

848 elif type(text) is list: comment_text.append(text[0]['value']) 

849 else: raise ValueError('Unexpected UniProt response format: text in comment is neither str or list') 

850 description = '\n\n'.join(comment_text) 

851 # Set the domain selection 

852 # The domain 'begin' and 'end' values may include non-numeric symbols such as '~', '>' or '<' 

853 # These values are usually ignored or replaced by '?' in the UniProt web client 

854 # There may be not numeric value at all (e.g. Q99497) 

855 # In these cases uniprot shows the domain but it does not allow to use its functionallities 

856 # e.g. you can not blast using the domain sequence 

857 # It makes not sense having a domain with unkown range to me so we skip these domains 

858 begin = feature['begin'].replace('~', '').replace('>', '').replace('<', '') 

859 end = feature['end'].replace('~', '').replace('>', '').replace('<', '') 

860 if begin == '' or end == '': 

861 continue 

862 selection = begin + '-' + end 

863 # If we already have a domain with the same name then join both domains 

864 # For instance, you may have several repetitions of the 'Disordered' region 

865 already_existing_domain = next((domain for domain in domains if domain['name'] == name), None) 

866 if already_existing_domain: 

867 already_existing_domain['selection'] += ', ' + selection 

868 continue 

869 # Otherwise, create a new domain 

870 domain = { 

871 'name': name, 

872 'selection': selection 

873 } 

874 # Add a description only if we succesfully mined it 

875 # Note that only features tagged as CHAIN have comments (not sure about this) 

876 if description: 

877 domain['description'] = description 

878 # Add the new domain to the list 

879 domains.append(domain) 

880 # Mine protein functions from Gene Ontology references 

881 # Get database references. It may no exist in some uniprot entries (e.g. P04058) 

882 db_references = parsed_response.get('dbReferences', []) 

883 # Get references from Gene Ontology (GO) only 

884 go_references = [ref for ref in db_references if ref['type'] == 'GO'] 

885 # A Gene Ontology entry may be one of three types: 

886 # Cellular Component (C), Molecular Function (F) and Biological Process (P) 

887 # In this case we are interested in protein function only so we will keep those with the 'F' header 

888 functions = [ref['properties']['term'][2:] for ref in go_references if ref['properties']['term'][0:2] == 'F:'] 

889 # Get a list of isoforms 

890 isoforms = None 

891 if comments_data: 

892 isoforms_comment = next((comment for comment in comments_data if comment['type'] == 'ALTERNATIVE_PRODUCTS'), None) 

893 if isoforms_comment: 

894 isoforms = [isoform['ids'][0] for isoform in isoforms_comment['isoforms']] 

895 # Set the final reference to be uploaded to the database 

896 return { 

897 'name': protein_name, 

898 'gene': gene_names, 

899 'organism': organism, 

900 'uniprot': uniprot_accession, 

901 'sequence': sequence, 

902 'domains': domains, 

903 'functions': functions, 

904 'isoforms': isoforms, 

905 'version': PROTEIN_REFERENCE_VERSION, 

906 }

909def pdb_to_uniprot(pdb_id: str) -> list[str | NoReferableException]: 

910 """Given a pdb Id, get its uniprot id (e.g. 6VW1 -> Q9BYF1, P0DTC2, P59594).""" 

911 # Set the request query 

912 query = '''query ($id: String!) { 

913 entry(entry_id: $id) { 

914 struct_keywords { 

915 pdbx_keywords 

916 } 

917 polymer_entities { 

918 rcsb_polymer_entity_container_identifiers { uniprot_ids } 

919 rcsb_entity_source_organism { scientific_name } 

920 entity_poly { pdbx_seq_one_letter_code } 

921 } 

922 } 

923 }''' 

924 # Request PDB data 

925 parsed_response = request_pdb_data(pdb_id, query) 

926 # The response may be None 

927 # e.g. an obsolete entry with no replacement 

928 if parsed_response is None: return [] 

929 # Check if it is a "de novo protein" according to the entry keywords 

930 de_novo_protein = False 

931 structure_keywords = parsed_response['struct_keywords'] 

932 if structure_keywords: 

933 pdb_keywords = structure_keywords.get('pdbx_keywords', None) 

934 if pdb_keywords == 'DE NOVO PROTEIN': 

935 de_novo_protein = True 

936 # Mine data 

937 uniprot_ids = [] 

938 # WARNING: Polymers do not come in the same order than in PDB 

939 # WARNING: You will not know the entity number by enumerating them 

940 for polymer in parsed_response['polymer_entities']: 

941 # Get the aminoacids sequence of this polymer (or chain) 

942 entity = polymer.get('entity_poly', None) 

943 sequence = entity.get('pdbx_seq_one_letter_code', None) if entity else None 

944 if sequence: 

945 sequence = normalize_protein_sequence(sequence) 

946 # Get the uniprot ids associated to this polymer (or chain) 

947 identifier = polymer['rcsb_polymer_entity_container_identifiers'] 

948 uniprots = identifier.get('uniprot_ids', None) 

949 # If there are not UniProt ids in this entity then it may be no referable 

950 # If we have a no referable entity then we must return an exception with its sequence 

951 # Beware that nucleic acids also fall in this section 

952 if not uniprots: 

953 # If this polymer, whatever it is, has not sequence then there is nothing we can do 

954 if not sequence: continue 

955 # Nueclic acid sequences should be all X at this point 

956 if all(letter == 'X' for letter in sequence): continue 

957 # Get the organisms 

958 organisms = polymer.get('rcsb_entity_source_organism', None) 

959 # Some synthetic constructs may have not defined organisms at all -> e.g. 3H11, entity 3 

960 if not organisms: 

961 uniprot_ids.append(NoReferableException(sequence)) 

962 continue 

963 # Get scientific names in lower caps since sometimes they are all upper caps 

964 scientific_names = set( 

965 [(organism.get('scientific_name') or '').lower() for organism in organisms]) 

966 # A synthetic construct may have a null scientific name -> e.g. 1LQ7 

967 if len(scientific_names) == 1 and '' in scientific_names: 

968 uniprot_ids.append(NoReferableException(sequence)) 

969 continue 

970 # Some synthetic construct have an object with an empty scientific name -> e.g. 1LQ7 

971 # If we have a synthetic construct then flag the sequence as no referable 

972 if 'synthetic construct' in scientific_names: 

973 uniprot_ids.append(NoReferableException(sequence)) 

974 continue 

975 # However some synthetic constructs may have an organism: the expression organism 

976 # e.g. 1P68 -> Escherichia Coli 

977 # If we reach this point but the whole entry is marked as a de novo protein then 

978 # we make the assumtion that this chain/polymer is synthetic 

979 if de_novo_protein: 

980 uniprot_ids.append(NoReferableException(sequence)) 

981 continue 

982 # Check if the sequence of this chain may belong to an antibody 

983 print(' Could this be an antibody?') 

984 is_antibody = False 

985 for reference_sequence in REFERENCE_ANTIBODY_SEQUENCES: 

986 if align(reference_sequence, sequence): 

987 is_antibody = True 

988 break 

989 print(f' I guess it {"is" if is_antibody else "is not"} an antibody') 

990 # If so, the also set this chain as no referable since antibodies have no UniProt id 

991 if is_antibody: 

992 uniprot_ids.append(NoReferableException(sequence)) 

993 continue 

994 # If we did not fall in any of the previous sections then continue 

995 continue 

996 # If we have multiple uniprots in a single entity then we skip them 

997 # Note tha they belong to an entity which is no referable (e.g. a chimeric entity) 

998 # See 5GY2 and 7E2Z labeled as chimeric entities and 6e67, which is not labeled likewise 

999 if len(uniprots) > 1: 

1000 warn(f'Multiple UniProt ids in the same entity in {pdb_id} -> Is this a chimeric entity?') 

1001 if not sequence: raise ValueError(f'Missing sequence with multiple UniProt ids in {pdb_id}') 

1002 uniprot_ids.append(NoReferableException(sequence)) 

1003 continue 

1004 # Normally, a polymer/chain will have one UniProt id only 

1005 uniprot_id = uniprots[0] 

1006 uniprot_ids.append(uniprot_id) 

1007 # Count how many UniProt ids we found 

1008 actual_uniprot_ids = [uniprot_id for uniprot_id in uniprot_ids if type(uniprot_id) is str] 

1009 if len(actual_uniprot_ids) > 0: 

1010 print(f' UniProt ids for PDB id {pdb_id}: ' + ', '.join(actual_uniprot_ids))
