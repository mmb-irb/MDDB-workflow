1"""Pydantic schema for the workflow inputs file (inputs.yaml / inputs.json). 

2 

3This module validates the user-provided inputs file and reports problems in a 

4clear, per-field way before the workflow proceeds. Validation is intentionally 

5lenient with unknown top-level fields (they are preserved and only warned about) 

6so legacy or forward-compatible files keep working, while the structured nested 

7fields (ligands, interactions, mds, links, customs...) are strongly typed since 

8mistakes there are the hardest to debug. 

9""" 

10

20def check_translation_value(value): 

21 """Make sure the translation is either the automatic flag or a list of 3 numbers.""" 

22 if value is None: 

23 return value 

24 if type(value) is str: 

25 if value != AUTOMATIC_FLAG: 

26 raise ValueError(f'Translation must be "{AUTOMATIC_FLAG}" or a list of 3 numbers, got "{value}".') 

27 return value 

28 if len(value) != 3: 

29 raise ValueError(f'Translation must be a list of 3 numbers (x y z), got {len(value)}.') 

30 return value

33class InputsBaseModel(BaseModel): 

34 """Base model tolerating (and preserving) unknown fields.""" 

35 

36 model_config = ConfigDict(extra='allow') 

37 

38 @model_validator(mode='before') 

39 @classmethod 

40 def check_is_mapping(cls, data): 

41 if not isinstance(data, dict): 

42 required_fields = [name for name, field in cls.model_fields.items() if field.is_required()] 

43 fields_hint = f' with fields: {", ".join(required_fields)}' if required_fields else '' 

44 raise ValueError(f'Each "{cls.__name__.lower()}" entry must be a dictionary{fields_hint}.') 

45 return data

48class Ligand(InputsBaseModel): 

49 """A ligand defined in the simulation. 

50 

51 Each ligand must have at least one database accession so it can be mapped. 

52 As a shorthand, a ligand may also be given as a bare PubChem CID (an int or 

53 a numeric string), which is expanded into a full entry during validation. 

54 """ 

55 

56 pubchem: Optional[Union[int, str]] = None 

57 drugbank: Optional[str] = None 

58 chembl: Optional[str] = None 

59 # Optional forced mapping through a list of VMD selections 

60 vmd_selection: Optional[Union[str, list[str]]] = None 

61 

62 @model_validator(mode='before') 

63 @classmethod 

64 def expand_pubchem_shorthand(cls, data): 

65 """Expand a bare PubChem CID (int or numeric string) into a full entry. 

66 A non-numeric bare string (a ligand name) is rejected. 

67 """ 

68 if type(data) is int or (type(data) is str and data.isnumeric()): 

69 return {'pubchem': str(data)} 

70 if type(data) is str: 

71 raise ValueError(f'A ligand name has been identified: {data}. ' 

72 'Please provide at least one of the following IDs: PubChem, DrugBank, ChEMBL.') 

73 # The forced selection field was previously read as 'selection' 

74 if type(data) is dict and 'selection' in data: 

75 raise ValueError('The ligand field "selection" is not supported. Use "vmd_selection" instead.') 

76 return data 

77 

78 @model_validator(mode='after') 

79 def check_has_accession(self) -> 'Ligand': 

80 if self.pubchem is None and self.drugbank is None and self.chembl is None: 

81 raise ValueError('Each ligand must have at least one of the following ' 

82 'attributes: "pubchem", "drugbank" or "chembl".') 

83 return self

238 if value is not None and value <= 0:

242 @field_validator('translation')

245 return check_translation_value(value)

247 @field_validator('skip_frames') 

248 @classmethod 

249 def check_skip_frames(cls, value): 

250 if value is None: 

251 return value 

252 for md_index, frames in value.items(): 

253 if md_index < 0: 

254 raise ValueError(f'Wrong MD index {md_index} in "skip_frames": it must not be a negative number.') 

255 if frames < 0: 

256 raise ValueError(f'Wrong number of frames for MD {md_index} in "skip_frames": it must not be a negative number.') 

257 return value 

258 

259 @field_validator('mercy', 'trust') 

260 @classmethod 

261 def check_test_flags(cls, value, info): 

262 if value is None or type(value) is bool: 

263 return value 

264 available = AVAILABLE_FAILURES if info.field_name == 'mercy' else AVAILABLE_CHECKINGS 

265 for flag in value: 

266 if flag not in available: 

267 raise ValueError(f'"{flag}" is not a valid "{info.field_name}" value. ' 

268 f'Available values: {", ".join(available)}.') 

269 return value 

270 

271 @field_validator('pdb_ids') 

272 @classmethod 

273 def check_pdb_ids(cls, value): 

274 if value is None: 

275 return value 

276 pdb_ids = [value] if isinstance(value, str) else value 

277 short_pdb_ids = [] 

278 for pdb_id in pdb_ids: 

279 # Extended ids with a classic 4 character equivalent are converted to it 

280 # This way they match the ids already used in other projects 

281 if re.match(EXTENDED_PDB_ID_FORMAT, pdb_id) and pdb_id[4:8] == '0000': 

282 pdb_id = pdb_id[8:] 

283 if not re.match(PDB_ID_FORMAT, pdb_id) and not re.match(EXTENDED_PDB_ID_FORMAT, pdb_id): 

284 raise ValueError(f'"{pdb_id}" does not look like a PDB id.') 

285 short_pdb_ids.append(pdb_id) 

286 return short_pdb_ids[0] if isinstance(value, str) else short_pdb_ids

293def format_validation_error(error: ValidationError) -> str: 

294 """Format a pydantic ValidationError into a readable, per-field message.""" 

295 lines = [f'Found {error.error_count()} problem(s) in the inputs file:'] 

296 for problem in error.errors(): 

297 # Build a dotted location path (e.g. "ligands -> 0 -> pubchem") 

298 location = ' -> '.join(str(part) for part in problem['loc']) or '(root)' 

299 # Custom validators raise plain ValueErrors, which pydantic prefixes with 

300 # "Value error, "; strip it since our messages are already self-contained 

301 message = problem['msg'].removeprefix('Value error, ') 

302 # Clarify the type and value that caused the error, when available 

303 detail = '' 

304 if 'input' in problem: 

305 value = problem['input'] 

306 detail = f' (got {type(value).__name__}: {value!r})' 

307 lines.append(f' - {location}: {message}{detail}') 

308 return '\n'.join(lines)

311def validate_inputs(inputs_data: dict, strict_unknown: bool = False) -> dict: 

312 """Validate a loaded inputs dictionary against the workflow schema. 

313 

314 Args: 

315 inputs_data (dict): The loaded inputs dictionary to validate. 

316 strict_unknown (bool): If True, unknown top-level fields will raise an 

317 InputError. If False, they will be preserved and only warned about. 

318 

319 """ 

320 # Report unknown top-level fields (legacy or typos) 

321 unknown_fields = [key for key in inputs_data if key not in KNOWN_INPUT_FIELDS] 

322 if unknown_fields: 

323 listed_fields = ', '.join(sorted(unknown_fields)) 

324 if strict_unknown: 

325 available_fields = ', '.join(sorted(KNOWN_INPUT_FIELDS)) 

326 raise InputError(f'Unrecognized input field(s): {listed_fields}.\n' 

327 f' Available inputs: {available_fields}') 

328 warn('Unknown field(s) in the inputs file (they will be ignored by the ' 

329 f'workflow): {listed_fields}') 

330 # Run the actual validation 

331 try: 

332 validated = WorkflowInputs.model_validate(inputs_data) 

333 except ValidationError as error: 

334 raise InputError(format_validation_error(error)) 

335 # Write back the normalized ligands 

336 if validated.ligands is not None: 

337 inputs_data['ligands'] = [ligand.model_dump(exclude_none=True) for ligand in validated.ligands] 

338 # Write back the PDB ids, since extended ids are converted to the classic ones 

339 if validated.pdb_ids is not None: 

340 inputs_data['pdb_ids'] = validated.pdb_ids 

341 return inputs_data
