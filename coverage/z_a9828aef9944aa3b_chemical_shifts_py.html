1# Chemical shifts analysis 

2# 

3# The chemical shifts analysis is carried by the 'LEGOLAS' software. LEGOLAS predicts the NMR chemical 

4# shifts of protein backbone atoms (H, HA, N, CA, CB and C) using an ensemble of 5 neural networks 

5# trained over atomic environment vectors (AEVs). The chemical shift is predicted for every frame and 

6# the standard deviation between the 5 models is returned as well as a measure of the uncertainty. 

7# 

8# https://github.com/roitberg-group/legolas 

9# 

10# This analysis was integrated by Claude 

11

32# LEGOLAS needs a PDB structure 

33LEGOLAS_STRUCTURE_FILENAME = 'structure.pdb' 

34 

35# DANI: Esto es provisional hasta que LEGOLAS esté en conda 

36LEGOLAS_COMMAND = 'source ~/miniforge3/bin/activate ~/miniforge3/envs/legolas; python ~/libraries/legolas/test/legolas.py {} -t {} -o csv' 

37 

38# Atom types predicted by LEGOLAS 

39# LEGOLAS sets the atom type by removing digits from the atom name (e.g. HA2 -> HA, H1 -> H) 

40LEGOLAS_ATOM_TYPES = {'H', 'HA', 'N', 'CA', 'CB', 'C'} 

41 

42# Residue names supported by LEGOLAS 

43# Water (HOH) is also supported but we do not include it since it would make LEGOLAS much slower 

44LEGOLAS_RESIDUE_NAMES = {'ALA', 'ARG', 'ASN', 'ASP', 'CYS', 'GLU', 'GLN', 'GLY', 'HIS', 'ILE', 

45 'LEU', 'LYS', 'MET', 'PHE', 'PRO', 'SER', 'THR', 'TRP', 'TYR', 'VAL'} 

46# Non-standard residue names which may be translated to a supported residue name 

47RESIDUE_NAME_ALIASES = { 

48 'HID': 'HIS', 'HIE': 'HIS', 'HIP': 'HIS', 'HSD': 'HIS', 'HSE': 'HIS', 'HSP': 'HIS', 

49 'CYX': 'CYS', 'CYM': 'CYS', 'ASH': 'ASP', 'GLH': 'GLU', 'LYN': 'LYS', 

50} 

51 

52# Number of decimals to keep in the output 

53OUTPUT_DECIMALS = 3

56def chemical_shifts( 

57 structure_file: 'File', 

58 trajectory_file: 'File', 

59 structure: 'Structure', 

60 pbc_selection: 'Selection', 

61 cg_selection: 'Selection', 

62 snapshots: int, 

63 output_directory: str, 

64 frames_limit: int = 1000, 

65 time_splits: int = 10, 

66): 

67 """Perform the chemical shifts analysis using LEGOLAS.""" 

68 warn('The "chemical shifts" analysis is not yet fully integrated. LEGOLAS is not installed in the conda enviornment') 

69 

70 # Every time split must have at least one frame 

71 if frames_limit < time_splits: 

72 raise InputError(f'Frames limit ({frames_limit}) must be greater than time splits ({time_splits})') 

73 

74 # Set the residues to be analyzed: protein residues with a residue name supported by LEGOLAS 

75 # Residues in PBC and coarse grain residues are excluded 

76 excluded_residue_indices = set(structure.get_selection_residue_indices(pbc_selection + cg_selection)) 

77 protein_residue_indices = sorted(structure.get_selection_residue_indices(structure.select_protein())) 

78 legolas_residue_indices = [] 

79 for residue_index in protein_residue_indices: 

80 if residue_index in excluded_residue_indices: continue 

81 residue = structure.residues[residue_index] 

82 if get_legolas_residue_name(residue.name) is None: 

83 warn(f'Residue {residue} is not supported by LEGOLAS and it will be skipped') 

84 continue 

85 legolas_residue_indices.append(residue_index) 

86 if len(legolas_residue_indices) == 0: 

87 print(' No protein residues to analyze') 

88 return 

89 

90 # Set the LEGOLAS working directory, where all LEGOLAS inputs and outputs will be 

91 legolas_directory = f'{output_directory}/legolas' 

92 if not exists(legolas_directory): mkdir(legolas_directory) 

93 

94 # Write a structure with only the analyzed residues 

95 # Residues are renamed to the names supported by LEGOLAS 

96 # Residues are renumbered so every residue has a unique number in LEGOLAS outputs 

97 # Note that LEGOLAS sorts the output by residue number, so this also guarantees the order is kept 

98 legolas_selection = structure.select_residue_indices(legolas_residue_indices) 

99 legolas_structure = structure.filter(legolas_selection) 

100 for r, residue in enumerate(legolas_structure.residues): 

101 residue.name = get_legolas_residue_name(residue.name) 

102 residue.number = r + 1 

103 residue.icode = '' 

104 legolas_structure_filepath = f'{legolas_directory}/{LEGOLAS_STRUCTURE_FILENAME}' 

105 # Note that residues are renumbered so chains are not needed to keep them apart 

106 legolas_structure.generate_pdb_file(legolas_structure_filepath, accept_limitations=True) 

107 

108 # Write a reduced trajectory with only the analyzed residues atoms 

109 reduced_trajectory_filepath, reduced_step, _ = get_reduced_trajectory( 

110 trajectory_file, 

111 snapshots, 

112 frames_limit, 

113 ) 

114 print(' Filtering trajectory for LEGOLAS') 

115 legolas_trajectory = mdt.load(reduced_trajectory_filepath, top=structure_file.path, 

116 atom_indices=legolas_selection.atom_indices) 

117 legolas_trajectory_filepath = f'{legolas_directory}/{trajectory_file.filename}' 

118 legolas_trajectory.save(legolas_trajectory_filepath) 

119 

120 # Run LEGOLAS 

121 print(' Running LEGOLAS') 

122 print(GREY_HEADER, end='') 

123 command = LEGOLAS_COMMAND.format(trajectory_file.filename, LEGOLAS_STRUCTURE_FILENAME) 

124 process = run(['bash', '-c', command], cwd=legolas_directory, stdout=PIPE, stderr=STDOUT) 

125 logs = process.stdout.decode() 

126 print(COLOR_END, end='') 

127 # LEGOLAS writes the output in the working directory and names it after the trajectory 

128 legolas_output_filepath = f'{legolas_directory}/{Path(trajectory_file.filename).stem}_cs.csv' 

129 if process.returncode != 0 or not exists(legolas_output_filepath): 

130 print(logs) 

131 raise ToolError('Something went wrong with LEGOLAS') 

132 

133 # Set the atoms we expect LEGOLAS to predict, grouped by atom type 

134 # Atoms are sorted by residue number and then by atom index, just like in the LEGOLAS output 

135 expected_atoms = { atom_type: [] for atom_type in LEGOLAS_ATOM_TYPES } 

136 for atom in sorted(legolas_structure.atoms, key=lambda atom: (atom.residue.number, atom.index)): 

137 atom_type = get_legolas_atom_type(atom.name) 

138 if atom_type not in LEGOLAS_ATOM_TYPES: continue 

139 expected_atoms[atom_type].append(atom) 

140 

141 # Read the LEGOLAS output 

142 # There is a row per atom and chemical shifts are lists with a value per frame 

143 # Rows are sorted by residue number and atom type (alphabetically) 

144 # Atoms with the same residue and atom type (e.g. H1, H2 and H3) are sorted by atom index 

145 # Note that cells are huge so we must increase the csv field size limit 

146 csv.field_size_limit(sys.maxsize) 

147 with open(legolas_output_filepath, 'r') as file: 

148 legolas_output = list(csv.DictReader(file)) 

149 # Note that atom indices here are still the indices in the LEGOLAS structure 

150 legolas_atom_indices = [] 

151 atom_type_counters = { atom_type: 0 for atom_type in LEGOLAS_ATOM_TYPES } 

152 for row in legolas_output: 

153 atom_type, residue_name = row['ATOM_TYPE'], row['RES_TYPE'] 

154 residue_number = int(row['SEQ_ID']) 

155 atom_type_atoms = expected_atoms[atom_type] 

156 atom_type_counter = atom_type_counters[atom_type] 

157 if atom_type_counter >= len(atom_type_atoms): 

158 raise ToolError(f'LEGOLAS returned more {atom_type} atoms than expected') 

159 expected_atom = atom_type_atoms[atom_type_counter] 

160 expected_residue = expected_atom.residue 

161 if residue_number != expected_residue.number or residue_name != expected_residue.name: 

162 raise ToolError(f'LEGOLAS atom {atom_type} in residue {residue_name} {residue_number} does not' 

163 f' match atom {expected_atom.name} in residue {expected_residue.name} {expected_residue.number}') 

164 legolas_atom_indices.append(expected_atom.index) 

165 atom_type_counters[atom_type] += 1 

166 for atom_type, atom_type_atoms in expected_atoms.items(): 

167 if atom_type_counters[atom_type] != len(atom_type_atoms): 

168 raise ToolError(f'LEGOLAS returned {atom_type_counters[atom_type]} {atom_type} atoms' 

169 f' but we expected {len(atom_type_atoms)}') 

170 # Map LEGOLAS structure atom indices to the original structure atom indices 

171 # Sort atoms by atom index 

172 atom_indices = [legolas_selection.atom_indices[atom_index] for atom_index in legolas_atom_indices] 

173 atom_order = sorted(range(len(atom_indices)), key=lambda row: atom_indices[row]) 

174 

175 # Set the time splits 

176 # The start will be always 0 since we start with the first frame 

177 start = 0 

178 # Calculate how many frames fall in every time split 

179 # Note that step refers to frames in the original trajectory, not in the reduced trajectory 

180 step = ceil(snapshots / time_splits) 

181 # Calculate how many time splits we will have at the end 

182 # Note that the last split may have less frames than the rest 

183 nsteps = ceil(snapshots / step) 

184 

185 # Parse chemical shifts and summarize them along all frames and along frames in every time split 

186 # Note that LEGOLAS returns a single value instead of a list when there is only one frame 

187 def parse_values(text: str) -> list[float]: 

188 values = json.loads(text) 

189 if type(values) != list: values = [values] 

190 return values 

191 # Split values according to the original trajectory frame they belong to 

192 def split_values(values: list[float]) -> list[list[float]]: 

193 splits = [ [] for _ in range(nsteps) ] 

194 for reduced_frame, value in enumerate(values): 

195 splits[reduced_frame * reduced_step // step].append(value) 

196 return splits 

197 def round_value(value: float) -> float: 

198 return round(value, OUTPUT_DECIMALS) 

199 chemical_shift_averages = [] 

200 chemical_shift_deviations = [] 

201 model_deviation_averages = [] 

202 timed_chemical_shift_averages = [] 

203 timed_chemical_shift_deviations = [] 

204 for row in legolas_output: 

205 chemical_shifts = parse_values(row['CHEMICAL_SHIFT']) 

206 chemical_shift_averages.append(round_value(mean(chemical_shifts))) 

207 chemical_shift_deviations.append(round_value(pstdev(chemical_shifts))) 

208 chemical_shift_splits = split_values(chemical_shifts) 

209 timed_chemical_shift_averages.append([ round_value(mean(split)) for split in chemical_shift_splits ]) 

210 timed_chemical_shift_deviations.append([ round_value(pstdev(split)) for split in chemical_shift_splits ]) 

211 # Note that the deviation returned by LEGOLAS is the standard deviation between its 5 models 

212 # i.e. it is a measure of the prediction uncertainty, not of the fluctuation along the trajectory 

213 model_deviations = parse_values(row['CHEMICAL_SHIFT_STD']) 

214 model_deviation_averages.append(round_value(mean(model_deviations))) 

215 

216 # Set the output analysis 

217 # Note that csdv is the deviation along frames while csmdv is the average deviation between LEGOLAS models 

218 # Note that timed values have a list per atom with a value per time split 

219 output_analysis = { 

220 'atndx': [atom_indices[row] for row in atom_order], 

221 'csav': [chemical_shift_averages[row] for row in atom_order], 

222 'csdv': [chemical_shift_deviations[row] for row in atom_order], 

223 'csmdv': [model_deviation_averages[row] for row in atom_order], 

224 'tcsav': [timed_chemical_shift_averages[row] for row in atom_order], 

225 'tcsdv': [timed_chemical_shift_deviations[row] for row in atom_order], 

226 'start': start, 

227 'step': step, 

228 } 

229 save_json(output_analysis, f'{output_directory}/{OUTPUT_CHEMICAL_SHIFTS_FILENAME}') 

230 

231 # Remove LEGOLAS files since they are heavy and we already have what we need 

232 rmtree(legolas_directory)

235def get_legolas_residue_name(residue_name: str) -> Optional[str]: 

236 """Get the residue name as LEGOLAS expects it, or None if it is not supported.""" 

237 residue_name = RESIDUE_NAME_ALIASES.get(residue_name, residue_name) 

238 if residue_name not in LEGOLAS_RESIDUE_NAMES: return None 

239 return residue_name

242def get_legolas_atom_type(atom_name: str) -> str: 

243 """Get the atom type as LEGOLAS sets it: the atom name without digits.""" 

244 return re.sub(r'\d', '', atom_name)
