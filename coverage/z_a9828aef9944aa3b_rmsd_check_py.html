563def compute_periodic_distances( 

564 positions : np.ndarray, 

565 box_vectors : np.ndarray, 

566 pairs : np.ndarray, 

567) -> np.ndarray: 

568 """Get the distance between the atoms of every pair, using the closest periodic image. 

569 Boxes may be rotated (e.g. in fitted trajectories), which MDtraj does not support. 

570 For this reason the whole frame is rotated back before calculating distances. 

571 A box full of zeros means there are no periodic boundaries. 

572 """ 

573 has_box = box_vectors.any() 

574 if has_box: positions, box_vectors = unrotate_box(positions, box_vectors) 

575 frame = mdt.Trajectory(positions[np.newaxis], None) 

576 if has_box: frame.unitcell_vectors = box_vectors[np.newaxis] 

577 return mdt.compute_distances(frame, pairs, periodic=True)[0]

580def get_pair_scales( 

581 trajectory_filename : str, 

582 pairs : np.ndarray, 

583 snapshots : int, 

584) -> np.ndarray: 

585 """Find how much the distance of every pair usually changes from one frame to the next. 

586 It is the median change in some jumps spread along the trajectory, so a few strange jumps do not affect it. 

587 """ 

588 jump_frames = np.unique(np.linspace(1, snapshots - 1, SCALE_SAMPLED_JUMPS).astype(int)) 

589 changes = [] 

590 with mdt.formats.XTCTrajectoryFile(trajectory_filename) as reader: 

591 for jump_frame in jump_frames: 

592 reader.seek(jump_frame - 1) 

593 xyz, _, _, boxes = reader.read(n_frames=2) 

594 # Trajectories with no box have no periodic boundaries, as a box full of zeros 

595 if boxes is None: boxes = np.zeros((2, 3, 3), dtype=np.float32) 

596 previous_distances = compute_periodic_distances(xyz[0], boxes[0], pairs) 

597 distances = compute_periodic_distances(xyz[1], boxes[1], pairs) 

598 changes.append(np.abs(distances - previous_distances)) 

599 scales = np.median(changes, axis=0) 

600 still_pairs = pairs[scales == 0] 

601 if len(still_pairs) > 0: 

602 raise ValueError(f'The distance between some pairs of atoms (e.g. atoms {still_pairs[0][0]} and {still_pairs[0][1]})' 

603 ' does not change in most frames. Are there repeated frames in the trajectory?') 

604 return scales

607def fit_gamma_shape(log_ratio : float) -> float: 

608 """Find the shape of the gamma distribution which fits best some values. 

609 The log ratio is the log of the mean of the values minus the mean of the logs of the values. 

610 It starts from an approximation which is then refined a few times. 

611 """ 

612 shape = (3 - log_ratio + math.sqrt((log_ratio - 3) ** 2 + 24 * log_ratio)) / (12 * log_ratio) 

613 for _ in range(10): 

614 shape -= (math.log(shape) - special.digamma(shape) - log_ratio) / (1 / shape - special.polygamma(1, shape)) 

615 return shape

618def gamma_log_densities(values : np.ndarray, log_values : np.ndarray, shape : float, scale : float) -> np.ndarray: 

619 """Get the log of the gamma distribution density for every value.""" 

620 return (shape - 1) * log_values - values / scale - shape * math.log(scale) - special.gammaln(shape)

623def check_bimodality(jumps : np.ndarray) -> tuple[float, np.ndarray, tuple[float, float]]: 

624 """Check if jumps are split in two groups: normal jumps and bigger jumps. 

625 When nothing is wrong squared jumps follow a gamma distribution: many small values and a tail of bigger values. 

626 So we fit one gamma and two gammas and check how much better two gammas fit, as a BIC difference. 

627 Both gammas have the same shape, so they differ only in the size of their jumps. 

628 Return the bimodality score, the probability of every jump to be in the group of bigger jumps 

629 and the average squared jump of each group, smaller group first. 

630 """ 

631 values = jumps ** 2 

632 # A jump may be exactly 0 because of the trajectory precision, but a gamma has no values at 0 

633 values = np.maximum(values, values[values > 0].min()) 

634 log_values = np.log(values) 

635 mean_log = log_values.mean() 

636 # Fit one gamma 

637 shape = fit_gamma_shape(math.log(values.mean()) - mean_log) 

638 one_gamma_log_likelihood = gamma_log_densities(values, log_values, shape, values.mean() / shape).sum() 

639 # Fit two gammas 

640 # Every start puts a different number of the biggest jumps in the bigger group and the best fit is kept 

641 best_log_likelihood = -np.inf 

642 for start_fraction in TWO_GROUPS_START_FRACTIONS: 

643 bigger_probabilities = (values >= np.quantile(values, 1 - start_fraction)).astype(float) 

644 log_likelihood = -np.inf 

645 for _ in range(TWO_GROUPS_MAX_ITERATIONS): 

646 # Update the gammas according to which jumps are in every group 

647 bigger_weight = bigger_probabilities.mean() 

648 smaller_mean = np.average(values, weights=1 - bigger_probabilities) 

649 bigger_mean = np.average(values, weights=bigger_probabilities) 

650 shape = fit_gamma_shape(bigger_weight * math.log(bigger_mean) 

651 + (1 - bigger_weight) * math.log(smaller_mean) - mean_log) 

652 # Update which jumps are in every group according to the gammas 

653 smaller_log_densities = math.log(1 - bigger_weight) + gamma_log_densities(values, log_values, shape, smaller_mean / shape) 

654 bigger_log_densities = math.log(bigger_weight) + gamma_log_densities(values, log_values, shape, bigger_mean / shape) 

655 total_log_densities = np.logaddexp(smaller_log_densities, bigger_log_densities) 

656 bigger_probabilities = np.exp(bigger_log_densities - total_log_densities) 

657 previous_log_likelihood = log_likelihood 

658 log_likelihood = total_log_densities.sum() 

659 if log_likelihood - previous_log_likelihood < TWO_GROUPS_TOLERANCE: break 

660 if log_likelihood > best_log_likelihood: 

661 best_log_likelihood = log_likelihood 

662 best_fit = (bigger_probabilities, smaller_mean, bigger_mean) 

663 bigger_probabilities, smaller_mean, bigger_mean = best_fit 

664 # The group which starts with the biggest jumps may end up being the smaller one 

665 if bigger_mean < smaller_mean: 

666 bigger_probabilities = 1 - bigger_probabilities 

667 smaller_mean, bigger_mean = bigger_mean, smaller_mean 

668 # One gamma has 2 parameters (shape and scale) and two gammas have 4 (shape, two scales and the weight) 

669 log_n = math.log(len(values)) 

670 one_gamma_bic = 2 * log_n - 2 * one_gamma_log_likelihood 

671 two_gammas_bic = 4 * log_n - 2 * best_log_likelihood 

672 return one_gamma_bic - two_gammas_bic, bigger_probabilities, (smaller_mean, bigger_mean)

675def pick_typical_jumps(report : dict) -> tuple[list[int], list[int]]: 

676 """Pick the most typical jumps of every group, i.e. the ones closest to the average of their group. 

677 Return the frame after every jump, starting at 0, with the most typical jump first. 

678 """ 

679 values = report['scaled_jumps'] ** 2 

680 is_bigger = report['bigger_probabilities'] > 0.5 

681 picked_frames = [] 

682 for in_group, group_mean in zip([ ~is_bigger, is_bigger ], report['group_means']): 

683 group_indices = np.where(in_group)[0] 

684 typical_indices = group_indices[np.argsort(np.abs(values[group_indices] - group_mean))] 

685 picked_frames.append([ int(index) + 1 for index in typical_indices[:SHOWN_JUMPS_PER_GROUP] ]) 

686 return tuple(picked_frames)

689def find_jumping_atoms( 

690 trajectory_filename : str, 

691 fragment : 'Selection', 

692 pairs : Optional[np.ndarray], 

693 jump_frame : int, 

694) -> 'Selection': 

695 """Find the atoms of the bonds which change the most in a jump, from the previous frame to the jump frame. 

696 Fragments with a single atom have no bonds, so the atom itself is returned. 

697 """ 

698 if pairs is None: return fragment 

699 with mdt.formats.XTCTrajectoryFile(trajectory_filename) as reader: 

700 reader.seek(jump_frame - 1) 

701 xyz, _, _, boxes = reader.read(n_frames=2) 

702 # Trajectories with no box have no periodic boundaries, as a box full of zeros 

703 if boxes is None: boxes = np.zeros((2, 3, 3), dtype=np.float32) 

704 previous_distances = compute_periodic_distances(xyz[0], boxes[0], pairs) 

705 distances = compute_periodic_distances(xyz[1], boxes[1], pairs) 

706 changes = np.abs(distances - previous_distances) 

707 biggest_changes = np.argsort(-changes)[:HIGHLIGHTED_BONDS] 

708 # Keep only bonds which change at least half as much as the bond which changes the most 

709 # Otherwise a single broken bond would be shown along with bonds which barely change 

710 biggest_changes = biggest_changes[changes[biggest_changes] >= changes[biggest_changes[0]] / 2] 

711 return Selection([ int(index) for index in np.unique(pairs[biggest_changes]) ])

730 

731 The shown frames are written to a small trajectory next to the script, in the same order.

739 for frame in shown_frames: 

740 reader.seek(frame) 

741 xyz, _, _, boxes = reader.read(n_frames=1) 

742 writer.write(xyz, box=boxes) 

743 commands = [ 

744 f'mol new {{{abspath(structure_filename)}}} waitfor all', 

745 # The frame from the structure file is removed so only the shown frames remain 

746 'animate delete all',

750 'mol representation Licorice', 

751 'mol color ColorID 4',

753 'mol material Opaque',

756 'mol representation CPK', 

757 'mol color Name', 

758 f'mol selection {{{jumping_atoms.to_vmd()}}}', 

759 'mol addrep $main', 

760 # Center the view in the fragment and zoom in 

761 f'animate goto {focus_frame}', 

762 f'set fragment [atomselect $main {{{fragment.to_vmd()}}}]',

764 ' lassign [measure minmax $atoms] lowest highest',

767 'display resetview', 

768 'molinfo $main set center_matrix [list [transoffset [vecscale -1 [measure center $fragment]]]]', 

769 f'set zoom [expr {{[get_size [atomselect $main all]] / max([get_size $fragment], {MIN_FOCUS_SIZE})}}]', 

770 'if {$zoom > 1} { scale by $zoom }',

772 commands += [ f'puts {{{note}}}' for note in notes ] 

773 with open(script_filename, 'w') as script_file:
