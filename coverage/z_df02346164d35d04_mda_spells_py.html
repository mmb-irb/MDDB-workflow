31def to_MDAnalysis_topology(standard_topology_path: str) -> 'Topology': 

32 """Create a MDAnalysis topology from a json topology file. 

33 

34 The json file should contain the following keys: 

35 - atom_names 

36 - atom_elements 

37 - atom_charges 

38 - atom_bonds (list of lists of atom indices) 

39 - residue_names 

40 - residue_numbers 

41 - chain_names 

42 - atom_residue_indices 

43 - residue_chain_indices 

44 """ 

45 topology = json.load(open(standard_topology_path)) 

46 

47 # transform bond to non redundant tuples 

48 bonds = [] 

49 for bond_from, bond_tos in enumerate(topology['atom_bonds']): 

50 # Atoms with missing bonds have no bonds 

51 if bond_tos == JSON_SERIALIZABLE_MISSING_BONDS: continue 

52 for bond_to in bond_tos: 

53 bond = tuple(sorted([bond_from, bond_to])) 

54 bonds.append(bond) 

55 # sorted(set(bonds)) if you want to check for duplicates 

56 topology['bonds'] = set(bonds) 

57 atom_2_chain = np.array(topology['residue_chain_indices'])[topology['atom_residue_indices']] 

58 topology['chainIDs'] = np.array(topology['chain_names'])[atom_2_chain] 

59 

60 attrs = [Atomnames(topology['atom_names']), 

61 Elements(topology['atom_elements']), 

62 Bonds(topology['bonds']), 

63 Resnames(topology['residue_names']), 

64 Resnums(topology['residue_numbers']), 

65 ChainIDs(topology['chainIDs']), 

66 Segids(topology['chain_names']), 

67 Resids(np.arange(len(topology['residue_names']))), 

68 ] 

69 if topology['atom_charges']: 

70 attrs.append(Charges(topology['atom_charges'])) 

71 

72 mda_top = Topology(n_atoms=len(topology['atom_names']), 

73 n_res=len(topology['residue_names']), 

74 n_seg=len(topology['chain_names']), 

75 attrs=attrs, 

76 atom_resindex=topology['atom_residue_indices'], 

77 residue_segindex=topology['residue_chain_indices'] 

78 ) 

79 return mda_top

96def set_structure_residues(universe: 'Universe', structure: 'Structure', trajectory_file: 'File') -> 'Universe': 

97 """Make a new universe whose residues are the residues in the structure. 

98 

99 MDAnalysis groups atoms in residues by their residue number in the PDB. 

100 When there are more than 9999 residues we write hexadecimal numbers, which MDAnalysis 

101 can not read, so it sets them all to 1 and joins consecutive residues with the same name. 

102 """ 

103 topology = universe._topology 

104 # Keep all atom values (names, elements, etc.) as they are 

105 attrs = [attr for attr in topology.attrs if isinstance(attr, AtomAttr) and not isinstance(attr, Atomindices)] 

106 # Each residue stays in the segment of its first atom 

107 residue_first_atoms = [residue.atom_indices[0] for residue in structure.residues] 

108 residue_numbers = [residue.number for residue in structure.residues] 

109 attrs += [ 

110 Resids(np.array(residue_numbers)), 

111 Resnums(np.array(residue_numbers)), 

112 Resnames(np.array([residue.name for residue in structure.residues], dtype=object)), 

113 ICodes(np.array([residue.icode or '' for residue in structure.residues], dtype=object)), 

114 Segids(universe.segments.segids), 

115 ] 

116 new_topology = Topology(n_atoms=len(universe.atoms), 

117 n_res=len(structure.residues), 

118 n_seg=len(universe.segments), 

119 attrs=attrs, 

120 atom_resindex=np.array([atom.residue_index for atom in structure.atoms]), 

121 residue_segindex=universe.atoms.segindices[residue_first_atoms]) 

122 return Universe(new_topology, trajectory_file.path)

125def get_mda_universe(structure_file: 'File', # To load in MDAnalysis 

126 trajectory_file: 'File', # To load in MDAnalysis 

127 structure: 'Structure', # To set the residues 

128 reference_bonds: list[list[int]], # To set the bonds 

129 charges: list[float]) -> 'Universe': # To set the charges 

130 """Create a MDAnalysis universe using data in the workflow.""" 

131 # Make MDAnalysis warnings and logs grey 

132 print(GREY_HEADER, end='\r') 

133 universe = Universe(structure_file.path, trajectory_file.path) 

134 universe = set_structure_residues(universe, structure, trajectory_file) 

135 # Set the atom bonds 

136 bonds = [] 

137 for bond_from, bond_tos in enumerate(reference_bonds): 

138 if bond_tos == MISSING_BONDS: continue 

139 for bond_to in bond_tos: 

140 bond = tuple(sorted([bond_from, bond_to])) 

141 bonds.append(bond) 

142 universe.add_TopologyAttr('bonds', set(bonds)) 

143 

144 # Set the charges 

145 if charges and charges != MISSING_CHARGES and len(charges) > 0: 

146 universe.add_TopologyAttr('charges', np.array(charges, dtype=np.float32)) 

147 

148 # No elements can happen if we use the faith flag 

149 if not hasattr(universe.atoms, 'elements'): 

150 universe.guess_TopologyAttrs(to_guess=['elements']) 

151 print(COLOR_END, end='\r') 

152 return universe

174def get_all_acyl_chains(residue: 'MDAnalysis.Residue', min_length=6, removeHs=True) -> list[list[int]]: 

175 """Find all groups of connected Carbon atoms within a residue which are not in rings. 

176 

177 Returns: 

178 list 

179 A list of lists, where each inner list contains atom indices of 

180 connected Carbon atoms forming a distinct group 

181 

182 """ 

183 def explore_carbon_group(start_atom, visited): 

184 """Explore a connected group of carbons.""" 

185 to_visit = [start_atom] 

186 group = set() 

187 while to_visit: 

188 current_atom = to_visit.pop(0) 

189 if current_atom.index in visited: 

190 continue 

191 visited.add(current_atom.index) 

192 if current_atom.element == 'C': 

193 group.add(current_atom.index) 

194 # Add all bonded carbon atoms of the residue to the visit queue 

195 # e.g. AMBER Lipid14 has the ester carbon of the PA tail in the PC residue 

196 for bond in current_atom.bonds: 

197 for bonded_atom in bond.atoms: 

198 if (bonded_atom.element == 'C' and 

199 bonded_atom.resindex == residue.resindex and 

200 bonded_atom.index not in visited): 

201 to_visit.append(bonded_atom) 

202 return list(group) 

203 

204 # Atoms in rings of the residue, also rings closed through other atoms (e.g. the O5 of pyranose sugars) 

205 residue_atoms = set(residue.atoms.indices.tolist()) 

206 graph = nx.Graph([bond for bond in residue.atoms.bonds.indices.tolist() if residue_atoms.issuperset(bond)]) 

207 ring_atoms = {index for cycle in nx.cycle_basis(graph) for index in cycle} 

208 # Get all Carbon atoms in the residue 

209 carbon_atoms = residue.atoms.select_atoms('element C') 

210 visited = set() 

211 if removeHs: 

212 # Remove ester carbons (without hydrogens) 

213 for at in carbon_atoms: 

214 if 'H' not in at.bonded_atoms.elements: 

215 visited.add(at.index) 

216 carbon_groups = [] 

217 # Find all distinct groups of connected carbons 

218 for carbon in carbon_atoms: 

219 if carbon.index not in visited: 

220 # Get all carbons connected to this one 

221 connected_carbons = explore_carbon_group(carbon, visited) 

222 # Skip rings; acyl chains are acyclic carbon chains 

223 if ring_atoms.intersection(connected_carbons): 

224 continue 

225 if len(connected_carbons) >= min_length: # Only add non-empty groups 

226 carbon_groups.append(sorted(connected_carbons)) 

227 return carbon_groups

230def get_acyl_chain_atom_names( 

231 residue: 'MDAnalysis.Residue', 

232 removeHs: bool = True, 

233 sort_key=None, 

234) -> list[list[str]]: 

235 """Get atom names for each acyl chain in a lipid residue. 

236 

237 Returns: 

238 List of lists of atom name strings, one per chain. 

239 

240 """ 

241 chains = get_all_acyl_chains(residue, removeHs=removeHs) 

242 result = [] 

243 for chain in chains: 

244 names = list(residue.universe.atoms[sorted(chain)].names) 

245 if sort_key is not None: 

246 names = sorted(names, key=sort_key) 

247 result.append(names) 

248 return result

251def get_tail_atom_group(residue: 'MDAnalysis.Residue', include_hydrogens: bool = True) -> 'MDAnalysis.AtomGroup': 

252 """Get the tail (acyl chain) AtomGroup for a lipid residue. 

253 

254 Finds connected carbon chains via `get_all_acyl_chains` and optionally 

255 includes bonded hydrogen atoms. 

256 

257 Args: 

258 residue: An MDAnalysis Residue representing a single lipid molecule. 

259 include_hydrogens: If True, hydrogen atoms bonded to tail carbons are 

260 included in the returned AtomGroup. 

261 

262 Returns: 

263 AtomGroup containing the tail atoms of the lipid. 

264 

265 """ 

266 chains = get_all_acyl_chains(residue) 

267 tail_idx = [idx for chain in chains for idx in chain] 

268 if not tail_idx: 

269 return residue.atoms[[]] # empty AtomGroup 

270 tail_ag = residue.universe.atoms[tail_idx] 

271 if include_hydrogens: 

272 h_indices = [ 

273 bonded.index 

274 for atom in tail_ag.atoms 

275 for bonded in atom.bonded_atoms 

276 if bonded.name.startswith('H') 

277 ] 

278 if h_indices: 

279 tail_ag = tail_ag | residue.universe.atoms[h_indices] 

280 return tail_ag
