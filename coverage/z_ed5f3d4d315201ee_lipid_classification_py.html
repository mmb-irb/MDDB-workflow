1"""Classify lipids from their structure following the LIPID MAPS categories. 

2 

3https://lipidmaps.org/resources/education/classification 

4 

5Each category is recognised by SMARTS of its core structure and, where it makes sense, 

6the hydrophobic chains attached to it, so drugs and other small molecules are not lipids: 

7 

8- GP glycerophospholipids: glycerol phosphate with acyl/alkyl chains. 

9- SP sphingolipids: sphingoid base (2-amino-1,3-diol with a long chain). 

10- GL glycerolipids: glycerol with acyl/alkyl chains. 

11- ST sterol lipids: steroid nucleus or secosteroid (vitamin D) core with an aliphatic side chain in the 

12 D ring, as in sterols, bile acids and vitamin D. Steroid hormones and steroid drugs are not lipids here. 

13- SL saccharolipids: sugars with acyl chains directly attached. 

14- PR prenol lipids: chains of 3+ head to tail isoprene units and retinoids/carotenoids with long chains. 

15- FA fatty acyls: a long hydrophobic chain making a large part of the molecule, two long chains 

16 (synthetic double tailed lipids, e.g. ionizable lipids) or the prostanoid skeleton. 

17- PK polyketides are not recognised: they share no core structure (flavonoids, macrolides, 

18 tetracyclines...) and they are drugs or natural products rather than membrane lipids. 

19 

20Peptides with acyl chains (lipopeptide antibiotics) are not saccharolipids nor fatty acyls. 

21 

22SMARTS use generic bonds and no hydrogen counts since the molecules come from MD structures, 

23whose InChIs may have unusual tautomers (e.g. amides as imidic acids) or charges. 

24"""

26from rdkit import Chem, rdBase

48# Atoms in rings larger than this are considered part of chains (e.g. archaeal macrocyclic tetraethers) 

49MAX_RING_SIZE = 30 

50# Minimum hydrophobic chain length (carbons only bonded to carbons) of the chains attached to lipid cores 

51# Hexanoyl chains have 5 of these carbons since the carbonyl carbon is not counted (e.g. DHPC) 

52MIN_CORE_CHAIN = 4 

53# Minimum hydrophobic chain length of the sphingoid base 

54MIN_SPHINGOID_CHAIN = 8 

55# Minimum hydrophobic chain length of the steroid side chain in the D ring, 0 to accept any steroid 

56# Bile acids have 4 and cholesterol 7, while steroid hormones have no side chain (estradiol, testosterone) 

57# or it has 1 carbon (progesterone, cortisol), so they are left out since they do not form membranes 

58MIN_STEROID_SIDE_CHAIN = 4 

59# Minimum hydrophobic chain length of fatty acyls (lauric acid has 11) 

60MIN_FATTY_CHAIN = 10 

61# Minimum fraction of heavy atoms in hydrophobic chains for a single chain fatty acyl 

62# Polyethylene glycol atoms are not counted since it is a polymeric head (e.g. PEG-lipids) 

63MIN_CHAIN_FRACTION = 0.3 

64# Molecules with this many peptide bonds are peptides 

65MIN_PEPTIDE_BONDS = 2 

66 

67# Carbon not in small rings 

68_C = f'[#6;!r{{3-{MAX_RING_SIZE}}}]' 

69# Oxygen not in small rings 

70_O = f'[#8;!r{{3-{MAX_RING_SIZE}}}]' 

71# Carbons with 2 and 3 heavy atoms bonded (CH2 and CH when they are saturated) 

72_CH2 = f'[#6;D2;!r{{3-{MAX_RING_SIZE}}}]' 

73_CH = f'[#6;D3;!r{{3-{MAX_RING_SIZE}}}]' 

74SMARTS = { 

75 # Glycerol phosphate (glycerol backbone in the sn-3 position, sn-1 in archaea) and glycerol backbone 

76 'glycerophosphate': f'[#15](~[#8])(~[#8])(~[#8])-{_O}-{_CH2}-{_CH}(-{_O})-{_CH2}-{_O}', 

77 'glycerol': f'{_CH2}(-{_O})-{_CH}(-{_O})-{_CH2}-{_O}', 

78 # Sphingoid base: CH2(O)-CH(N)-CH(O)-C (1-hydroxy may be phosphorylated or glycosylated, 3-hydroxy may be oxo) 

79 'sphingoid': f'{_CH2}(-[#8])-{_CH}(-[#7;!r{{3-{MAX_RING_SIZE}}}])-{_C}(~{_O})-{_C}', 

80 # Steroid nucleus (gonane): fused 6-6-6-5 rings 

81 'steroid': '[#6]1~[#6]~[#6]~[#6]2~[#6](~[#6]~1)~[#6]~[#6]~[#6]1~[#6]~2~[#6]~[#6]~[#6]2~[#6]~[#6]~[#6]~[#6]~1~2', 

82 # Secosteroid (vitamin D): A ring linked by a diene to the CD rings 

83 'secosteroid': '[#6]1~[#6]~[#6]~[#6]~[#6]~[#6]~1~[#6;!R]~[#6;!R]~[#6]1~[#6]~[#6]~[#6]~[#6]2~[#6]~[#6]~[#6]~[#6]~1~2', 

84 # Pyranose and furanose rings 

85 'pyranose': '[#6;R]1~[#6;R]~[#6;R]~[#6;R]~[#6;R]~[#8;R]~1', 

86 'furanose': '[#6;R]1~[#6;R]~[#6;R]~[#6;R]~[#8;R]~1', 

87 # Three head to tail isoprene units: methyl branches every 4 carbons, with unbranched carbons between them 

88 'isoprenoid': f'{_CH}(-[#6;D1])~{_CH2}~{_CH2}~{_CH2}~{_CH}(-[#6;D1])~{_CH2}~{_CH2}~{_CH2}~{_CH}(-[#6;D1])', 

89 # Retinoids and carotenoids: 1,1,5-trimethylcyclohexene with a polyene chain with methyl branches 

90 'retinoid': '[#6;D1]-[#6;R]1(-[#6;D1])~[#6;R]~[#6;R]~[#6;R]~[#6;R](-[#6;D1])~[#6;R]~1-[#6;!R]~[#6;!R]~[#6;!R](-[#6;D1])~[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R](-[#6;D1])', 

91 # Prostanoid skeleton: cyclopentane with two chains in adjacent carbons, one ending in a carboxyl 

92 'prostanoid': '[#6;r5]1(-[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R](~[#8])~[#8])~[#6;r5](-[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R]~[#6;!R])~[#6;r5]~[#6;r5]~[#6;r5]~1', 

93 # Polyethylene glycol unit 

94 'peg': f'{_O}-{_CH2}-{_CH2}-{_O}', 

95 # Peptide bond between two amino acids: N-CA-C(=O)-N 

96 'peptide': '[#7]-[#6;X4]-[#6;X3](~[#8])~[#7]-[#6;X4]', 

97} 

98PATTERNS = {name: Chem.MolFromSmarts(smarts) for name, smarts in SMARTS.items()}

101def inchi_to_mol(inchi: str) -> Optional['Chem.Mol']: 

102 """Build the RDKit molecule of an InChI, keeping only its largest component (e.g. no counter ions).""" 

103 try: 

104 # Block the RDKit InChI warnings 

105 with rdBase.BlockLogs(): 

106 mol = Chem.MolFromInchi(inchi) 

107 except Exception: 

108 return None 

109 if mol is None or mol.GetNumAtoms() == 0: return None 

110 fragments = Chem.GetMolFrags(mol, asMols=True) 

111 return max(fragments, key=lambda fragment: fragment.GetNumHeavyAtoms())

114class HydrophobicChains: 

115 """Hydrophobic chains of a molecule: connected carbons out of small rings which are only bonded to carbons. 

116 

117 Carbons bonded to heteroatoms (e.g. carbonyl, glycerol or polyethylene glycol carbons) break the chains, 

118 so a hexadecanoyl chain has 15 carbons. 

119 """ 

120 

121 def __init__(self, mol: 'Chem.Mol'): 

122 self.mol = mol 

123 ring_info = mol.GetRingInfo() 

124 self.small_ring_atoms = {atom.GetIdx() for atom in mol.GetAtoms() 

125 if 0 < ring_info.MinAtomRingSize(atom.GetIdx()) <= MAX_RING_SIZE} 

126 self.atoms = {atom.GetIdx() for atom in mol.GetAtoms() if atom.GetAtomicNum() == 6 

127 and atom.GetIdx() not in self.small_ring_atoms 

128 and all(neighbor.GetAtomicNum() in (1, 6) for neighbor in atom.GetNeighbors())} 

129 self.neighbors = {index: [neighbor.GetIdx() for neighbor in mol.GetAtomWithIdx(index).GetNeighbors() 

130 if neighbor.GetIdx() in self.atoms] for index in self.atoms} 

131 # Chain (connected component) of each atom, the length of the longest path and the size of each chain 

132 self.chain_of = {} 

133 self.lengths = [] 

134 self.sizes = [] 

135 for index in self.atoms: 

136 if index in self.chain_of: continue 

137 farthest, _, component = self._bfs(index) 

138 _, length, _ = self._bfs(farthest) 

139 for atom in component: self.chain_of[atom] = len(self.lengths) 

140 self.lengths.append(length) 

141 self.sizes.append(len(component)) 

142 

143 def _bfs(self, start: int) -> tuple[int, int, set[int]]: 

144 """Get the farthest atom from the start atom, its distance in atoms and the visited atoms.""" 

145 distances = {start: 1} 

146 queue = deque([start]) 

147 farthest = start 

148 while queue: 

149 current = queue.popleft() 

150 for neighbor in self.neighbors[current]: 

151 if neighbor in distances: continue 

152 distances[neighbor] = distances[current] + 1 

153 if distances[neighbor] > distances[farthest]: farthest = neighbor 

154 queue.append(neighbor) 

155 return farthest, distances[farthest], set(distances) 

156 

157 @property 

158 def longest(self) -> int: 

159 """Length of the longest chain.""" 

160 return max(self.lengths, default=0) 

161 

162 def count_longer_than(self, min_length: int) -> int: 

163 """Count the chains with at least the given length.""" 

164 return sum(length >= min_length for length in self.lengths) 

165 

166 def substituent_length(self, start: int, excluded: set[int]) -> int: 

167 """Get the longest chain in the substituent starting at the given atom. 

168 The substituent is made of the carbons out of small rings connected to the start atom, without crossing 

169 heteroatoms nor the excluded atoms (e.g. the lipid core), so it includes carbonyl and hydroxylated carbons. 

170 """ 

171 if start in excluded: return 0 

172 visited = {start} 

173 queue = deque([start]) 

174 length = 0 

175 while queue: 

176 current = queue.popleft() 

177 atom = self.mol.GetAtomWithIdx(current) 

178 if atom.GetAtomicNum() != 6 or current in self.small_ring_atoms: continue 

179 if current in self.chain_of: 

180 length = max(length, self.lengths[self.chain_of[current]]) 

181 for neighbor in atom.GetNeighbors(): 

182 index = neighbor.GetIdx() 

183 if index in visited or index in excluded: continue 

184 visited.add(index) 

185 queue.append(index) 

186 return length 

187 

188 def oxygen_substituent_length(self, oxygen: int, excluded: set[int]) -> int: 

189 """Get the longest chain linked to an oxygen (or nitrogen), as in esters, ethers and amides.""" 

190 atom = self.mol.GetAtomWithIdx(oxygen) 

191 return max((self.substituent_length(neighbor.GetIdx(), excluded) for neighbor in atom.GetNeighbors() 

192 if neighbor.GetAtomicNum() == 6 and neighbor.GetIdx() not in excluded), default=0)

195def classify_lipid(inchi: str) -> Optional[str]: 

196 """Get the LIPID MAPS category of the molecule of an InChI, or None if it is not a lipid.""" 

197 mol = inchi_to_mol(inchi) 

198 if mol is None: return None 

199 chains = HydrophobicChains(mol) 

200 if has_acylated_glycerol(mol, chains, PATTERNS['glycerophosphate']): 

201 return GLYCEROPHOSPHOLIPIDS 

202 if has_sphingoid_base(mol, chains): 

203 return SPHINGOLIPIDS 

204 if has_acylated_glycerol(mol, chains, PATTERNS['glycerol']): 

205 return GLYCEROLIPIDS 

206 if has_sterol_core(mol, chains): 

207 return STEROL_LIPIDS 

208 if mol.HasSubstructMatch(PATTERNS['isoprenoid']) or ( 

209 mol.HasSubstructMatch(PATTERNS['retinoid']) and chains.longest >= MIN_FATTY_CHAIN): 

210 return PRENOL_LIPIDS 

211 if len(mol.GetSubstructMatches(PATTERNS['peptide'])) >= MIN_PEPTIDE_BONDS: 

212 return None 

213 if has_acylated_sugar(mol, chains): 

214 return SACCHAROLIPIDS 

215 if is_fatty_acyl(mol, chains): 

216 return FATTY_ACYLS 

217 return None

220def has_acylated_glycerol(mol: 'Chem.Mol', chains: HydrophobicChains, pattern: 'Chem.Mol') -> bool: 

221 """Check if a glycerol backbone has an oxygen linked to a chain (acyl, alkyl or alkenyl).""" 

222 for match in mol.GetSubstructMatches(pattern): 

223 # The phosphate oxygens are not part of the glycerol 

224 oxygens = [index for index in match if mol.GetAtomWithIdx(index).GetAtomicNum() == 8][-3:] 

225 if any(chains.oxygen_substituent_length(oxygen, set(match)) >= MIN_CORE_CHAIN for oxygen in oxygens): 

226 return True 

227 return False

230def has_sphingoid_base(mol: 'Chem.Mol', chains: HydrophobicChains) -> bool: 

231 """Check if there is a sphingoid base with its long chain.""" 

232 for match in mol.GetSubstructMatches(PATTERNS['sphingoid']): 

233 if chains.substituent_length(match[-1], set(match[:-1])) >= MIN_SPHINGOID_CHAIN: 

234 return True 

235 return False

238def has_sterol_core(mol: 'Chem.Mol', chains: HydrophobicChains) -> bool: 

239 """Check if there is a steroid or secosteroid core with the minimum side chain in the D ring, if any.""" 

240 for pattern in [PATTERNS['steroid'], PATTERNS['secosteroid']]: 

241 for match in mol.GetSubstructMatches(pattern): 

242 core = set(match) 

243 # The last 5 atoms of both patterns are the D ring 

244 for ring_atom in match[-5:]: 

245 for neighbor in mol.GetAtomWithIdx(ring_atom).GetNeighbors(): 

246 if chains.substituent_length(neighbor.GetIdx(), core) >= MIN_STEROID_SIDE_CHAIN: 

247 return True 

248 return False

251def has_acylated_sugar(mol: 'Chem.Mol', chains: HydrophobicChains) -> bool: 

252 """Check if a sugar ring has a fatty acid directly linked as ester or amide.""" 

253 sugar_rings = mol.GetSubstructMatches(PATTERNS['pyranose']) + mol.GetSubstructMatches(PATTERNS['furanose']) 

254 for ring in sugar_rings: 

255 ring_atoms = set(ring) 

256 # Ring carbons and their exocyclic carbons (e.g. C6 of hexoses) 

257 carbons = {index for index in ring if mol.GetAtomWithIdx(index).GetAtomicNum() == 6} 

258 carbons |= {neighbor.GetIdx() for index in list(carbons) for neighbor in mol.GetAtomWithIdx(index).GetNeighbors() 

259 if neighbor.GetAtomicNum() == 6 and not neighbor.IsInRing()} 

260 for carbon in carbons: 

261 for heteroatom in mol.GetAtomWithIdx(carbon).GetNeighbors(): 

262 if heteroatom.GetAtomicNum() not in (7, 8) or heteroatom.GetIdx() in ring_atoms: continue 

263 for acyl in heteroatom.GetNeighbors(): 

264 if acyl.GetIdx() == carbon or not is_acyl_carbon(acyl): continue 

265 if chains.substituent_length(acyl.GetIdx(), ring_atoms | {carbon, heteroatom.GetIdx()}) >= MIN_CORE_CHAIN: 

266 return True 

267 return False

270def is_acyl_carbon(atom: 'Chem.Atom') -> bool: 

271 """Check if a carbon is a carbonyl carbon of an ester or amide (also as imidic acid tautomer).""" 

272 if atom.GetAtomicNum() != 6 or atom.IsInRing(): return False 

273 heteroatoms = [neighbor for neighbor in atom.GetNeighbors() if neighbor.GetAtomicNum() in (7, 8)] 

274 return len(heteroatoms) >= 2 and any(neighbor.GetAtomicNum() == 8 for neighbor in heteroatoms)

277def is_fatty_acyl(mol: 'Chem.Mol', chains: HydrophobicChains) -> bool: 

278 """Check if the molecule is a fatty acyl: a long hydrophobic chain making a large part of the molecule, 

279 two long chains (e.g. synthetic ionizable and cationic lipids) or the prostanoid skeleton. 

280 """ 

281 if chains.count_longer_than(MIN_FATTY_CHAIN) >= 2: 

282 return True 

283 if mol.HasSubstructMatch(PATTERNS['prostanoid']): 

284 return True 

285 if chains.longest < MIN_FATTY_CHAIN: 

286 return False 

287 peg_atoms = {index for match in mol.GetSubstructMatches(PATTERNS['peg']) for index in match} 

288 chain_atoms = sum(size for size, length in zip(chains.sizes, chains.lengths) if length >= MIN_CORE_CHAIN) 

289 return chain_atoms / (mol.GetNumHeavyAtoms() - len(peg_atoms)) >= MIN_CHAIN_FRACTION
