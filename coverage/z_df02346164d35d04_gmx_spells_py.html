25def run_gromacs( 

26 command : str, 

27 user_input : Optional[str] = None, 

28 expected_output_filepath : Optional[str] = 'auto', 

29 show_output_logs : bool = False, 

30 show_error_logs : bool = False) -> tuple[str, str]: 

31 """ Set a function to call gromacs in a more confortable and standarized way: 

32 - Standard gromacs executable, which may be provided by the user 

33 - Gromacs mass fixes by using a custom atommass.dat file with extended atom names 

34 - Hidden unnecessary output logs and grey-colored necessary ones 

35 - Missing output checks 

36 Then return both output and error logs. """ 

37 

38 # Set the working directory as the path for GROMACS where to look for supplementary data files 

39 # Here we will make a copy of the atommass.dat file 

40 environ["GMXLIB"] = GLOBALS["working_directory"] 

41 

42 # Run a fix for gromacs masses if not done before 

43 fix_gromacs_masses() 

44 

45 # In case we have user input we must open a process to then pipe it in the gromacs process 

46 if user_input: 

47 # The -e option allows the interpretation of '\', which is critial for the make_ndx command 

48 # WARNING: Explicity split by whitespaces 

49 # WARNING: If not specified then breaklines are also removed, which is ciritcal for make_ndx 

50 user_input_process = Popen([ "echo", "-e", *user_input.split(' ') ], stdout=PIPE) 

51 

52 # Get the time at the moment we start to run the command 

53 # WARNING: We must truncate this value, because the file mtime is truncated as well 

54 # Otherwise it may happen that gromacs runs in less than 1 seconds and the start time is higher 

55 start_time = time().__trunc__() 

56 

57 # If error is to be shown the color it in grey 

58 # This is usually used to see progress logs 

59 if show_error_logs: print(GREY_HEADER) 

60 

61 # Set the gromacs process 

62 process = run([ GROMACS_EXECUTABLE, *command.split(), '-quiet' ], 

63 stdin = user_input_process.stdout if user_input else None, 

64 stdout = PIPE, stderr = PIPE if not show_error_logs else None) 

65 

66 # Get the gromacs process output 

67 output_logs = process.stdout.decode() 

68 # Get also error logs if necessary 

69 error_logs = process.stderr.decode() if not show_error_logs else None 

70 

71 # End the grey coloring 

72 if show_error_logs: print(COLOR_END) 

73 

74 # In case the expected output is set as 'auto' we must guess it from the command 

75 # Normally output comes after the '-o' option 

76 if expected_output_filepath == 'auto': 

77 command_splits = command.split() 

78 if '-o' not in command_splits: expected_output_filepath = None 

79 else: 

80 option_flag_index = command_splits.index('-o') 

81 expected_output_filepath = command_splits[option_flag_index + 1] 

82 

83 # If an output file was expected then check it was created 

84 if expected_output_filepath: 

85 # Make sure output exists 

86 output_exists = exists(expected_output_filepath) 

87 # Make sure output have been generated by the gromacs command and it is not an old output 

88 output_is_old = False 

89 output_is_empty = False 

90 if output_exists: 

91 output_time = getmtime(expected_output_filepath) 

92 output_is_old = output_time < start_time 

93 output_is_empty = getsize(expected_output_filepath) == 0 

94 # Gromacs may create the output file and then fail, leaving it empty or incomplete 

95 gromacs_failed = process.returncode != 0 

96 # If output was not generated then we report the problem 

97 if not output_exists or output_is_old or output_is_empty or gromacs_failed: 

98 # Remove the broken output generated by this command, so it is not reused in further runs 

99 if output_exists and not output_is_old: 

100 remove(expected_output_filepath) 

101 # If we recognize the error and can provide any clue then this is the moment 

102 error_match = findall(GROMACS_MASSES_ERROR, error_logs) if error_logs else None 

103 if error_match: 

104 warn(f'We are missing the masses for some atoms: {", ".join(error_match)}\n' + \ 

105 ' Are they dummy atoms?\n' + \ 

106 ' If so, please select them using the "dummy_selection" input field.\n' + \ 

107 ' Dummy atoms will be considered to have mass = 0.\n' + \ 

108 ' Are they Coarse Grain (CG) beads?\n' + \ 

109 ' If so, please select them using the "cg_selection" input field.\n' + \ 

110 ' Coarse Grain beads will be considered to have mass = 1.\n' + \ 

111 f' If you want to set custom masses for these atoms then please edit the {GROMACS_MASSES_FILENAME} file in the project directory.') 

112 # If we don't know the error then simply output all the logs 

113 else: 

114 print(output_logs) 

115 print(error_logs) 

116 # Recreate the exact command 

117 final_command = f'{GROMACS_EXECUTABLE} {command}' 

118 if user_input: final_command += f' (with user input "{user_input}")' 

119 if gromacs_failed: reason = f'exit code {process.returncode}' 

120 elif output_is_empty: reason = 'empty output file' 

121 else: reason = 'missing output file' 

122 raise ToolError(f'Something went wrong with Gromacs ({reason}) while running "{final_command}"') 

123 

124 # If all was good then show final logs but only if it was requested 

125 if show_output_logs: print(output_logs) 

126 

127 # Return outputs 

128 return output_logs, error_logs

419def get_tpr_charges (tpr_filepath : str) -> list[float]: 

420 """ Get tpr atom charges. 

421 This works for the new tpr format (tested in 122). """ 

422 # Read the TPR 

423 tpr_content, tpr_error_logs = get_tpr_content(tpr_filepath) 

424 # Mine the atomic charges 

425 charges = [] 

426 # Iterate tpr content lines 

427 for line in tpr_content.split('\n'): 

428 # Skip everything which is not atomic charges data 

429 if line[0:16] != ' atom': continue 

430 # Parse the line to get only charges 

431 match = search(GROMACS_TPR_ATOM_CHARGES_REGEX, line) 

432 if match: charges.append(float(match[1])) 

433 # If we successfully got atom charges then return them 

434 if len(charges) > 0: return charges 

435 # If there are no charges at the end then something went wrong 

436 print(tpr_error_logs) 

437 raise RuntimeError(f'Charges extraction from tpr file "{tpr_filepath}" has failed')

439# Regular expresion to mine atom bonds 

440GROMACS_TPR_ATOM_BONDS_REGEX = r"^\s*([0-9]*) type=[0-9]* \((BONDS|CONSTR|CONNBONDS)\)\s*([0-9]*)\s*([0-9]*)$" 

441# Set a regular expression for SETTLE bonds, used for rigid waters 

442# ---- From the paper ------------------------------------------------------------------------------------ 

443# The SETTLE can be applied to a four-point water model like TIP4P5 which has the fourth point with 

444# a certain charge and no mass if the force acting on the fourth point is distributed onto the other three 

445# points with masses in a reasonable manner. 

446# S. Miyamoto and P.A. Kollman, “SETTLE: An analytical version of the SHAKE and RATTLE algorithms for rigid 

447# water models,” J. Comp. Chem., 13 952–962 (1992) 

448# -------------------------------------------------------------------------------------------------------- 

449# So it may happen that we encounter SETTLE with 4 atoms 

450# This is not yet supported, but at least we check if this is happening to raise an error when found 

451# NOTE: Amber topologies (prmtop) have a similar issue with SHAKE 

452# Rigid waters (e.g. TIP3P) include a H-H bond to keep the geometry rigid, which is not a covalent bond 

453# Bonds mined through pytraj (tools/get_bonds.py) keep it, so the stabonds/cohbonds tests fail since it is 

454# never found when guessing bonds from coordinates (e.g. PLAS-20K, where 2 waters are kept in the system) 

455# Until this is handled, filter the water (-filt) or discard H-H bonds when mining prmtop bonds 

456GROMACS_TPR_SETTLE_REGEX = r"^\s*([0-9]*) type=[0-9]* \(SETTLE\)\s*([0-9]*)\s*([0-9]*)\s*([0-9]*)\s*([0-9]*)$"

461def get_tpr_bonds (tpr_filepath : str) -> list[ tuple[int, int] ]: 

462 """ Get tpr atom bonds. """ 

463 # Read and parse the TPR 

464 parsed_tpr = read_and_parse_tpr(tpr_filepath) 

465 topology = parsed_tpr['gromacs-topology'] 

466 # Get the actual bonds 

467 tpr_bonds = topology['bond']['value'] 

468 # Get virtual sites with bond-like annotations 

469 # These are not actual bonds, but some CG models such as Martini use them as such 

470 # If not consider, some beads may remain non-bonded to anything thus breaking our logic 

471 # DANI: I check for the field to exist just to support old versions of the tool 

472 for virtual_size_field in VIRTUAL_SITE_FIELDS: 

473 virtuals = topology.get(virtual_size_field, None) 

474 if not virtuals: continue 

475 # Add a bond between the first atom index and each of the rest of atom indices 

476 for virtual in virtuals['value']: 

477 main = virtual[0] 

478 for other in virtual[1:]: 

479 fake_bond = [main, other] 

480 tpr_bonds.append(fake_bond) 

481 return tpr_bonds

529def generate_frames_ndx (frames : list[int], filename : str): 

530 """ Generate a ndx file with a selection of frames. """ 

531 # Add a header 

532 content = '[ frames ]\n' 

533 count = 0 

534 for frame in frames: 

535 # Add a breakline each 15 indices 

536 count += 1 

537 if count == 15: 

538 content += '\n'

721# ---- XTC low-level utilities (Claude) ------------------------------------------------------------------ 

722# 

723# XTC is the GROMACS compressed trajectory format. Each frame is an XDR 

724# (big-endian) record with the following layout: 

725# 

726# int32 magic (always 1995) 

727# int32 natoms 

728# int32 step 

729# float32 time 

730# float32[9] box (3×3 matrix, nm, row-major — diagonal = Lx, Ly, Lz) 

731# int32 natoms (repeated) 

732# -- if natoms <= 9 the coordinates are not compressed: -- 

733# float32[3*natoms] coordinates 

734# -- otherwise: -- 

735# float32 precision 

736# int32[3] minint 

737# int32[3] maxint 

738# int32 smallidx 

739# int32 ncoord_bytes (byte count of the compressed coordinate block) 

740# bytes[ceil(ncoord_bytes/4)*4] (3dfcoord-compressed integers, XDR-padded) 

741# 

742# Because ncoord_bytes is stored in the header we can skip the compressed 

743# block entirely and jump straight to the next frame. This avoids the full 

744# 3dfcoord decompression and yields ~50× speed-up over mdtraj/pytraj when 

745# only box dimensions or a frame count are needed. 

746# 

747# Note: ncoord_bytes is frame-specific (depends on coordinate range) so 

748# frame sizes are not constant in general. 

749# -------------------------------------------------------------------------------------------------------- 

750 

751_XTC_MAGIC = 1995 

752# Byte offsets within a frame (after the 4-byte magic): 

753# natoms(4) + step(4) + time(4) + box(36) = 48 → box starts at byte 12 of this block 

754_XTC_PRE_BOX = 48 # bytes to read after magic: covers natoms/step/time/box 

755_XTC_BOX_OFFSET = 12 # offset of box within that 48-byte block 

756# Compressed block header before ncoord_bytes: precision(4) + minint(12) + maxint(12) + smallidx(4) 

757_XTC_COMPRESSED_HEADER = 32 

758# Systems with this many atoms or less store their coordinates without compression 

759_XTC_MAX_UNCOMPRESSED_ATOMS = 9 

760# XDR integers are 32 bits big-endian 

761_XTC_INT = struct.Struct('>i') 

762 

763def _skip_xtc_coordinates (f) -> bool: 

764 """Move the file cursor from right after the box to the start of the next frame. 

765 Return False if the file ends before. 

766 

767 Made by Claude. 

768 """ 

769 raw = f.read(4) 

770 if len(raw) < 4: return False 

771 natoms = struct.unpack('>i', raw)[0] 

772 if natoms <= _XTC_MAX_UNCOMPRESSED_ATOMS: 

773 f.seek(3 * 4 * natoms, 1) 

774 return True 

775 f.seek(_XTC_COMPRESSED_HEADER, 1) 

776 raw = f.read(4) 

777 if len(raw) < 4: return False 

778 ncoord_bytes = struct.unpack('>i', raw)[0] 

779 f.seek(((ncoord_bytes + 3) // 4) * 4, 1) 

780 return True

782def get_xtc_frame_count(xtc_path: str, verbose: bool = True) -> int: 

783 """Count the number of frames in an XTC file without loading coordinates. 

784 

785 Scans the file by reading only the small per-frame header and skipping the 

786 compressed coordinate block via the stored byte-count field. 

787 ~3× faster than pytraj and requires no topology file. 

788 

789 Made by Claude. 

790 """ 

791 if verbose: print('-> Counting number of frames') 

792 count = 0 

793 magic_bytes = struct.pack('>i', _XTC_MAGIC) 

794 with open(xtc_path, 'rb') as f: 

795 while True: 

796 if f.read(4) != magic_bytes: 

797 break 

798 f.seek(_XTC_PRE_BOX, 1) 

799 if not _skip_xtc_coordinates(f): 

800 break 

801 count += 1 

802 if verbose: print(f' Frames: {count}') 

803 return count

806def count_xtc_frames (trajectory_file : 'File') -> int:

809# Byte offsets within a frame, counting from the magic number 

810_XTC_COORDINATES_OFFSET = 4 + _XTC_PRE_BOX + 4 # magic + header + repeated natoms 

811_XTC_NCOORD_BYTES_OFFSET = _XTC_COORDINATES_OFFSET + _XTC_COMPRESSED_HEADER

813def _iterate_xtc_frame_bounds (mapped_file : mmap.mmap) -> Generator[tuple[int, int], None, None]: 

814 """Yield the start and end byte of every frame in a memory mapped XTC file. 

815 Stop at the first incomplete or corrupted frame. 

816 

817 Made by Claude. 

818 """ 

819 size = len(mapped_file) 

820 start = 0 

821 while start + _XTC_COORDINATES_OFFSET <= size: 

822 if _XTC_INT.unpack_from(mapped_file, start)[0] != _XTC_MAGIC: return 

823 natoms = _XTC_INT.unpack_from(mapped_file, start + _XTC_COORDINATES_OFFSET - 4)[0] 

824 if natoms <= _XTC_MAX_UNCOMPRESSED_ATOMS: 

825 end = start + _XTC_COORDINATES_OFFSET + 3 * 4 * natoms 

826 else: 

827 if start + _XTC_NCOORD_BYTES_OFFSET + 4 > size: return 

828 ncoord_bytes = _XTC_INT.unpack_from(mapped_file, start + _XTC_NCOORD_BYTES_OFFSET)[0] 

829 end = start + _XTC_NCOORD_BYTES_OFFSET + 4 + ((ncoord_bytes + 3) // 4) * 4 

830 # Make sure the frame is complete, which may not be the case of the last frame 

831 if end > size: return 

832 yield start, end 

833 start = end

835def copy_xtc_frames (input_xtc_path : str, output_xtc_path : str, frames : Callable[[int], bool] | list[int]) -> int: 

836 """Write a new XTC file with a selection of frames from another XTC file. 

837 

838 Frames are copied byte by byte, without decompressing coordinates, so the output 

839 is identical to the one from 'gmx trjconv' but much faster (5-50× in our tests). 

840 Frames may be a list of frame indices (0-based) or a function which says if a given frame index is to be kept. 

841 Frames are always written in their original order. 

842 Input and output files may be the same. 

843 Return the number of written frames. 

844 

845 Made by Claude. 

846 """ 

847 # If frames are a list then set the selection function and the last frame to read 

848 last_frame = None 

849 if not callable(frames): 

850 selected_frames = set(frames) 

851 last_frame = max(selected_frames, default=-1) 

852 frames = selected_frames.__contains__ 

853 # Write to an auxiliar file first in case the input and output files are the same 

854 # Keep it in the same directory so it can be renamed 

855 auxiliar_output_path = join(dirname(output_xtc_path), f'.{basename(output_xtc_path)}.incomplete') 

856 written_frames = 0 

857 # Count the complete frames we read and the byte where the last of them ends 

858 read_frames = 0 

859 end = 0 

860 with open(input_xtc_path, 'rb') as input_file, open(auxiliar_output_path, 'wb') as output_file: 

861 size = fstat(input_file.fileno()).st_size 

862 # Note that empty files can not be memory mapped 

863 if size > 0: 

864 with mmap.mmap(input_file.fileno(), 0, access=mmap.ACCESS_READ) as mapped_file: 

865 for frame, (start, end) in enumerate(_iterate_xtc_frame_bounds(mapped_file)): 

866 if last_frame is not None and frame > last_frame: break 

867 read_frames += 1 

868 if frames(frame): 

869 output_file.write(mapped_file[start:end]) 

870 written_frames += 1 

871 # If we read the whole trajectory then there must be no remaining bytes 

872 else: 

873 if end < size: 

874 remove(auxiliar_output_path) 

875 raise RuntimeError(f'Trajectory {input_xtc_path} is corrupted: ' 

876 f'after {read_frames} complete frames (byte {end}) there is an incomplete or not valid frame ' 

877 f'({size - end} remaining bytes). This may happen if the simulation or a file transfer was interrupted.') 

878 rename(auxiliar_output_path, output_xtc_path) 

879 return written_frames

881# ---- Box shape classification (Claude) ----------------------------------------------------------------- 

882# 

883# The same periodic box can be written with different sets of box vectors. 

884# e.g. AMBER writes the truncated octahedron with three 109.47° angles while 

885# GROMACS writes it with 70.53°, 109.47° and 70.53°, yet both are the same box. 

886# Thus comparing the box vectors against templates is not reliable. 

887# 

888# Instead we look at the periodic translations of the box: every integer 

889# combination of box vectors moves the system onto one of its periodic images. 

890# The three shortest translations which do not lie in the same line or plane 

891# are the same whatever the box vectors we were given, so their lengths and 

892# angles characterise the box shape. 

893# Angles are folded into [0°, 90°] since a translation and its opposite are 

894# equivalent (e.g. 60° and 120° both count as 60°). 

895# 

896# Expected shortest translations for every shape: 

897# cubic — 3 equal lengths, angles 90° 90° 90° 

898# orthogonal — any lengths, angles 90° 90° 90° 

899# hexagonal — 2 equal lengths at 60°, the third one perpendicular to both 

900# dodecahedral — 3 equal lengths, angles 60° 60° 60° or 60° 60° 90° 

901# octahedral — 3 equal lengths, angles 70.53° 70.53° 70.53° (truncated octahedron) 

902# triclinic — anything else 

903# -------------------------------------------------------------------------------------------------------- 

904 

905# Integer combinations of box vectors used to enumerate periodic translations 

906# Coefficients up to 2 are enough for any reasonably shaped box (e.g. those written by MD engines) 

907# Tested also with boxes rewritten through random combinations of their vectors 

908_TRANSLATION_COEFFICIENTS = np.array( 

909 [(i, j, k) for i in range(-2, 3) for j in range(-2, 3) for k in range(-2, 3) if (i, j, k) != (0, 0, 0)], 

910 dtype=np.float64) 

911# Tolerances when comparing lengths (relative) and angles (degrees) 

912# They absorb float32 precision and the small fluctuations of isotropic pressure coupling 

913_BOX_SHAPE_LENGTH_TOLERANCE = 0.01 

914_BOX_SHAPE_ANGLE_TOLERANCE = 1.0 

915# Angle between the shortest translations of a truncated octahedron: acos(1/3) 

916_OCTAHEDRAL_ANGLE = float(np.degrees(np.arccos(1 / 3))) 

917 

918# Number of frames classified at once, to limit memory usage (~50 MB per chunk) 

919_BOX_SHAPE_CHUNK_SIZE = 2000

921def get_box_shapes (box_matrices : np.ndarray) -> list[str]: 

922 """Classify the shape of many periodic boxes given their box matrices (n_boxes, 3, 3), rows are box vectors. 

923 

924 Returns a list with one of 'cubic', 'orthogonal', 'hexagonal', 'dodecahedral', 'octahedral' 

925 or 'triclinic' for every box. See the comment block above for the logic. 

926 All boxes are processed at once with numpy operations since classifying boxes one by one 

927 would be too slow for long trajectories with a dynamic box. 

928 

929 Made by Claude. 

930 """ 

931 box_matrices = np.asarray(box_matrices, dtype=np.float64).reshape(-1, 3, 3) 

932 shapes = [] 

933 for start in range(0, len(box_matrices), _BOX_SHAPE_CHUNK_SIZE): 

934 shapes += _classify_box_shapes(box_matrices[start : start + _BOX_SHAPE_CHUNK_SIZE]) 

935 return shapes

944def _classify_box_shapes (box_matrices : np.ndarray) -> list[str]: 

945 """Classify a chunk of box matrices (n_boxes, 3, 3). Used by get_box_shapes. 

946 

947 Made by Claude. 

948 """ 

949 n_boxes = len(box_matrices) 

950 rows = np.arange(n_boxes) 

951 # All periodic translations of every box and their lengths 

952 translations = np.einsum('tj,bjk->btk', _TRANSLATION_COEFFICIENTS, box_matrices) # (n_boxes, n_translations, 3) 

953 lengths = np.linalg.norm(translations, axis=2) # (n_boxes, n_translations) 

954 # Pick the shortest translation, then the shortest one not in its line, 

955 # then the shortest one not in the plane of the previous two 

956 # A small relative threshold makes float precision errors not count as out of line/plane 

957 first_index = np.argmin(lengths, axis=1) 

958 first = translations[rows, first_index] 

959 first_length = lengths[rows, first_index] 

960 out_of_line = np.linalg.norm(np.cross(first[:, np.newaxis, :], translations), axis=2) \ 

961 > 1e-4 * first_length[:, np.newaxis] * lengths 

962 second = translations[rows, np.argmin(np.where(out_of_line, lengths, np.inf), axis=1)] 

963 normal = np.cross(first, second) 

964 out_of_plane = np.abs(np.einsum('btk,bk->bt', translations, normal)) \ 

965 > 1e-4 * np.linalg.norm(normal, axis=1)[:, np.newaxis] * lengths 

966 third = translations[rows, np.argmin(np.where(out_of_plane, lengths, np.inf), axis=1)] 

967 shortest = np.stack([first, second, third], axis=1) # (n_boxes, 3, 3) 

968 # These three translations must enclose the same volume as the box itself 

969 # Otherwise they skip some periodic images and they do not represent the box 

970 # This is not expected to happen, but in such case we can not tell the shape 

971 box_volumes = np.abs(np.linalg.det(box_matrices)) 

972 represents_box = np.abs(np.abs(np.linalg.det(shortest)) - box_volumes) <= 1e-3 * box_volumes 

973 # Get lengths and folded angles 

974 # angles[:, k] is the angle between the two translations other than k 

975 shortest_lengths = np.linalg.norm(shortest, axis=2) # (n_boxes, 3) 

976 def folded_angle (i, j): 

977 cosine = np.abs(np.sum(shortest[:, i] * shortest[:, j], axis=1)) \ 

978 / (shortest_lengths[:, i] * shortest_lengths[:, j]) 

979 return np.degrees(np.arccos(np.minimum(cosine, 1.0))) 

980 angles = np.stack([folded_angle(1, 2), folded_angle(0, 2), folded_angle(0, 1)], axis=1) 

981 def equal_lengths (i, j): 

982 a, b = shortest_lengths[:, i], shortest_lengths[:, j] 

983 return np.abs(a - b) <= _BOX_SHAPE_LENGTH_TOLERANCE * np.maximum(a, b) 

984 def angle_is (angle, target): 

985 return np.abs(angle - target) <= _BOX_SHAPE_ANGLE_TOLERANCE 

986 all_lengths_equal = equal_lengths(0, 1) & equal_lengths(1, 2) & equal_lengths(0, 2) 

987 # Check every shape 

988 all_right_angles = np.all(angle_is(angles, 90), axis=1) 

989 sorted_angles = np.sort(angles, axis=1) 

990 is_dodecahedral = all_lengths_equal & angle_is(sorted_angles[:, 0], 60) & angle_is(sorted_angles[:, 1], 60) \ 

991 & (angle_is(sorted_angles[:, 2], 60) | angle_is(sorted_angles[:, 2], 90)) 

992 is_octahedral = all_lengths_equal & np.all(angle_is(angles, _OCTAHEDRAL_ANGLE), axis=1) 

993 # Hexagonal: two equal translations at 60° and the third one perpendicular to both 

994 is_hexagonal = np.zeros(n_boxes, dtype=bool) 

995 for k in range(3): 

996 i, j = [index for index in range(3) if index != k] 

997 is_hexagonal |= equal_lengths(i, j) & angle_is(angles[:, k], 60) \ 

998 & angle_is(angles[:, i], 90) & angle_is(angles[:, j], 90) 

999 # The first matching condition wins 

1000 shapes = np.select( 

1001 [~represents_box, all_right_angles & all_lengths_equal, all_right_angles, 

1002 is_dodecahedral, is_octahedral, is_hexagonal], 

1003 ['triclinic', 'cubic', 'orthogonal', 'dodecahedral', 'octahedral', 'hexagonal'], 

1004 default='triclinic') 

1005 return shapes.tolist()

1016 """Read the simulation box from every frame of an XTC file. 

1017 

1018 Skips compressed coordinates entirely (~50× faster than mdtraj/pytraj). 

1019 

1020 Returns (box, box_size, is_constant, is_orthogonal, box_shape) where: 

1021 box — The three box vectors in Ångstroms, as a tuple of three 

1022 (x, y, z) tuples (rows of the box matrix). They fully 

1023 define the periodic cell whatever its shape. 

1024 If the box is dynamic this is the flag 'dyn' instead. 

1025 box_size — (x, y, z) tuple in Ångstroms with the diagonal of the box 

1026 matrix. For orthogonal boxes these are the box lengths, 

1027 but for triclinic boxes they are not the vector lengths. 

1028 For constant boxes this is the per-frame value; for 

1029 variable boxes this is the per-dimension maximum. 

1030 is_constant — True when the box vectors do not change significantly 

1031 across frames (relative std < 0.1 % in all elements). 

1032 is_orthogonal — True when every off-diagonal element of the box matrix 

1033 is ~0 in every frame (rectangular box). 

1034 box_shape — 'cubic', 'orthogonal', 'hexagonal', 'dodecahedral', 

1035 'octahedral' or 'triclinic' (see get_box_shape). 

1036 If a dynamic box changes its shape along the trajectory 

1037 then all shapes are joined in order of appearance 

1038 (e.g. 'dodecahedral / triclinic'). 

1039 All five are None if no valid XTC frame is found or there is no box. 

1040 

1041 Made by Claude. 

1042 """ 

1043 # Box matrix of every frame, as 9 floats in nm (row-major) 

1044 box_matrices: list[tuple[float, ...]] = [] 

1045 magic_bytes = struct.pack('>i', _XTC_MAGIC) 

1046 

1047 with open(xtc_path, 'rb') as f: 

1048 while True: 

1049 if f.read(4) != magic_bytes: 

1050 break 

1051 data = f.read(_XTC_PRE_BOX) 

1052 if len(data) < _XTC_PRE_BOX: 

1053 break 

1054 # Box is a 3×3 float32 matrix (nm); rows are the box vectors 

1055 box_matrices.append(struct.unpack('>9f', data[_XTC_BOX_OFFSET : _XTC_BOX_OFFSET + 36])) 

1056 if not _skip_xtc_coordinates(f): 

1057 break 

1058 

1059 if not box_matrices: 

1060 return None, None, None, None, None 

1061 

1062 matrices = np.array(box_matrices, dtype=np.float64).reshape(-1, 3, 3) * 10.0 # (n_frames, 3, 3) Å 

1063 

1064 # Trajectories without a periodic box store an all-zero box matrix 

1065 if not np.any(matrices): 

1066 if verbose: print(' There is no simulation box') 

1067 return None, None, None, None, None 

1068 

1069 diagonals = np.diagonal(matrices, axis1=1, axis2=2) # (n_frames, 3) Å 

1070 

1071 # Off-diagonal elements (triclinic tilt); 1e-3 Å is well below XTC coordinate precision 

1072 off_diagonal_mask = ~np.eye(3, dtype=bool) 

1073 is_orthogonal = bool(np.all(np.abs(matrices[:, off_diagonal_mask]) < 1e-3)) 

1074 

1075 # Every element varies less than 0.1 % of the mean length of its vector 

1076 if matrices.shape[0] > 1: 

1077 vector_lengths = np.linalg.norm(matrices, axis=2).mean(axis=0) # (3,) Å 

1078 variation = np.std(matrices, axis=0) / vector_lengths[:, np.newaxis] 

1079 is_constant = bool(np.all(variation < 0.001)) 

1080 else: 

1081 is_constant = True 

1082 

1083 if is_constant: 

1084 box = tuple(tuple(float(value) for value in vector) for vector in matrices[0]) 

1085 box_size = tuple(float(value) for value in diagonals[0]) 

1086 box_shape = get_box_shape(matrices[0]) 

1087 else: 

1088 box = 'dyn' 

1089 box_size = tuple(float(value) for value in diagonals.max(axis=0)) 

1090 # Pressure coupling may deform the box along the trajectory (e.g. semi-isotropic) 

1091 # Classify every frame and keep the different shapes in order of appearance 

1092 box_shape = ' / '.join(dict.fromkeys(get_box_shapes(matrices))) 

1093 

1094 # Make a summary of the final box 

1095 if verbose: 

1096 if is_constant: 

1097 print(f' Simulation box size is constant ({box_size[0]:.2f} × {box_size[1]:.2f} × {box_size[2]:.2f} Å)') 

1098 else: 

1099 print(f' Simulation box varies. Maximum size: ({box_size[0]:.2f} × {box_size[1]:.2f} × {box_size[2]:.2f} Å)') 

1100 

1101 print(f' Simulation box shape: {box_shape}') 

1102 

1103 return box, box_size, is_constant, is_orthogonal, box_shape
