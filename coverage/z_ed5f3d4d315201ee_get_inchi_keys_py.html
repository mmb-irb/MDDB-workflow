38def is_ferroheme(mda_atoms: 'MDAnalysis.AtomGroup') -> bool: 

39 """Check if the given MDAnalysis AtomGroup corresponds to a ferroheme molecule.""" 

40 try: 

41 # Create a copy of the universe to remove the bonds safely 

42 resatoms = MDAnalysis.Merge(mda_atoms) 

43 resatoms.delete_bonds(resatoms.select_atoms('element Fe').bonds) 

44 # Convert to basic RDKit molecule (no bond order nor formal charges) 

45 mol = resatoms.select_atoms('not element Fe').convert_to.rdkit(inferrer=None) 

46 # Add hydrogen to first two nitrogens in exchange 

47 # for the removed Fe bonds. Which N to does not matter 

48 # as we standardize later 

49 mol_editable = Chem.RWMol(mol) 

50 n_atoms = [at for at in mol_editable.GetAtoms() if at.GetSymbol() == 'N'] 

51 for n_atom in n_atoms[:2]: 

52 # Add H atom bonding to N 

53 h_idx = mol_editable.AddAtom(Chem.Atom('H')) 

54 mol_editable.AddBond(n_atom.GetIdx(), h_idx, Chem.BondType.SINGLE) 

55 # Convert back to regular molecule 

56 mol_with_h = mol_editable.GetMol() 

57 # rdDepictor.Compute2DCoords(mol_with_h) # Optional: compute 2D coordinates for visualization 

58 # Use MDAnalysisInferrer to get formal charge of the molecule 

59 mol_mda = MDAnalysisInferrer()(mol) 

60 # DetermineBondOrders can crash so we first check if the standardization 

61 # gives us ferroheme directly without it 

62 standar_cid = pubchem_standardization(Chem.MolToInchi(mol_mda)) 

63 if standar_cid and any(cid['pubchem'] in ['4971', '5353910', '25246109'] for cid in standar_cid): 

64 return True 

65 formal_charge = Chem.GetFormalCharge(mol_mda) 

66 DetermineBondOrders(mol_with_h, charge=formal_charge, maxIterations=1000) 

67 unch_mol = rdMolStandardize.ChargeParent(mol_with_h) 

68 inchi = Chem.MolToInchi(unch_mol) 

69 standar_cid = pubchem_standardization(inchi) 

70 return standar_cid[0]['pubchem'] == '4971' # CID for ferroheme without Fe 

71 except Exception: 

72 return False

75def residue_to_inchi(task: tuple['MDAnalysis.AtomGroup', int]) -> tuple[str, str, int]: 

76 """Process a single residue to get its InChI key and related information.""" 

77 resatoms, resindices = task 

78 # Convert to RDKIT and get InChI data 

79 error = None 

80 try: 

81 res_RD = resatoms.convert_to.rdkit(force=True) 

82 except KeyError as e: 

83 tb_str = traceback.format_exc() 

84 # Error that happend with accesoin A01J7 (united atoms) 

85 if 'bond.SetBondType(RDBONDORDER[order])' in tb_str: 

86 error = f'Invalid bond order {e}, failed to convert to RDKit.' 

87 return ('', '', resindices, error) 

88 except Exception: 

89 try: 

90 res_RD = resatoms.convert_to.rdkit(inferrer=None) 

91 except Exception as e: 

92 raise e 

93 if 'Fe' in set(resatoms.atoms.elements) and len(resatoms.atoms) > 1: 

94 # Metallo proteins are not well handled by RDKit/InChI so we hardcode 

95 # the InChI key for ferroheme that is the only case we have found 

96 if is_ferroheme(resatoms): 

97 inchikey = 'KABFMIBPWCXCRK-UHFFFAOYSA-L' 

98 inchi = 'InChI=1S/C34H34N4O4.Fe/c1-7-21-17(3)25-13-26-19(5)23(9-11-33(39)40)31(37-26)16-32-24(10-12-34(41)42)20(6)28(38-32)15-30-22(8-2)18(4)27(36-30)14-29(21)35-25;/h7-8,13-16H,1-2,9-12H2,3-6H3,(H4,35,36,37,38,39,40,41,42);/q;+2/p-2' 

99 else: 

100 error = 'Non-ferroheme residues with Fe are not supported.' 

101 return ('', '', resindices, error) 

102 else: 

103 formal_charge = (int(resatoms.atoms.charges.sum().round()) 

104 if hasattr(resatoms.atoms, 'charges') 

105 else None) 

106 # For charged residues, DetermineBondOrders is better as we 

107 # can set the formal charge on the molecule. Step needed for A026E 

108 # If MDAnalysisInferrer infers the correct formal charge, we skip this step 

109 if formal_charge and Chem.GetFormalCharge(res_RD) != formal_charge: 

110 # Try/except because DetermineBondOrders can fail 

111 try: 

112 res_RD_copy = Chem.Mol(res_RD) 

113 DetermineBondOrders(res_RD_copy, charge=formal_charge, maxIterations=1000) 

114 res_RD = res_RD_copy 

115 except Exception: 

116 pass 

117 # Calculate InChI key and string 

118 inchikey = Chem.MolToInchiKey(res_RD) 

119 # rdinchi.MolToInchi so it doesnt print the warnings 

120 inchi, retcode, message, logs, aux = Chem.rdinchi.MolToInchi(res_RD) 

121 if not inchi: raise InChIFailure(resindices, message) 

122 return (inchikey, inchi, resindices, error)

125def report_inchi_failure(failure: InChIFailure, structure: 'Structure', structure_file: 'File', script_filename: str) -> str: 

126 """Describe the molecule whose InChI could not be found and write a VMD script to see it.""" 

127 residues = [structure.residues[index] for index in failure.resindices] 

128 atoms = [atom for residue in residues for atom in residue.atoms] 

129 # Formula with carbon and hydrogen first and the rest in alphabetical order 

130 element_counts = Counter(atom.element for atom in atoms) 

131 elements = [e for e in ('C', 'H') if e in element_counts] + sorted(e for e in element_counts if e not in ('C', 'H')) 

132 formula = ''.join(f'{e}{element_counts[e]}' for e in elements) 

133 name_counts = Counter(residue.name for residue in residues) 

134 names = ', '.join(f'{name} x{count}' for name, count in name_counts.items()) 

135 labels = ', '.join(residue.label for residue in residues[:MAX_LISTED_RESIDUES]) 

136 if len(residues) > MAX_LISTED_RESIDUES: labels += ', ...' 

137 classes = ', '.join(sorted({ residue.classification for residue in residues })) 

138 # Write a VMD script for the user to see the molecule 

139 molecule_atoms = Selection(sorted(atom.index for atom in atoms)) 

140 commands = [ 

141 '# Molecule whose InChI could not be found', 

142 f'mol new {{{abspath(structure_file.path)}}} waitfor all', 

143 'set main [molinfo top]', 

144 'mol representation Licorice 0.2', 

145 'mol color Name', 

146 f'mol selection {{{molecule_atoms.to_vmd()}}}', 

147 'mol material Opaque', 

148 'mol addrep $main', 

149 *get_focus_commands(molecule_atoms), 

150 ] 

151 with open(script_filename, 'w') as script_file: 

152 script_file.write('\n'.join(commands) + '\n') 

153 return ('Failed to find the InChI of a molecule.\n' 

154 f' InChI says: {failure.message or "no message"}\n' 

155 f' Size: {len(atoms)} atoms in {len(residues)} residues\n' 

156 f' Formula: {formula}\n' 

157 f' Classification: {classes}\n' 

158 f' Residue names: {names}\n' 

159 f' Residues: {labels}\n' 

160 f'To see the molecule run:\n vmd -e {script_filename}')

163def generate_inchikeys( 

164 universe: 'MDAnalysis.Universe', 

165 structure: 'Structure', 

166 structure_file: 'File', 

167 cg_selection: 'Selection', 

168 ua_selection: 'Selection', 

169 parallel: bool = True, 

170) -> dict[str, InChIKeyData]: 

171 """Generate a dictionary mapping InChI keys to residue information for non-standard residues. 

172 

173 This function uses MDAnalysis to parse the input structure and topology files and identifies 

174 residues that are not classified as 'ion', 'solvent', 'nucleic', or 'protein'. For each 

175 identified residue, it converts the structure to RDKit format to obtain the InChI key 

176 and InChI string. The resulting data is stored in dictionaries to map InChI keys to residue 

177 details and residue names to InChI keys. PDB coordinates are necessary to distinguish stereoisomers. 

178 

179 Args: 

180 universe (Universe): The MDAnalysis Universe object containing the structure and topology. 

181 structure (Structure): The Structure object containing residues. 

182 structure_file (File): The structure file, used to see the molecule in VMD if its InChI fails. 

183 cg_selection (Selection): Selection object for coarse-grained regions. 

184 ua_selection (Selection): Selection object for united atom regions, which have no explicit hydrogens. 

185 parallel (bool): Whether to use multiprocessing (default True). Uses more memory, set to False to process residues serially. 

186 

187 Returns: 

188 dict: A dictionary mapping InChI keys to InChIKeyData objects. 

189 

190 Notes: 

191 The function also performs consistency checks, warning if multiple residue names 

192 map to the same InChI key or if multiple InChI keys map to the same residue name, 

193 which can indicate mismatched residue definitions or stereoisomers. 

194 

195 """ 

196 socket_timeout = SocketTimeout(120) 

197 socket_timeout.start() 

198 if not hasattr(universe.atoms, 'charges'): 

199 warn('Topology file does not have charges, InChI keys may be unreliable.') 

200 

201 # 1) Prepare residue data for parallel processing 

202 # First group residues that are bonded together 

203 tasks = [] 

204 residues = structure.residues 

205 

206 # Get coarse grain and united atom indices 

207 # United atoms are skipped since the conversion to RDKit requires all hydrogens to be explicit 

208 coarse_gran_atom_indices = set(cg_selection.atom_indices) | set(ua_selection.atom_indices) 

209 

210 # Fragment = residues that are bonded together 

211 fragments = universe.atoms.fragments 

212 for i, fragment in enumerate(fragments): 

213 

214 # If the fragment contains coarse grain or united atom regions then skip it 

215 fragment_atom_indices = set([int(index) for index in fragment.indices]) 

216 if coarse_gran_atom_indices.intersection(fragment_atom_indices): continue 

217 

218 resindices = fragment.residues.resindices.tolist() 

219 

220 # Continue to the next fragment if any of its 

221 # residues are of a disallowed classification 

222 classes = {residues[resindex].classification for resindex in resindices} 

223 if (classes.intersection({'solvent'}) or 

224 (classes.intersection({'dna', 'rna', 'protein'}) and len(resindices) > 1)): 

225 # print(f'Skipping fragment {i} with classes {classes} and residues {resindices}') 

226 continue 

227 

228 # Select residues atoms with MDAnalysis 

229 resatoms = universe.residues[resindices].atoms 

230 # Skip residues that only contain dummy atoms (Dm) as tey have no element and RDKit breaks 

231 if set(resatoms.types) == {'Dm'}: 

232 continue 

233 # If you pass a residue selection to a parallel worker, you a passing a whole MDAnalysis 

234 # universe, slowing the process down because you have to pickle the object 

235 # To avoid this we create 

236 resatoms = MDAnalysis.Merge(resatoms).universe.atoms 

237 # Convert to RDKit and get InChI data 

238 tasks.append((resatoms, resindices)) 

239 

240 # Remove the VMD script from a previous run, if any 

241 script_filename = dirname(abspath(structure_file.path)) + '/' + INCHI_FAILURE_VIEW_FILENAME 

242 remove_view_script(script_filename) 

243 results = [] 

244 # Every spawned worker imports the whole workflow again (about 200 MB each), so there is 

245 # one per CPU of this process at most. Counting all the CPUs of a cluster node instead of 

246 # the ones of the SLURM job starts dozens of workers, more than the job memory can hold. 

247 # Starting workers is slow, so a single residue or a single CPU is processed directly 

248 workers = min(len(tasks), get_available_cpus()) 

249 try: 

250 if parallel and workers > 1: 

251 # Execute tasks in parallel 

252 # Use 'spawn' context to avoid fork-safety issues with threading (RDKit?) 

253 # https://pythonspeed.com/articles/python-multiprocessing/ 

254 ctx = multiprocessing.get_context('spawn') 

255 with ctx.Pool(processes=workers) as pool: 

256 try: 

257 print(f'Processing {len(tasks)} residues to get InChI keys (parallel mode, {workers} workers)...') 

258 async_results = pool.map_async(residue_to_inchi, tasks) 

259 # Add timeout (e.g., 60 seconds = 1 minute per task) 

260 results = async_results.get(timeout=60 * len(tasks)) 

261 except multiprocessing.TimeoutError: 

262 warn('Pool timeout - some tasks did not complete.') 

263 pool.terminate() # Force kill all workers 

264 pool.join() 

265 raise 

266 except Exception: 

267 pool.terminate() 

268 pool.join() 

269 raise 

270 else: 

271 print(f'Processing {len(tasks)} residues to get InChI keys (sequential mode)...') 

272 for task in tasks: 

273 results.append(residue_to_inchi(task)) 

274 except InChIFailure as failure: 

275 raise RuntimeError(report_inchi_failure(failure, structure, structure_file, script_filename)) from failure 

276 

277 # 2) Process results and build dictionaries 

278 inchikeys: dict[str, InChIKeyData] = {} # To see if different name for same residue 

279 name_2_key = {} # To see if different residues under same name 

280 errors = {} 

281 for (inchikey, inchi, resindices, error) in results: 

282 if error: 

283 errors.setdefault(error, []).extend(resindices) 

284 continue 

285 # Get or create the entry for this InChI key 

286 data = inchikeys.setdefault(inchikey, InChIKeyData(inchi=inchi)) 

287 

288 # Add residue index to the list 

289 data.resindices.extend(resindices) 

290 # Add residue name to the list. For multi residues we join the names 

291 resnames = '-'.join(sorted([residues[index].name for index in resindices])) 

292 data.resnames.add(resnames) 

293 # Add residue class to the list 

294 if len(resindices) > 1: 

295 classes = tuple(set([residues[index].classification for index in resindices])) 

296 data.classification.add(classes) 

297 # Glucolipids saved the groups of residues the form a 'fragment' to solve a 

298 # problem with FATSLiM later (ex: A01IR, A01J5) 

299 data.fragments.append(list(map(int, resindices))) 

300 else: 

301 data.classification.add(residues[resindices[0]].classification) 

302 

303 # Incorrect residue name, stereoisomers, loss of atoms... 

304 name_2_key.setdefault(resnames, []).append(inchikey) 

305 if errors: 

306 for error, resindices in errors.items(): 

307 warn(f'Error processing residues {resindices}: {error}') 

308 # 3) Check data coherence 

309 for inchikey, data in inchikeys.items(): 

310 # Check if there are multiple names for the same InChI key 

311 if len(data.resnames) > 1: 

312 warn(f'Same residue with different names:\n {inchikey} -> {str(data.resnames)}') 

313 data.molname = list(data.resnames)[0] # Just pick one name 

314 # Check if there are multiple classifications for the same InChI key 

315 if len(data.classification) > 1: 

316 warn('Same residue with different classifications:\n' 

317 f'{inchikey} + -> {str(data.classification)} for names {str(data.resnames)}') 

318 # Check if there are multiple fragments length for the same InChI key 

319 if len(data.fragments) == 0: 

320 data.frag_len = 1 

321 else: 

322 data.moltype = 'fragment' 

323 frag_lens = set([len(fragment) for fragment in data.fragments]) 

324 assert len(frag_lens) == 1, \ 

325 f'Fragments of different lengths for InChI key {inchikey}: {str(frag_lens)}' 

326 data.frag_len = frag_lens.pop() 

327 

328 # Check if there are multiple InChI keys for the same name 

329 for name, keys in name_2_key.items(): 

330 keys = list(set(keys)) 

331 if len(keys) < 2: continue 

332 counts = {} 

333 # Count the number of fragments 

334 for key in keys: 

335 # If there are not fragments, we use the number of residues 

336 counts[key] = (inchikeys[key].frag_len 

337 if inchikeys[key].frag_len > 1 

338 else len(inchikeys[key].resindices)) 

339 # Format the counts for printing 

340 key_counts = '\n'.join([f'\t{k}: {c: >4}' for k, c in counts.items()]) 

341 warn(f'The fragment {name} has more than one InChi key:\n' 

342 f'{key_counts}') 

343 socket_timeout.stop() 

344 return inchikeys

367def make_inchikey_reference( 

368 inchikey: str, 

369 inchi: str, 

370 lipid_reference: Optional[dict], 

371 ligand_reference: Optional[dict], 

372) -> dict: 

373 """Make the database reference of an InChI key from its lipid and ligand references. 

374 

375 Lipids have their LIPID MAPS category and whether they were forced by the user, so lipids missing 

376 in the lipid databases (e.g. synthetic ionizable lipids) can be told apart from other molecules. 

377 References of lipids made while the lipid databases were not available get an incomplete version, 

378 so they are not reused from the MDDB database and they are remade later. 

379 """ 

380 lipid_reference = lipid_reference or {} 

381 incomplete = lipid_reference.get('version') == INCOMPLETE_REFERENCE_VERSION 

382 reference = { 

383 'inchikey': inchikey, 

384 'inchi': inchi, 

385 'lipid': {'category': lipid_reference.get('category'), 'forced': lipid_reference.get('forced', False)} 

386 if lipid_reference else None, 

387 'swisslipids': lipid_reference.get('swisslipids'), 

388 'lipidmaps': lipid_reference.get('lipidmaps'), 

389 'ligand': ligand_reference, 

390 'version': INCOMPLETE_REFERENCE_VERSION if incomplete else INCHI_REFERENCE_VERSION 

391 } 

392 # Sort dictionary entries for consistency when uploading to database 

393 return {k: dict(sorted(v.items())) if type(v) is dict else v for k, v in reference.items()}

396def generate_inchi_references( 

397 inchikeys: dict[str, 'InChIKeyData'], 

398 lipid_references: dict[str, dict], 

399 ligand_references: dict[str, dict], 

400 membrane_map: dict, 

401 structure: 'Structure', 

402 output_file: 'File', 

403 mercy: list[str] = [], 

404) -> list[dict]: 

405 """Generate InChI references for the database. 

406 Ligands must be matched to PubChem, so the workflow fails otherwise unless the ligands mercy flag is passed. 

407 Lipids in membranes are not ligands, so they do not need to be in PubChem (e.g. synthetic PEG-lipids). 

408 """ 

409 inchikey_references = [] # For the database 

410 inchikey_map = [] # For the workflow 

411 # Add the user selected ligands to the references, if any 

412 for k, v in ligand_references.items(): 

413 if v.get('forced', False): 

414 inchikeys[k] = InChIKeyData(inchi=v['inchi'], resnames=set(v['resnames']), resindices=v['resindices']) 

415 for inchikey, res_data in inchikeys.items(): 

416 # If there is force ligands, the inchikey may have changed 

417 ref_inchikey = ligand_references.get(inchikey, {}).get('inchikey', inchikey) 

418 ref_inchi = ligand_references.get(inchikey, {}).get('inchi', res_data.inchi) 

419 inchikey_references.append(make_inchikey_reference(ref_inchikey, ref_inchi, 

420 lipid_references.get(inchikey), ligand_references.get(inchikey))) 

421 

422 # Get residue indices from ligand forced selections if available 

423 resindices = ligand_references.get(inchikey, {}).get('resindices', list(map(int, res_data.resindices))) 

424 inchikey_map.append({ 

425 'inchikey': ref_inchikey, # InChI with the possible corrections to find the ligand reference 

426 'name': list(res_data.resnames)[0], # For rmsds workflow analysis 

427 'residue_indices': resindices, 

428 'is_lipid': inchikey in lipid_references, 

429 'is_ligand': False, # Updated later 

430 'generated_inchikey': inchikey, # Original inchikey generated from the structure 

431 }) 

432 

433 # Use inchikeys to 'classify' residues as ligands 

434 no_mem_lipid = Selection(membrane_map['no_mem_lipid']) 

435 for inchikey_data in inchikey_map: 

436 inchikey = inchikey_data['generated_inchikey'] 

437 if inchikey not in ligand_references: 

438 continue 

439 # Skip ions 

440 formula = ligand_references[inchikey].get('formula', 'missing formula') 

441 is_ion = match(ION_FORMULA, formula) 

442 if is_ion: 

443 continue 

444 if inchikey_data['is_lipid']: 

445 ligand_selection = structure.select_residue_indices(inchikey_data['residue_indices']) 

446 # Take only the lipids that are not part of a membrane (fatty-like ligands) 

447 no_mem_ligand_selection = ligand_selection & no_mem_lipid 

448 # If most of the lipid is in a membrane, we do not mark it as a ligand 

449 if len(no_mem_ligand_selection) / len(ligand_selection) < 0.5: 

450 continue 

451 inchikey_data['is_ligand'] = True 

452 

453 # Check that ligands were matched to PubChem 

454 not_matched_ligands = [] 

455 large_aa = [] 

456 for inchikey_data in inchikey_map: 

457 key = inchikey_data['generated_inchikey'] 

458 if not inchikey_data['is_ligand'] or 'pubchem' in ligand_references[key]: continue 

459 resids = inchikeys[key].resindices 

460 molname = inchikeys[key].molname 

461 is_large_aa = any(structure.residues[resid].is_large_aa() for resid in resids) 

462 if is_large_aa: 

463 large_aa.append((molname, resids, key)) 

464 else: 

465 not_matched_ligands.append((molname, resids, key)) 

466 warn(f'Ligand {molname} could not be matched to any PubChem ID{" and has an aminoacid-like structure" if is_large_aa else ""}. Residues: {resids}') 

467 large_aa_fail = large_aa and LARGE_AMINOACID_FLAG not in mercy 

468 not_matched_fail = not_matched_ligands and LIGANDS_MATCH_FLAG not in mercy 

469 if large_aa_fail or not_matched_fail: 

470 fail_msg = 'Some ligands could not be matched to PubChem IDs.\n\tPlease, provide PubChem IDs for all ligands or a PDB code where they are present.' 

471 if large_aa_fail:
