1# Pockets analysis 

2# 

3# The pockets analysis is carried by the 'Fpocket' software. Fpocket is a fast open source protein 

4# pocket (cavity) detection algorithm based on Voronoi tessellation. 

5# 

6# The analysis is performed over 100 frames selected along the trajectory. First of all, an 'MDpocket' 

7# analysis is run over these frames thus generating an output grid file. This grid contains a measure 

8# of frequency of how many times the pocket was open during the trajectory. The most stable and wide 

9# pockets are selected from this grid and then analyzed again with MDpocket individually. This second 

10# run returns dynamic data of the pocket volume and drugability score, among others. 

11# 

12# Vincent Le Guilloux, Peter Schmidtke and Pierre Tuffery, “Fpocket: An open source platform for ligand 

13# pocket detection”, BMC Bioinformatics 2009, 10:168 

14# 

15# Peter Schmidtke & Xavier Barril “Understanding and predicting druggability. A high-throughput method 

16# for detection of drug binding sites.”, J Med Chem, 2010, 53(15):5858-67 

17# 

18# Peter Schmldtke, Axel Bidon-Chanal, Javier Luque, Xavier Barril, “MDpocket: open-source cavity detection 

19# and characterization on molecular dynamics trajectories.”, Bioinformatics. 2011 Dec 1;27(23):3276-85

154 # First, mine the grid dimensions and origin 

155 dimensions = None 

156 dimensions_pattern = "counts ([0-9]+) ([0-9]+) ([0-9]+)" 

157 origin = None 

158 origin_pattern = "origin ([.0-9-]+) ([.0-9-]+) ([.0-9-]+)" 

159 for line in file: 

160 search = re.search(dimensions_pattern, line) 

161 if search != None: 

162 dimensions = (int(search.group(1)),int(search.group(2)),int(search.group(3))) 

163 search = re.search(origin_pattern, line) 

164 if search != None: 

165 origin = (float(search.group(1)),float(search.group(2)),float(search.group(3))) 

166 if origin and dimensions:

169 # This may happend when one of the dimensions is negative, which an mdpocket error

173 raise Exception('Failed to mine dimensions')

177 # Next, mine the grid values 

178 grid_values = [] 

179 grid_values_pattern = "^([.0-9]+) ([.0-9]+) ([.0-9]+) $" 

180 # The last line of the grid may have less than 3 values 

181 last_line_grid_values_pattern_1 = "^([.0-9]+) ([.0-9]+) $" 

182 last_line_grid_values_pattern_2 = "^([.0-9]+) $" 

183 for line in file: 

184 search = re.search(grid_values_pattern, line) 

185 if (search != None): 

186 grid_values.append(float(search.group(1))) 

187 grid_values.append(float(search.group(2))) 

188 grid_values.append(float(search.group(3))) 

189 continue 

190 # This should only happend at the end 

191 search = re.search(last_line_grid_values_pattern_1, line) 

192 if (search != None): 

193 grid_values.append(float(search.group(1))) 

194 grid_values.append(float(search.group(2))) 

195 continue 

196 search = re.search(last_line_grid_values_pattern_2, line) 

197 if (search != None): 

198 grid_values.append(float(search.group(1))) 

199 continue

201 # Make sure we mined as many grid values as expected 

202 # Otherwise some line was not recognized and the grid indices would be shifted 

203 expected_values = dimensions[0] * dimensions[1] * dimensions[2] 

204 if len(grid_values) != expected_values: 

205 raise Exception(f'Failed to mine grid values: expected {expected_values} but found {len(grid_values)}')

207 # Set a function to get the value of a given 'x, y, z' grid point 

208 # Grid points are disposed first in 'z' order, then 'y', and finally 'x' 

209 xl, yl, zl = dimensions 

210 def get_index (x : int, y : int, z : int) -> int: 

211 index = x*yl*zl + y*zl + z 

212 return index

214 # Get the coordinates of the colliding points 

215 # Set the x, y and z index limit (i.e. the limit - 1) 

216 xil, yil, zil = (xl-1, yl-1, zl-1) 

217 def get_colliding_points (point : tuple) -> list: 

218 x, y, z = point[0], point[1], point[2] 

219 colliding_points = [ 

220 (x-1, y-1, z-1), 

221 (x-1, y-1, z ), 

222 (x-1, y-1, z+1), 

223 (x-1, y , z-1), 

224 (x-1, y , z ), 

225 (x-1, y , z+1), 

226 (x-1, y+1, z-1), 

227 (x-1, y+1, z ), 

228 (x-1, y+1, z+1), 

229 (x , y-1, z-1), 

230 (x , y-1, z ), 

231 (x , y-1, z+1), 

232 (x , y , z-1), 

233 (x , y , z+1), 

234 (x , y+1, z-1), 

235 (x , y+1, z ), 

236 (x , y+1, z+1), 

237 (x+1, y-1, z-1), 

238 (x+1, y-1, z ), 

239 (x+1, y-1, z+1), 

240 (x+1, y , z-1), 

241 (x+1, y , z ), 

242 (x+1, y , z+1), 

243 (x+1, y+1, z-1), 

244 (x+1, y+1, z ), 

245 (x+1, y+1, z+1), 

246 ] 

247 if x <= 0: 

248 colliding_points = [ point for point in colliding_points if point[0] != x-1 ] 

249 if x >= xil: 

250 colliding_points = [ point for point in colliding_points if point[0] != x+1 ] 

251 if y <= 0: 

252 colliding_points = [ point for point in colliding_points if point[1] != y-1 ] 

253 if y >= yil: 

254 colliding_points = [ point for point in colliding_points if point[1] != y+1 ] 

255 if z <= 0: 

256 colliding_points = [ point for point in colliding_points if point[2] != z-1 ] 

257 if z >= zil: 

258 colliding_points = [ point for point in colliding_points if point[2] != z+1 ] 

259 return colliding_points

261 # Set a cuttoff value to consider a point valid

264 # Classify all non-zero values by 'pocket' groups according to if they are connected 

265 # Store the grid point indices of each pocket by its pocket number 

266 # Note that a grid point may belong to several pockets 

267 pockets = {} 

268 pockets_count = 0 

269 # Keep track of the points already used to expand any pocket so they do not start a new pocket 

270 # Note that pocket edge points which were tagged but not expanded may still start a new pocket 

271 expanded_points = set()

273 # Given a point which belongs to a pocket, find all pocket points connected to it 

274 # This connection must be wide enought for a water molecule to fit in 

275 # Grid points are 1 Ångstrom wide and a water molecule has a diameter of 2.75 Ångstroms aprox 

276 # So, pocket points must be connected by a net of points at least 3 points wide in all dimensions (x,y,z) 

277 # Tag all points with the same pocket number 

278 def set_pocket (start_point : tuple, pocket_number : int): 

279 # Tag the start point itself, since only colliding points are tagged below 

280 pocket_indices = { get_index(*start_point) } 

281 points = [start_point] 

282 # Keep also a set of the points for fast membership checks 

283 visited_points = { start_point } 

284 for point in points: 

285 # Get current point colliders 

286 colliding_points = get_colliding_points(point) 

287 # Filter only the pocket points 

288 pocket_points = [] 

289 for point in colliding_points: 

290 index = get_index(*point) 

291 value = grid_values[index] 

292 # If it is a pocket then set it as part of the current pocket 

293 if value >= cuttoff: 

294 # Tag it as the current pocket 

295 pocket_indices.add(index) 

296 pocket_points.append(point) 

297 # In case all its surrounding points are pockets add the points to the list so they can keep expanding the pocket 

298 # This is done because if all surrounding points are pockets it means we have a wide enought region of the pocket 

299 # It is a 3 x 3 x 3 region. 

300 # Otherwise stop here since the surrounding points may not we wide enough to expand the pocket 

301 if len(pocket_points) < 26: 

302 continue 

303 # Filter points which are not in the pockets list already 

304 new_points = [ point for point in pocket_points if point not in visited_points ] 

305 # Update the points list 

306 points += new_points 

307 visited_points.update(new_points) 

308 pockets[pocket_number] = pocket_indices 

309 expanded_points.update(visited_points) 

310 

311 # Find out if a point is surrounded by all pocket points so it could be considered a pocket alone 

312 def is_pocket_base (start_point : tuple) -> bool: 

313 # Get current point colliders 

314 colliding_points = get_colliding_points(start_point) 

315 # Points in the grid border have less colliding points and they can not be a pocket base 

316 # Note that they could not expand the pocket either 

317 if len(colliding_points) < 26: 

318 return False 

319 # Search if these points match the cutoff 

320 # If only 1 point does not match then this is not a pocket base 

321 for point in colliding_points: 

322 index = get_index(*point) 

323 value = grid_values[index] 

324 if value < cuttoff: 

325 return False 

326 return True 

327 

328 # Save also each point coordinates 

329 for x in range(xl): 

330 for y in range(yl): 

331 for z in range(zl): 

332 index = get_index(x,y,z) 

333 # If it is not a pocket then pass 

334 value = grid_values[index] 

335 if value < cuttoff: 

336 continue 

337 # If it is a pocket but it has been already expanded by another pocket then pass 

338 point = (x,y,z) 

339 if point in expanded_points: 

340 continue 

341 # If it is not wide enought to start a pocket then pass 

342 if not is_pocket_base(point): 

343 continue 

344 # If none of the previous values was marked as a pocket then we set a new pocket number 

345 pockets_count += 1 

346 set_pocket(start_point=point, pocket_number=pockets_count)
