145def parse_automatic_selection(selection_string: Optional[str]) -> tuple[bool, Optional[str]]: 

146 """Check if a selection string is the automatic flag, which may be followed by an additional selection. 

147 e.g. auto + resname LIG 

148 Return whether the selection is automatic and the additional selection string, if any. 

149 """ 

150 if not isinstance(selection_string, str): return False, None

155 if not additional: 

156 raise InputError(f'Selection "{selection_string}" is missing a selection after the "+"') 

157 return True, additional

208def recursive_transformer(target_object: dict | list | tuple, transformer: Optional[Callable] = None) -> dict | list | tuple: 

209 """Recursive transformer for nested dicts and lists. 

210 Note that a new object is created to prevent mutating the original. 

211 If no transformer function is passed then it becomes a recursive cloner. 

212 WARNING: the transformer function could mutate some types of the original object if not done properly. 

213 """ 

214 object_type = type(target_object) 

215 # Get a starting object and entries iterator depending on the object type 

216 if object_type is dict: 

217 clone = {} 

218 entries = target_object.items() 

219 elif object_type is list or object_type is tuple: 

220 # Note that if it is a tuple we make a list anyway and we convert it to tuple at the end 

221 # WARNING: Note that tuples do not support reassigning items 

222 clone = [None for i in range(len(target_object))] 

223 entries = enumerate(target_object) 

224 else: ValueError(f'The recursive cloner should only be applied to object and lists, not to {object_type}') 

225 # Iterate the different entries in the object 

226 for index_or_key, value in entries: 

227 # Get he value type 

228 value_type = type(value) 

229 # If it is a dict or list then call the transformer recursively 

230 if value_type in OBJECT_TYPES: clone[index_or_key] = recursive_transformer(value, transformer) 

231 # If it is not an object type then apply the transformer to it 

232 else: clone[index_or_key] = transformer(value) if transformer else value 

233 # If it was a tuple then make the conversion now 

234 if object_type == tuple: clone = tuple(clone) 

235 return clone

243def json_serializer(object: dict | list | tuple) -> dict | list | tuple: 

244 """Serialize a standard JSON with support for additional types.""" 

245 def serializer(value): 

246 # If we have exceptions then convert them to text with an appropiate header 

247 if type(value) is Exception: 

248 return f'{EXCEPTION_HEADER}{value}' 

249 if type(value) is NoReferableException: 

250 return f'NoReferableException: {value.sequence}' 

251 # This must be done before the set check because asdict() will create dicts with sets inside 

252 if is_dataclass(value) and not isinstance(value, type): 

253 dict_value = asdict(value) 

254 return recursive_transformer(dict_value, serializer) 

255 if isinstance(value, set): 

256 return list(value) 

257 # If the type is not among the ones we check then assume it is already serializable 

258 return value 

259 object_clone = recursive_transformer(object, serializer) 

260 return object_clone

263def json_deserializer(object: dict | list | tuple) -> dict | list | tuple: 

264 """Deserialize a standard JSON with support for additional types.""" 

265 def deserializer(value): 

266 # Check if there is any value which was adapted to become JSON serialized and restore it 

267 if type(value) is str and value[0:11] == EXCEPTION_HEADER: 

268 # WARNING: Do not declare new exceptions here but use the constant ones 

269 # WARNING: Otherwise further equality comparisions will fail 

270 exception_message = value[11:] 

271 standard_exception = next((exception for exception in STANDARD_EXCEPTIONS if str(exception) == exception_message), None) 

272 if standard_exception is None: 

273 raise ValueError(f'Exception "{exception_message}" is not among standard exceptions') 

274 return standard_exception 

275 if type(value) is str and value[0:22] == 'NoReferableException: ': 

276 sequence = value[22:] 

277 return NoReferableException(sequence) 

278 # If the type is not among the ones we check then assume it is already deserialized 

279 return value 

280 object_clone = recursive_transformer(object, deserializer) 

281 return object_clone

309def check_database_document_size(filepath: str): 

310 """If the file is to be loaded in the database as a document, make sure it does not exceed the size limit. 

311 Size is measured as the BSON encoding of every document, which is what MongoDB actually stores. 

312 """ 

313 filename = os.path.basename(filepath) 

314 is_document = any(re.match(pattern, filename) for pattern in DATABASE_DOCUMENT_PATTERNS) 

315 is_document_list = any(re.match(pattern, filename) for pattern in DATABASE_DOCUMENT_LIST_PATTERNS) 

316 if not is_document and not is_document_list: return 

317 with open(filepath, 'r') as file: 

318 content = json.load(file) 

319 documents = content if is_document_list else [content] 

320 for document in documents: 

321 size = get_bson_size(document) 

322 if size <= DATABASE_DOCUMENT_SIZE_LIMIT: continue 

323 raise RuntimeError(f'Output file {filepath} is to be loaded in the database but a document is too big: ' 

324 f'{size / 1024 ** 2:.2f} Mb (limit is {DATABASE_DOCUMENT_SIZE_LIMIT / 1024 ** 2:.2f} Mb).\n' 

325 ' MongoDB documents can not exceed 16 Mb and, anyway, such a big document would make the service slow.\n' 

326 ' The format of this output should be reconsidered.')

416def ranger(numbers: list[int], unit_separator : str = ',', range_separator : str = '-') -> str: 

417 """Given a list with numbers, create a string where number in a row are represented rangedly. 

418 

419 Example: 

420 [1, 3, 5, 6, 7, 8] => "1,3,5-8" 

421 

422 """ 

423 # Remove duplicates and sort numbers 

424 sorted_numbers = sorted(list(set(numbers))) 

425 # Get the number of numbers in the list 

426 number_count = len(sorted_numbers) 

427 # If there is only one number then finish here 

428 if number_count == 1: 

429 return str(sorted_numbers[0]) 

430 # Start the parsing otherwise 

431 ranged = '' 

432 last_number = -1 

433 # Iterate numbers 

434 for i, number in enumerate(sorted_numbers): 

435 # Skip this number if it was already included in a previous serie 

436 if i <= last_number: continue 

437 # Add current number to the ranged string 

438 ranged += unit_separator + str(number) 

439 # Now iterate numbers after the current number 

440 next_index = i+1 

441 for j, next_number in enumerate(sorted_numbers[next_index:], next_index): 

442 # Set the length of the serie 

443 length = j - i 

444 # End of the serie 

445 if next_number - number != length: 

446 # The length here is the length which broke the serie 

447 # i.e. if the length here is 2 the actual serie length is 1 

448 serie_length = length - 1 

449 if serie_length > 1: 

450 last_serie_number = j - 1 

451 previous_number = sorted_numbers[last_serie_number] 

452 ranged += range_separator + str(previous_number) 

453 last_number = last_serie_number 

454 break 

455 # End of the selection 

456 if j == number_count - 1: 

457 if length > 1: 

458 ranged += range_separator + str(next_number) 

459 last_number = j 

460 # Remove the first coma before returning the ranged string 

461 return ranged[1:]

535def url_to_source_filename(url: str, is_sample: bool = False, verbose: bool = True) -> str: 

536 """Set the filename of an input file downloaded from an input URL. 

537 

538 In this scenario we are free to set our own paths or filenames. 

539 Note that the original name will usually be the very same output filename. 

540 In order to avoid both filenames being the same we will add a header here. 

541 This headers includes also a reference to the database and project. 

542 Thus if there is a new accession then we rely on the filenames to know 

543 if we already have the correct ones or if we must download again. 

544 """ 

545 url_splits = url.split('/') 

546 # Get the name of the source database/service 

547 # This will be at the start of the URL, right after the protocol 

548 source_site = url_splits[2] 

549 source_alias = source_site.split('.')[0].split('-')[0] 

550 # Get the project accession 

551 # Accession in the URL is placed in .../projects/ACCESSION/... 

552 accession = 'unknown-accession' 

553 accession_split_index = url_splits.index('projects') + 1 

554 accession = url_splits[accession_split_index] 

555 # Get the original filename, which will be at the end of the URL 

556 original_filename = url_splits[-1] 

557 # Edit the filename if this is a sample 

558 filename = original_filename 

559 if is_sample: 

560 filename_splits = filename.split('.') 

561 if filename_splits == 1: filename = filename + '_SAMPLE' 

562 else: filename = '.'.join(filename_splits[0:-1]) + '_SAMPLE.' + filename_splits[-1] 

563 # Set the final name 

564 source_filename = f'source_{source_alias}_{accession}_{filename}' 

565 if verbose: print(f'{url} -> {source_filename}') 

566 return source_filename

609 """ 

610 numpy_dtype = SUPPORTED_BINARY_TYPES.get(dtype, None) 

611 if not numpy_dtype: 

612 raise ValueError(f'Not supported data type {dtype}, please select one of these: {", ".join(SUPPORTED_BINARY_TYPES.keys())}') 

613 values = numpy.asarray(data) 

614 # Make sure integer values fit in the requested type, since numpy would silently overflow them 

615 if 'int' in dtype and len(values) > 0: 

616 limits = numpy.iinfo(numpy_dtype) 

617 if values.min() < limits.min or values.max() > limits.max: 

618 raise ValueError(f'Values out of range for {dtype} ({limits.min} - {limits.max}): min {values.min()}, max {values.max()}') 

619 values.astype(numpy_dtype).tofile(filepath) 

620 # Write the companion metadata file, including the data type so readers know how to parse it 

621 companion_filepath = filepath + '.meta.json' 

622 save_json({ **metadata, 'dtype': dtype }, companion_filepath)

625class CaptureOutput (object): 

626 """Capture all stdout or stderr within a code region even if it comes from another non-python threaded process. 

627 

628 https://stackoverflow.com/questions/24277488/in-python-how-to-capture-the-stdout-from-a-c-shared-library-to-a-variable 

629 """ 

630 escape_char = "\b" 

631 def __init__(self, stream: str = 'stdout'): 

632 # Get sys stdout or stderr 

633 if not hasattr(sys, stream): 

634 raise ValueError(f'Unknown stream "{stream}". Expected stream value is "stdout" or "stderr"') 

635 self.original_stream = getattr(sys, stream) 

636 self.original_streamfd = self.original_stream.fileno() 

637 self.captured_text = "" 

638 # Create a pipe so the stream can be captured: 

639 self.pipe_out, self.pipe_in = os.pipe() 

640 def __enter__(self): 

641 self.captured_text = "" 

642 # Save a copy of the stream: 

643 self.streamfd = os.dup(self.original_streamfd) 

644 # Replace the original stream with our write pipe: 

645 os.dup2(self.pipe_in, self.original_streamfd) 

646 return self 

647 def __exit__(self, type, value, traceback): 

648 # Make sure there is at least some output or the 'readOutput' may hang forever 

649 # This has been observed only when running the workflow from Jupyter Notebook 

650 # Print the escape character to make the readOutput method stop: 

651 self.original_stream.write(' ' + self.escape_char) 

652 # Flush the stream to make sure all our data goes in before 

653 # the escape character: 

654 self.original_stream.flush() 

655 self.readOutput() 

656 # Close the pipe: 

657 os.close(self.pipe_in) 

658 os.close(self.pipe_out) 

659 # Restore the original stream: 

660 os.dup2(self.streamfd, self.original_streamfd) 

661 # Close the duplicate stream: 

662 os.close(self.streamfd) 

663 def readOutput(self): 

664 while True: 

665 content = os.read(self.pipe_out, os.O_RDONLY) 

666 char = content.decode(self.original_stream.encoding) 

667 # print(char) 

668 # char = os.read(self.pipe_out, 1).decode(self.original_stream.encoding) 

669 if not char or self.escape_char in char: 

670 break 

671 self.captured_text += char

708def retry_request(func=None, *, retry_not_found: bool = False, 

709 max_retries: int = MAX_RETRIES, retry_delay: float = RETRY_DELAY): 

710 """Decorator to retry HTTP requests with exponential backoff. 

711 

712 Network errors, server errors (5xx) and too many requests errors (429) are retried. 

713 Other HTTP errors (e.g. 404) are raised directly since retrying would give the same result. 

714 Not found errors (404) are also retried with 'retry_not_found', for things which should exist, 

715 since a server may answer 404 for a while (e.g. when it is restarting). 

716 Network and server errors are tried 'max_retries' times in total, waiting 'retry_delay' seconds 

717 after the first failure and twice as long after every next one. 

718 """ 

719 if func is None: return lambda func: retry_request(func, retry_not_found=retry_not_found, 

720 max_retries=max_retries, retry_delay=retry_delay) 

721 @wraps(func) 

722 def wrapper(*args, **kwargs): 

723 attempt = 0 

724 while True: 

725 try: 

726 return func(*args, **kwargs) 

727 # Catch network-related exceptions for retry 

728 except (RateLimitError, 

729 requests.exceptions.RequestException, 

730 http.client.RemoteDisconnected, http.client.IncompleteRead, 

731 ConnectionError, TimeoutError, URLError) as error: 

732 status = _get_http_status(error) 

733 # Errors in our request would fail again, except for too many requests 

734 retried_status = status == 429 or (retry_not_found and status == 404) 

735 if status is not None and status < 500 and not retried_status: 

736 raise 

737 rate_limited = isinstance(error, RateLimitError) or status == 429 

738 max_attempts = RATE_LIMIT_RETRIES if rate_limited else max_retries 

739 attempt += 1 

740 if attempt >= max_attempts: 

741 print(f"Request failed after {attempt} attempts: {error}") 

742 raise 

743 if rate_limited: 

744 delay = _get_error_retry_after(error) or RATE_LIMIT_DELAY * RETRY_BACKOFF ** (attempt - 1) 

745 else: 

746 delay = retry_delay * RETRY_BACKOFF ** (attempt - 1) 

747 print(f"Request failed (attempt {attempt}/{max_attempts}): {error}. Retrying in {delay:.1f}s...") 

748 time.sleep(delay) 

749 return wrapper

753def request_pdb_data(pdb_id: str, query: str) -> dict | None: 

754 """Request data to the PDB GraphQL API. 

755 

756 Note that this function may be used for either PDB ids or PDB molecule ids, depending on the query. 

757 The query parameter may be constructed using the following page: 

758 https://data.rcsb.org/graphql/index.html 

759 """ 

760 # Entries have 4 characters or the extended format (e.g. pdb_00001abc), anything else is a ligand 

761 is_entry = len(pdb_id) == 4 or re.match(EXTENDED_PDB_ID_FORMAT, pdb_id) 

762 data_key = 'entry' if is_entry else 'chem_comp' 

763 # Set the request URL 

764 request_url = 'https://data.rcsb.org/graphql' 

765 # Set the POST data 

766 post_data = { 

767 "query": query, 

768 "variables": {"id": pdb_id} 

769 } 

770 # Send the request 

771 try: 

772 response = requests.post(request_url, json=post_data) 

773 except requests.exceptions.ConnectionError as error: 

774 raise ConnectionError('No internet connection :(') from None 

775 if response.status_code == 429: 

776 raise RateLimitError(f'Too many requests to the PDB API for {pdb_id}', 

777 retry_after=get_retry_after(response.headers)) 

778 if response.status_code != 200: 

779 raise RequestError(f'Something went wrong with the PDB API request for {pdb_id} ' 

780 f'(error {response.status_code}): {response.text}') 

781 # Get the response 

782 parsed_response = json.loads(response.text)['data'][data_key] 

783 if parsed_response is None: 

784 new_pdb_id = request_replaced_pdb(pdb_id) 

785 if new_pdb_id: 

786 warn(f'PDB {pdb_id} is obsolete. {new_pdb_id} will be used instead.') 

787 parsed_response = request_pdb_data(new_pdb_id, query) 

788 else: 

789 warn(f'PDB id {pdb_id} not found') 

790 return parsed_response

878def read_ndict(nested_dict: dict, nested_key: str, placeholder=KeyError('Missing nested key')): 

879 """Read a value in a nested dictionary using a single combined key. 

880 Return the placeholder if any key in the path does not exist. 

881 """ 

882 keys = nested_key.split('.') 

883 value = nested_dict 

884 for key in keys: 

885 # support list indices 

886 if key.isdigit(): 

887 if type(value) is not list: return placeholder 

888 index = int(key) 

889 value = value[index] 

890 # support dict keys 

891 else: 

892 if type(value) is not dict: return placeholder 

893 value = value.get(key, placeholder) 

894 if value == placeholder: return placeholder 

895 return value

898def write_ndict(nested_dict: dict, nested_key: str, value): 

899 """Write a value in a nested dictionary using a single combined key. 

900 Raise an error if any key in the path is missing. 

901 """ 

902 keys = nested_key.split('.') 

903 nested_keys = keys[0:-1] 

904 next_target = nested_dict 

905 for k, key in enumerate(nested_keys): 

906 # support list indices 

907 if key.isdigit(): 

908 if type(next_target) is not list: 

909 raise ValueError(f'{".".join(nested_keys[0:k])} should be a list, but it is {next_target}') 

910 index = int(key) 

911 next_target = next_target[index] 

912 # support dict keys 

913 else: 

914 if type(next_target) is not dict: 

915 raise ValueError(f'{".".join(nested_keys[0:k])} should be a dict, but it is {next_target}') 

916 missing_key_error = KeyError(f'Missing nested key {key}') 

917 next_target = next_target.get(key, missing_key_error) 

918 if next_target == missing_key_error: raise missing_key_error 

919 field = keys[-1] 

920 next_target[field] = value

1014def timeout(seconds=60): 

1015 """Add a timeout to a function using a separate process. 

1016 

1017 Unlike the signal-based approach, this works in: 

1018 - Non-main threads 

1019 - Multiprocessing / cluster environments 

1020 - Any OS (not just Unix) 

1021 """ 

1022 def decorator(func): 

1023 @wraps(func) 

1024 def wrapper(*args, **kwargs): 

1025 result_queue = multiprocessing.Queue() 

1026 process = multiprocessing.Process( 

1027 target=_timeout_worker, 

1028 args=(func, result_queue, args, kwargs), 

1029 ) 

1030 process.start() 

1031 process.join(timeout=seconds) 

1032 

1033 if process.is_alive(): 

1034 process.terminate() 

1035 process.join(timeout=5) 

1036 # Force kill if it didn't terminate gracefully 

1037 if process.is_alive(): 

1038 process.kill() 

1039 process.join() 

1040 raise TimeoutError(f"Function '{func.__name__}' timed out after {seconds} seconds.") 

1041 

1042 if result_queue.empty(): 

1043 raise RuntimeError(f"Function '{func.__name__}' terminated without returning a result.") 

1044 

1045 status, payload = result_queue.get_nowait() 

1046 if status == 'error': 

1047 raise payload 

1048 return payload 

1049 return wrapper 

1050 return decorator
