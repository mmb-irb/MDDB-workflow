700# Versions 

701def _add_versions(version1: str, version2: str) -> int: 

702 """Combine two version strings into a single integer version.""" 

703 # Split the version strings into lists of integers 

704 v1_parts = list(map(int, version1.split('.'))) 

705 v2_parts = list(map(int, version2.split('.'))) 

706 v3_parts = [str(p1+p2) for p1, p2 in zip(v1_parts, v2_parts)] 

707 v3 = '.'.join(v3_parts) 

708 return v3
