15# Words in the force field names which tell us it is a coarse grain force field 

16CG_FORCE_FIELD_KEYWORDS = ['martini', 'sirah', 'elnedyn', 'oxdna', '3spn']

18# Number of heavy atoms in every residue inside a chain 

19# Atomistic force fields have at least these atoms (hydrogens apart) 

20PROTEIN_HEAVY_ATOMS = { 

21 'A': 5, 'R': 11, 'N': 8, 'D': 8, 'C': 6, 'Q': 9, 'E': 9, 'G': 4, 'H': 10, 'I': 8, 

22 'L': 8, 'K': 9, 'M': 8, 'F': 11, 'P': 7, 'S': 6, 'T': 7, 'W': 14, 'Y': 12, 'V': 7, 

23} 

24NUCLEIC_HEAVY_ATOMS = { 'A': 21, 'C': 19, 'G': 22, 'T': 20, 'U': 20 }

26# Atom names which every atomistic residue has 

27PROTEIN_MAIN_ATOM_NAMES = [{'N', 'CA', 'C'}] 

28NUCLEIC_MAIN_ATOM_NAMES = [{"C3'", "C4'"}, {'C3*', 'C4*'}]

30# If a group of residues has less atoms than this ratio of the expected heavy atoms then it is coarse grain 

31CG_ATOM_COUNT_RATIO = 0.6 

32# If atoms in a group of residues have less than this ratio of residues with the main atom names then it is coarse grain 

33CG_ATOM_NAMES_RATIO = 0.5 

34# Atomistic bonds are always shorter than this distance (Å) while coarse grain bonds are longer 

35CG_BOND_LENGTH = 2.2 

36# Number of frames along the trajectory where distances are measured

38FRAMES_LIMIT = 10

51def get_residue_groups(structure: 'Structure') -> dict[str, list['Residue']]: 

52 """Split residues in groups which are expected to be either all coarse grain or all atomistic. 

53 Protein and nucleic residues are grouped by chain. 

54 The rest of residues are grouped by residue name and number of atoms, so for instance all lipids of a kind go together. 

55 """ 

56 groups = {} 

57 for residue in structure.residues: 

58 if residue.name in PROTEIN_RESIDUE_NAME_LETTERS: 

59 group_name = f'protein in chain "{residue.chain.name}"' 

60 elif residue.name in NUCLEIC_RESIDUE_NAME_LETTERS: 

61 group_name = f'nucleic in chain "{residue.chain.name}"' 

62 else: 

63 group_name = f'residue {residue.name} ({residue.atom_count} atoms)' 

64 groups.setdefault(group_name, []).append(residue) 

65 return groups

68def is_polymer(residues: list['Residue']) -> bool: 

69 return residues[0].name in PROTEIN_RESIDUE_NAME_LETTERS or residues[0].name in NUCLEIC_RESIDUE_NAME_LETTERS

72def vote_by_atom_names(residues: list['Residue']) -> Optional[str]: 

73 """Check if residues have the atom names which every atomistic residue has. 

74 This only works for protein and nucleic residues. 

75 """ 

76 if not is_polymer(residues): return None 

77 main_atom_names = PROTEIN_MAIN_ATOM_NAMES if residues[0].name in PROTEIN_RESIDUE_NAME_LETTERS else NUCLEIC_MAIN_ATOM_NAMES 

78 matching_residues = 0 

79 for residue in residues: 

80 atom_names = set(atom.name for atom in residue.atoms) 

81 if any(names <= atom_names for names in main_atom_names): matching_residues += 1 

82 return CG if matching_residues / len(residues) < CG_ATOM_NAMES_RATIO else AA

85def vote_by_atom_count(residues: list['Residue']) -> Optional[str]: 

86 """Compare the number of atoms with the number of heavy atoms we would expect. 

87 This only works for protein and nucleic residues. 

88 """ 

89 if not is_polymer(residues): return None 

90 is_protein = residues[0].name in PROTEIN_RESIDUE_NAME_LETTERS 

91 expected_atom_count = 0 

92 for residue in residues: 

93 if is_protein: expected_atom_count += PROTEIN_HEAVY_ATOMS.get(PROTEIN_RESIDUE_NAME_LETTERS[residue.name], 0) 

94 else: expected_atom_count += NUCLEIC_HEAVY_ATOMS.get(NUCLEIC_RESIDUE_NAME_LETTERS[residue.name], 0) 

95 # Residues with an unusual letter (e.g. X) have no expected number of atoms 

96 if expected_atom_count == 0: return None 

97 atom_count = sum(residue.atom_count for residue in residues) 

98 return CG if atom_count / expected_atom_count < CG_ATOM_COUNT_RATIO else AA

101def get_frames_coordinates(structure: 'Structure', trajectory_file: 'File') -> np.ndarray: 

102 """Get atom coordinates in a few frames along the trajectory.""" 

103 # Write the structure in a mmCIF file to be used as topology for pytraj 

104 auxiliar_mmcif_filepath = get_auxiliar_filepath('.cg_guess.cif') 

105 structure.generate_mmcif_file(auxiliar_mmcif_filepath) 

106 snapshots = count_xtc_frames(trajectory_file) 

107 reduced_trajectory, _, _ = get_reduced_pytraj_trajectory(auxiliar_mmcif_filepath, trajectory_file.path, snapshots, FRAMES_LIMIT) 

108 # Copy every frame, since pytraj may reuse the same coordinates buffer for every frame (e.g. with few frames) 

109 coordinates = np.array([ frame.xyz.copy() for frame in reduced_trajectory.iterframe() ]) 

110 os.remove(auxiliar_mmcif_filepath) 

111 return coordinates

114def vote_by_topology_bonds(residues: list['Residue'], coordinates: np.ndarray, topology_bonds: Optional[list]) -> Optional[str]: 

115 """Measure the length of the topology bonds along frames.""" 

116 if not topology_bonds: return None 

117 atom_indices = set(atom_index for residue in residues for atom_index in residue.atom_indices) 

118 bonds = [ (atom_index, bonded_index) for atom_index in atom_indices 

119 if topology_bonds[atom_index] != MISSING_BONDS 

120 for bonded_index in topology_bonds[atom_index] 

121 if atom_index < bonded_index and bonded_index in atom_indices ] 

122 if not bonds: return None 

123 bonds = np.array(bonds) 

124 lengths = np.linalg.norm(coordinates[:, bonds[:, 0]] - coordinates[:, bonds[:, 1]], axis=2) 

125 return CG if np.median(np.median(lengths, axis=0)) > CG_BOND_LENGTH else AA

128def vote_by_closest_atoms(residues: list['Residue'], coordinates: np.ndarray) -> Optional[str]: 

129 """Measure the distance from every atom to its closest atom, as a guess of its bond length. 

130 Single atom residues are skipped unless they are part of a chain (e.g. one bead per nucleotide). 

131 Otherwise the closest atom of an ion would be another ion far away. 

132 """ 

133 if not is_polymer(residues): residues = [ residue for residue in residues if residue.atom_count > 1 ] 

134 atom_indices = [ atom_index for residue in residues for atom_index in residue.atom_indices ] 

135 if len(atom_indices) < 2: return None 

136 distances = [ cKDTree(frame[atom_indices]).query(frame[atom_indices], k=2)[0][:, 1] for frame in coordinates ] 

137 return CG if np.median(np.median(distances, axis=0)) > CG_BOND_LENGTH else AA

140def guess_cg_selection( 

141 structure: 'Structure', 

142 trajectory_file: 'File', 

143 topology_file: Union['File', Exception], 

144 input_force_fields: Optional[list[str] | str], 

145 verbose: bool = False, 

146) -> 'Selection': 

147 """Guess which atoms are coarse grain beads. 

148 

149 If all input force fields are coarse grain then all atoms are coarse grain. 

150 Otherwise residues are split in groups (see get_residue_groups) and every group is checked in 4 ways: 

151 1. Topology bonds: coarse grain bonds are much longer than atomistic bonds 

152 2. Closest atoms: in atomistic residues every atom has a bonded atom very close 

153 3. Atom count: coarse grain residues have much less atoms than heavy atoms in the real residue 

154 4. Atom names: atomistic residues have standard names (e.g. CA in proteins) 

155 Distances are measured in a few frames along the trajectory and we keep the median. 

156 The most voted option wins. In case of tie, votes are prioritized in the same order. 

157 Atoms which are already flagged with the coarse grain element make their group coarse grain. 

158 """ 

159 if has_only_cg_force_fields(input_force_fields): 

160 if verbose: print(f' All force fields are coarse grain ({input_force_fields}) -> All atoms are CG') 

161 return structure.select_all() 

162 # Get topology bonds, if any 

163 topology_bonds = mine_topology_bonds(topology_file) 

164 if topology_bonds and len(topology_bonds) != structure.atom_count: 

165 raise InputError(f'Topology atom count ({len(topology_bonds)}) does not match the structure atom count ({structure.atom_count})') 

166 coordinates = get_frames_coordinates(structure, trajectory_file) 

167 # Decide every group of residues 

168 groups = get_residue_groups(structure) 

169 decisions = {} 

170 undecided_groups = {} 

171 for group_name, residues in groups.items(): 

172 if any(atom.element == CG_ATOM_ELEMENT for residue in residues for atom in residue.atoms): 

173 decisions[group_name] = CG 

174 if verbose: print(f' {group_name}: atoms are already flagged as CG') 

175 continue 

176 votes = [ 

177 vote_by_topology_bonds(residues, coordinates, topology_bonds), 

178 vote_by_closest_atoms(residues, coordinates), 

179 vote_by_atom_count(residues), 

180 vote_by_atom_names(residues), 

181 ] 

182 valid_votes = [ vote for vote in votes if vote ] 

183 if not valid_votes: 

184 undecided_groups[group_name] = residues 

185 continue 

186 counts = Counter(valid_votes) 

187 decision = valid_votes[0] if counts[CG] == counts[AA] else counts.most_common(1)[0][0] 

188 decisions[group_name] = decision 

189 if verbose: print(f' {group_name}: {decision} (votes: bonds {votes[0]}, closest atoms {votes[1]}, atom count {votes[2]}, atom names {votes[3]})') 

190 # Groups with no votes left are single atom residues (e.g. ions) so their atoms tell us nothing 

191 # If the rest of the system agrees then use the same decision 

192 # In hybrid systems single atom water is coarse grain, since atomistic water has always 3 atoms or more 

193 # The rest (e.g. ions) are considered atomistic 

194 if undecided_groups: 

195 decided_values = set(decisions.values()) 

196 if not decided_values: 

197 raise InputError('Could not guess if residues are coarse grain or not since they all have a single atom.\n' 

198 ' Please provide the coarse grain atom selection (VMD syntax) instead of "auto".\n' 

199 ' Use the "-cg" argument or set the "cg_selection" field in the inputs file.\n' 

200 ' e.g. -cg "resname W ION"') 

201 for group_name, residues in undecided_groups.items(): 

202 if len(decided_values) == 1: 

203 decisions[group_name] = next(iter(decided_values)) 

204 reason = 'like the rest of the system' 

205 elif residues[0].name in STANDARD_SOLVENT_RESIDUE_NAMES: 

206 decisions[group_name] = CG 

207 reason = 'single atom water in a hybrid system' 

208 else: 

209 decisions[group_name] = AA 

210 reason = 'single atom residue in a hybrid system' 

211 if verbose: print(f' {group_name}: {decisions[group_name]} ({reason})') 

212 # Join all coarse grain atoms in a single selection 

213 atom_indices = [ atom_index 

214 for group_name, residues in groups.items() if decisions[group_name] == CG 

215 for residue in residues for atom_index in residue.atom_indices ] 

216 return Selection(sorted(atom_indices))

219# Atoms closer than this distance (Å) are considered bonded when looking for united atoms 

220UA_BOND_LENGTH = 1.9 

221# Hydrogens closer than this distance (Å) to a carbon are considered bonded to it 

222UA_HYDROGEN_BOND_LENGTH = 1.3 

223# If less than this ratio of carbons which should have hydrogens actually have them then it is united atom 

224UA_HYDROGENS_RATIO = 0.5

227def vote_ua_by_hydrogens(atoms: list['Atom'], coordinates: np.ndarray) -> Optional[bool]: 

228 """Check if carbons are missing their hydrogens. 

229 Carbons bonded to 1 or 2 heavy atoms should always have hydrogens in atomistic force fields. 

230 Note that other carbons may have no hydrogens (e.g. C=O carbons) so they are not checked. 

231 """ 

232 elements = [ atom.element or atom.guess_element() for atom in atoms ] 

233 carbon_indices = [ atom.index for atom, element in zip(atoms, elements) if element == 'C' ] 

234 if not carbon_indices: return None 

235 heavy_indices = [ atom.index for atom, element in zip(atoms, elements) if element != 'H' ] 

236 hydrogen_indices = [ atom.index for atom, element in zip(atoms, elements) if element == 'H' ] 

237 # Count the bonded heavy atoms of every carbon along frames (the carbon itself is also counted so we subtract it) 

238 heavy_counts = [ [ len(neighbours) - 1 for neighbours in cKDTree(frame[heavy_indices]).query_ball_point(frame[carbon_indices], r=UA_BOND_LENGTH) ] 

239 for frame in coordinates ] 

240 must_have_hydrogens = np.median(heavy_counts, axis=0) <= 2 

241 if not np.any(must_have_hydrogens): return None 

242 if not hydrogen_indices: return True 

243 hydrogen_distances = [ cKDTree(frame[hydrogen_indices]).query(frame[carbon_indices])[0] for frame in coordinates ] 

244 have_hydrogens = np.median(hydrogen_distances, axis=0) < UA_HYDROGEN_BOND_LENGTH 

245 return np.mean(have_hydrogens[must_have_hydrogens]) < UA_HYDROGENS_RATIO

248def guess_ua_selection( 

249 structure: 'Structure', 

250 trajectory_file: 'File', 

251 cg_selection: 'Selection', 

252 verbose: bool = False, 

253) -> 'Selection': 

254 """Guess which atoms are united atoms: carbons with their hydrogens absorbed. 

255 Residues are split in the same groups used to guess coarse grain (see get_residue_groups). 

256 Groups including coarse grain atoms are skipped. 

257 Note that the whole group is selected, including atoms which are not united atoms (e.g. oxygens). 

258 """ 

259 coordinates = get_frames_coordinates(structure, trajectory_file) 

260 cg_atom_indices = set(cg_selection.atom_indices) 

261 atom_indices = [] 

262 for group_name, residues in get_residue_groups(structure).items(): 

263 atoms = [ atom for residue in residues for atom in residue.atoms ] 

264 if any(atom.index in cg_atom_indices for atom in atoms): continue 

265 is_united_atom = vote_ua_by_hydrogens(atoms, coordinates) 

266 if is_united_atom is None: continue 

267 if verbose: print(f' {group_name}: {"UA" if is_united_atom else "AA"}') 

268 if is_united_atom: atom_indices += [ atom.index for atom in atoms ] 

269 return Selection(sorted(atom_indices))
